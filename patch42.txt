# ============================================================
# Louvo - lot 42 : HISTORY dans le menu 3 traits
#
# Le RGS de Stake n'expose aucune route d'historique (9 routes, aucune
# ne liste les parties). Le History est donc une liste LOCALE de la
# session courante, exactement comme le decrivent vos notes :
#   betID, mise, mode, payout, payoutMultiplier, heure locale
# et relecture locale du round au clic, sans /wallet/play.
#
# A ne pas confondre avec le Replay officiel Stake (lot 40), qui recoit
# un Event ID de simulation math dans l'URL. Le betID stocke ici n'est
# jamais envoye a /bet/replay.
#
# CE QUI EST MODIFIE :
#   - NOUVEAU apps/louvo/src/game/stateHistory.svelte.ts   (le magasin)
#   - NOUVEAU apps/louvo/src/components/LouvoHistoryPanel.svelte
#   - game/actor.ts            : enregistre chaque tour dans onPlayGame
#   - game/typesEmitterEvent.ts: nouvel evenement 'historyShow'
#   - components/LouvoSettingsMenu.svelte : ligne HISTORY + panneau agrandi
#   - components/Game.svelte   : affiche le panneau
#
# Aucun package partage n'est touche.
# L'enregistrement est enferme dans un try/catch : il ne peut pas
# casser un tour meme si le RGS renvoie une forme inattendue.
#
# Rapport ecrit apres chaque ligne. AUCUN BUILD.
#
# Usage :
#   1) deposer ce fichier a la RACINE de web-sdk
#   2) cd /workspaces/web-sdk && python3 patch42.txt
#   3) envoyer rapport48.txt
# ============================================================
import os
import sys
import traceback

ROOT = os.path.dirname(os.path.abspath(__file__))
os.chdir(ROOT)

RAPPORT = "rapport48.txt"
report = []
files = {}


def flush():
    try:
        with open(RAPPORT, "w") as f:
            f.write("\n".join(report) + "\n")
    except OSError:
        pass


def say(line=""):
    report.append(line)
    print(line)
    try:
        sys.stdout.flush()
    except Exception:
        pass
    flush()


def load(p):
    if p not in files:
        with open(p, "r", encoding="utf-8") as f:
            files[p] = f.read()
    return files[p]


def save_all():
    for p, content in files.items():
        with open(p, "w", encoding="utf-8") as f:
            f.write(content)


def rep(p, old, new, tag):
    txt = load(p)
    if new in txt:
        say("  %-46s deja applique" % tag)
        return
    n = txt.count(old)
    if n != 1:
        raise AssertionError("%s : motif trouve %d fois (attendu 1) dans %s" % (tag, n, p))
    files[p] = txt.replace(old, new, 1)
    say("  %-46s OK" % tag)


def ecrire(chemin, contenu, tag):
    if os.path.exists(chemin) and open(chemin, encoding="utf-8").read() == contenu:
        say("  %-46s deja en place" % tag)
        return
    with open(chemin, "w", encoding="utf-8") as f:
        f.write(contenu)
    say("  %-46s ecrit (%d lignes)" % (tag, len(contenu.split("\n"))))


STORE = '''import { stateBet } from 'state-shared';
import { API_AMOUNT_MULTIPLIER } from 'constants-shared/bet';

// ------------------------------------------------------------------
// Historique LOCAL de la session courante.
//
// Le RGS de Stake n'expose aucune route d'historique : les 9 routes du
// SDK sont /wallet/authenticate, /wallet/play, /wallet/end-round,
// /wallet/balance, /bet/event, /bet/action, /bet/replay/..., /game/search
// et /session/start. On garde donc nous-memes les tours joues.
//
// A ne PAS confondre avec le Replay officiel Stake, qui recoit un Event ID
// de simulation math dans l'URL. Le betID stocke ici n'est pas un Event ID
// et n'est jamais envoye a /bet/replay.
// ------------------------------------------------------------------
export type HistoryRound = {
	key: number;
	id: string;
	date: string;
	time: string;
	mode: string;
	betAmount: number;
	winAmount: number;
	multiplier: number;
	bet: unknown;
};

const MAX_ROUNDS = 50;
let counter = 0;

export const stateHistory = $state({ rounds: [] as HistoryRound[] });

const deux = (n: number) => String(n).padStart(2, '0');

export const recordRound = (bet: unknown) => {
	// Enregistrer l'historique ne doit JAMAIS pouvoir casser un tour :
	// tout est enferme dans un try/catch.
	try {
		const raw = bet as {
			betID?: number | string;
			roundID?: number | string;
			amount?: number;
			payout?: number;
			payoutMultiplier?: number;
			mode?: string;
		};

		const id = String(raw?.betID ?? raw?.roundID ?? '');

		// Une reprise de partie rejoue le meme round : on ne l'ajoute pas deux fois.
		if (id && stateHistory.rounds[0]?.id === id) return;

		const betAmount =
			typeof raw?.amount === 'number' && raw.amount > 0
				? raw.amount / API_AMOUNT_MULTIPLIER
				: stateBet.wageredBetAmount;
		const payout = typeof raw?.payout === 'number' ? raw.payout / API_AMOUNT_MULTIPLIER : null;
		const multiplier =
			typeof raw?.payoutMultiplier === 'number'
				? raw.payoutMultiplier
				: payout !== null && betAmount > 0
					? payout / betAmount
					: 0;
		const winAmount = payout !== null ? payout : betAmount * multiplier;

		const now = new Date();
		counter += 1;

		stateHistory.rounds.unshift({
			key: counter,
			id: id || '-',
			date: deux(now.getDate()) + '/' + deux(now.getMonth() + 1),
			time: deux(now.getHours()) + ':' + deux(now.getMinutes()) + ':' + deux(now.getSeconds()),
			mode: raw?.mode || stateBet.activeBetModeKey || 'BASE',
			betAmount,
			winAmount,
			multiplier,
			bet,
		});

		if (stateHistory.rounds.length > MAX_ROUNDS) stateHistory.rounds.length = MAX_ROUNDS;
	} catch (error) {
		console.error('history: enregistrement impossible', error);
	}
};
'''
