#!/usr/bin/env python3
# ============================================================
# Louvo - lot 45 : correction du bandeau REPLAY UNAVAILABLE
#
# LE PROBLEME (prouve par l'onglet Reseau) :
#   GET /bet/replay/<uuid>/6/base/502 renvoie 200 OK + JSON.
#   Le replay se joue donc correctement. Mais LouvoReplayBar lisait
#   stateBet.betToResume, que la machine a etats (resumeGame) remet a
#   null des qu'elle commence a rejouer - parfois avant que le composant
#   ait eu le temps de le lire. Course perdue -> faux message d'erreur.
#
# LA CORRECTION :
#   On capture le round dans actor.ts, sur onResumeGameActive, qui recoit
#   le round BRUT avant que quoi que ce soit ne l'efface. Le composant lit
#   cette capture en priorite et ne retombe sur betToResume qu'en secours.
#
# BONUS : le round rejoue n'est plus enregistre dans le History (c'est une
# relecture, pas une partie jouee).
#
# Rapport ecrit apres chaque ligne. AUCUN BUILD.
#
# Usage :
#   1) deposer ce fichier a la RACINE de web-sdk
#   2) cd /workspaces/web-sdk && python3 patch45.txt
#   3) envoyer rapport51.txt
# ============================================================
import os
import sys
import traceback

ROOT = os.path.dirname(os.path.abspath(__file__))
os.chdir(ROOT)

RAPPORT = "rapport51.txt"
report = []
files = {}


def flush():
    try:
        with open(RAPPORT, "w") as f:
            f.write("\n".join(report) + "\n")
    except OSError:
        pass


def say(line=""):
    report.append(line)
    print(line)
    try:
        sys.stdout.flush()
    except Exception:
        pass
    flush()


def load(p):
    if p not in files:
        with open(p, "r", encoding="utf-8") as f:
            files[p] = f.read()
    return files[p]


def save_all():
    for p, content in files.items():
        with open(p, "w", encoding="utf-8") as f:
            f.write(content)


def rep(p, old, new, tag):
    txt = load(p)
    if new in txt:
        say("  %-46s deja applique" % tag)
        return
    n = txt.count(old)
    if n != 1:
        raise AssertionError("%s : motif trouve %d fois (attendu 1) dans %s" % (tag, n, p))
    files[p] = txt.replace(old, new, 1)
    say("  %-46s OK" % tag)


def main():
    # ---------- 1. Le magasin ----------
    say("===== 1. stateHistory.svelte.ts : capture du round de replay =====")
    s = "apps/louvo/src/game/stateHistory.svelte.ts"
    load(s)

    rep(s,
        "import { stateBet } from 'state-shared';",
        "import { stateBet, stateUi } from 'state-shared';",
        "import stateUi")

    rep(s,
        "export const stateHistory = $state({ rounds: [] as HistoryRound[] });",
        "export const stateHistory = $state({ rounds: [] as HistoryRound[] });\n"
        "\n"
        "// Round du replay officiel Stake. Capture en amont depuis actor.ts\n"
        "// (onResumeGameActive), AVANT que la machine a etats ne remette\n"
        "// stateBet.betToResume a null : sinon la barre de replay lit un\n"
        "// objet deja efface et croit a tort que l'Event ID est invalide.\n"
        "export const stateReplay = $state({\n"
        "\tpayout: null as number | null,\n"
        "\tpayoutMultiplier: null as number | null,\n"
        "\tloaded: false,\n"
        "});\n"
        "\n"
        "export const captureReplayRound = (bet: unknown) => {\n"
        "\ttry {\n"
        "\t\tconst raw = bet as { payout?: number; payoutMultiplier?: number };\n"
        "\t\tif (typeof raw?.payout === 'number') stateReplay.payout = raw.payout;\n"
        "\t\tif (typeof raw?.payoutMultiplier === 'number')\n"
        "\t\t\tstateReplay.payoutMultiplier = raw.payoutMultiplier;\n"
        "\t\tstateReplay.loaded = true;\n"
        "\t} catch (error) {\n"
        "\t\tconsole.error('replay: capture impossible', error);\n"
        "\t}\n"
        "};",
        "stateReplay + captureReplayRound")

    rep(s,
        "\ttry {\n\t\tconst raw = bet as {\n\t\t\tbetID?: number | string;",
        "\ttry {\n"
        "\t\t// Un round rejoue n'est pas une partie jouee : on ne l'ajoute pas.\n"
        "\t\tif (stateUi.config.mode === 'replay') return;\n"
        "\n"
        "\t\tconst raw = bet as {\n\t\t\tbetID?: number | string;",
        "History ignore le mode replay")

    # ---------- 2. Le point de capture ----------
    say("")
    say("===== 2. actor.ts : capture avant effacement =====")
    a = "apps/louvo/src/game/actor.ts"
    load(a)

    rep(a,
        "import { recordRound } from './stateHistory.svelte';",
        "import { recordRound, captureReplayRound } from './stateHistory.svelte';",
        "import captureReplayRound")

    rep(a,
        "\tonResumeGameActive: (betToResume) => convertTorResumableBet(betToResume),",
        "\tonResumeGameActive: (betToResume) => {\n"
        "\t\t// Replay officiel Stake : le round brut passe ICI, avec payout et\n"
        "\t\t// payoutMultiplier, avant que resumeGame ne vide stateBet.betToResume.\n"
        "\t\tcaptureReplayRound(betToResume);\n"
        "\t\treturn convertTorResumableBet(betToResume);\n"
        "\t},",
        "hook onResumeGameActive")

    # ---------- 3. La barre de replay ----------
    say("")
    say("===== 3. LouvoReplayBar.svelte : lecture de la capture =====")
    r = "apps/louvo/src/components/LouvoReplayBar.svelte"
    load(r)

    rep(r,
        "\timport { getContext } from '../game/context';",
        "\timport { getContext } from '../game/context';\n"
        "\timport { stateReplay } from '../game/stateHistory.svelte';",
        "import stateReplay")

    rep(r,
        "\t\t\tif (!seenRound) timedOut = true;",
        "\t\t\tif (!seenRound && !stateReplay.loaded) timedOut = true;",
        "delai tient compte de la capture")

    rep(r,
        "\tconst betText = $derived(numberToCurrencyString(stateBet.betAmount));\n"
        "\tconst winText = $derived(\n"
        "\t\tpayout === null ? '-' : numberToCurrencyString(payout / API_AMOUNT_MULTIPLIER),\n"
        "\t);\n"
        "\tconst multiplierText = $derived.by(() => {\n"
        "\t\tif (payoutMultiplier !== null) return `${payoutMultiplier.toFixed(2)}x`;\n"
        "\t\tif (payout !== null && stateBet.betAmount > 0)\n"
        "\t\t\treturn `${(payout / API_AMOUNT_MULTIPLIER / stateBet.betAmount).toFixed(2)}x`;\n"
        "\t\treturn '-';\n"
        "\t});",
        "\t// La capture faite dans actor.ts prime : elle arrive toujours, alors que\n"
        "\t// betToResume peut avoir ete efface avant qu'on ait pu le lire.\n"
        "\tconst ready = $derived(stateReplay.loaded || seenRound);\n"
        "\tconst effPayout = $derived(stateReplay.payout ?? payout);\n"
        "\tconst effMultiplier = $derived(stateReplay.payoutMultiplier ?? payoutMultiplier);\n"
        "\n"
        "\tconst betText = $derived(numberToCurrencyString(stateBet.betAmount));\n"
        "\tconst winText = $derived(\n"
        "\t\teffPayout === null ? '-' : numberToCurrencyString(effPayout / API_AMOUNT_MULTIPLIER),\n"
        "\t);\n"
        "\tconst multiplierText = $derived.by(() => {\n"
        "\t\tif (effMultiplier !== null) return `${effMultiplier.toFixed(2)}x`;\n"
        "\t\tif (effPayout !== null && stateBet.betAmount > 0)\n"
        "\t\t\treturn `${(effPayout / API_AMOUNT_MULTIPLIER / stateBet.betAmount).toFixed(2)}x`;\n"
        "\t\treturn '-';\n"
        "\t});",
        "valeurs issues de la capture")

    rep(r,
        "\t{#if timedOut}",
        "\t{#if timedOut && !ready}",
        "message d'erreur conditionne")

    save_all()

    say("")
    say("===== 4. VERIFICATION =====")
    controles = (
        ("apps/louvo/src/game/stateHistory.svelte.ts", "captureReplayRound"),
        ("apps/louvo/src/game/stateHistory.svelte.ts", "mode === 'replay'"),
        ("apps/louvo/src/game/actor.ts", "captureReplayRound(betToResume)"),
        ("apps/louvo/src/components/LouvoReplayBar.svelte", "stateReplay.loaded"),
        ("apps/louvo/src/components/LouvoReplayBar.svelte", "effMultiplier"),
        ("apps/louvo/src/components/LouvoReplayBar.svelte", "timedOut && !ready"),
    )
    tout_ok = True
    for chemin, motif in controles:
        present = motif in open(chemin, encoding="utf-8").read()
        if not present:
            tout_ok = False
        say("  %-64s %s" % (os.path.basename(chemin) + " : " + motif,
                            "OK" if present else "*** ABSENT ***"))
    return tout_ok


try:
    ok = main()
except Exception:
    say("")
    say("===== ERREUR PYTHON - AUCUN FICHIER ENREGISTRE =====")
    for l in traceback.format_exc().rstrip().split("\n"):
        say(l)
    ok = False

say("")
say("VERDICT : %s" % ("OK - build, puis testez avec un Event ID entre 0 et 20000"
                     if ok else "ECHEC - envoyez rapport51.txt"))
print("")
print("FINI - envoyez rapport51.txt")
