     2		import { onMount } from 'svelte';
     3		import { Container, Sprite, BitmapText } from 'pixi-svelte';
     4		import MatchDuelClash from './MatchDuelClash.svelte';
     5		import SuperlikeBarilletDisplay from './SuperlikeBarilletDisplay.svelte';
     6		import MatchOutcomeAnimation from './MatchOutcomeAnimation.svelte';
     7		import { SYMBOL_WIDTH, SYMBOL_HEIGHT, BOARD_DIMENSIONS } from '../game/constants';
     8	
     9		type Props = {
    10			assetKey: string;
    11			multiplierText?: string;
    12			duelValues?: [number, number];
    13			duelWinner?: number;
    14			likes?: number;
    15			closeToken?: number;
    16			x?: number;
    17			y?: number;
    18			durationInMs?: number;
    19			holdMs?: number;
    20			zIndex?: number;
    21			oncomplete?: () => void;
    22		};
    23	
    24		const props: Props = $props();
    25		const durationInMs = props.durationInMs ?? 350;
    26		const holdMs = props.holdMs ?? 1400;
    27	
    28		// Taille d'un rouleau complet : 1 colonne de large, toute la hauteur du plateau
    29		const bannerWidth = SYMBOL_WIDTH;
    30		const bannerHeight = SYMBOL_HEIGHT * BOARD_DIMENSIONS.y;
    31	
    32		let duelResolved = $state(false);
    33		const biggerWon = $derived(
    34			props.duelValues && props.duelWinner !== undefined
    35				? props.duelWinner === Math.max(props.duelValues[0], props.duelValues[1])
    36				: true,
    37		);
    38	
    39		let scale = $state(0.6);
    40		let alpha = $state(0);
    41	
    42		const easeOutBack = (t: number) => {
    43			const c1 = 1.70158;
    44			const c3 = c1 + 1;
    45			return 1 + c3 * Math.pow(t - 1, 3) + c1 * Math.pow(t - 1, 2);
    46		};
    47	
    48		const animateTo = (targetScale: number, targetAlpha: number, duration: number) =>
    49			new Promise<void>((resolve) => {
    50				const startScale = scale;
    51				const startAlpha = alpha;
    52				const start = performance.now();
    53	
    54				const step = (now: number) => {
    55					const elapsed = now - start;
    56					const t = Math.min(elapsed / duration, 1);
    57					const eased = easeOutBack(t);
    58	
    59					scale = startScale + (targetScale - startScale) * eased;
    60					alpha = startAlpha + (targetAlpha - startAlpha) * Math.min(elapsed / duration, 1);
    61	
    62					if (t < 1) {
    63						requestAnimationFrame(step);
    64					} else {
    65						scale = targetScale;
    66						alpha = targetAlpha;
    67						resolve();
    68					}
    69				};
    70	
    71				requestAnimationFrame(step);
    72			});
    73	
    74		let resolveHold = () => {};
    75	
    76		onMount(() => {
    77			(async () => {
    78				await animateTo(1, 1, durationInMs);
    79	
    80				if (holdMs > 0) {
    81					if (holdMs === Infinity) {
    82						// Pas de minuteur du tout : attend uniquement forceClose
    83						// (prochain spin), jamais de fermeture automatique.
    84						await new Promise<void>((r) => (resolveHold = r));
    85					} else {
    86						await Promise.race([
    87							new Promise<void>((r) => setTimeout(r, holdMs)),
    88							new Promise<void>((r) => (resolveHold = r)),
    89						]);
    90					}
    91					await animateTo(1, 0, Math.min(durationInMs, 250));
    92				}
    93	
    94				props.oncomplete?.();
    95			})();
    96		});
    97	
    98		// Valeur de reference capturee au montage : tout changement ulterieur
    99		// de closeToken (peu importe l'ancienne valeur booleenne ratee en
   100		// turbo/super turbo) declenche la fermeture, sans jamais pouvoir etre
   101		// rate meme si plusieurs spins s'enchainent tres vite.
   102		const initialCloseToken = props.closeToken;
   103		$effect(() => {
   104			if (props.closeToken !== undefined && props.closeToken !== initialCloseToken) resolveHold();
   105		});
   106	</script>
   107	
   108	<Container x={props.x} y={props.y} alpha={alpha} scale={scale} zIndex={props.zIndex}>
   109		{#if props.assetKey === 'matchReveal' && duelResolved}
   110			<MatchOutcomeAnimation {biggerWon} width={bannerWidth} height={bannerHeight} />
   111		{:else}
   112			<Sprite anchor={0.5} key={props.assetKey} width={bannerWidth} height={bannerHeight} />
   113		{/if}
   114	
   115		{#if props.assetKey === 'matchReveal' && props.duelValues && props.duelWinner !== undefined}
   116			<MatchDuelClash
   117				duelValues={props.duelValues}
   118				winner={props.duelWinner}
   119				oncomplete={() => (duelResolved = true)}
   120			/>
   121		{:else if props.multiplierText}
   122			<BitmapText
   123				anchor={0.5}
   124				y={0}
   125				text={props.multiplierText}
   126				style={{
   127					fontFamily: 'gold', fill: 0xff2d6a,
   128					fontSize: 60,
   129				}}
   130			/>
   131		{/if}
   132	
   133		{#if props.assetKey === 'superlikeReveal' && props.likes}
   134			<SuperlikeBarilletDisplay likes={props.likes} />
   135		{/if}
   136	</Container>
