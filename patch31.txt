#!/usr/bin/env python3
# ============================================================
# Louvo - lot 31 : le tour suivant attend la FIN REELLE de la
# presentation des lignes de gains.
#
# L'attente etait fixe (1520 ms = UNE ligne) alors que les lignes
# s'enchainent avec un decalage : avec beaucoup de lignes elle se
# terminait avant la derniere. En vitesse normale la surbrillance
# des symboles absorbait le retard ; en turbo/super turbo elle est
# quasi instantanee -> le tour suivant partait par-dessus
# l'animation de gain inachevee (constate en MATCH).
#
# Usage :
#   1) deposer ce fichier a la RACINE de web-sdk
#   2) cd /workspaces/web-sdk && python3 patch31.txt
#   3) envoyer rapport34.txt
# Build coupe automatiquement (5 min max), pas de Ctrl+C.
# ============================================================
import os
import subprocess
import sys

T = "\t"
ROOT = os.path.dirname(os.path.abspath(__file__))
os.chdir(ROOT)

BOOK = "apps/louvo/src/game/bookEventHandlerMap.ts"

files = {}
journal = []


def load(p):
    if p not in files:
        with open(p) as f:
            files[p] = f.read()


def rep(p, old, new, tag):
    load(p)
    if new and new not in old:
        applied = new in files[p]
    else:
        applied = old not in files[p]
    if applied:
        journal.append("%s : deja applique, saute" % tag)
        return
    n = files[p].count(old)
    if n != 1:
        raise AssertionError("%s ECHEC : %d occurrence(s) du texte attendu dans %s" % (tag, n, p))
    files[p] = files[p].replace(old, new, 1)
    journal.append("%s : OK" % tag)


def run_patches():
    rep(BOOK,
        "\n".join([
            T*3 + "// Duree totale de l'animation d'une ligne (voir WinLineReveal.svelte) :",
            T*3 + "// 120 (apparition) + 700 (maintien) + 150 (ligne disparait) + 300 (attente) + 250 (montant disparait)",
            T*3 + "await new Promise((r) => setTimeout(r, 120 + 700 + 150 + 300 + 250));",
        ]),
        "\n".join([
            T*3 + "// Duree d'UNE ligne (voir WinLineReveal.svelte) : 120 (apparition)",
            T*3 + "// + 700 (maintien) + 150 (ligne disparait) + 300 (attente) + 250 (montant",
            T*3 + "// disparait). Les lignes s'enchainent avec WIN_LINE_STAGGER d'ecart : la",
            T*3 + "// presentation complete dure donc plus longtemps des qu'il y en a",
            T*3 + "// plusieurs. L'attente etait FIXE (une seule ligne) : en turbo/super turbo,",
            T*3 + "// ou la surbrillance des symboles est quasi instantanee et n'absorbe plus",
            T*3 + "// le retard, le tour suivant partait par-dessus l'animation inachevee.",
            T*3 + "const WIN_LINE_DURATION = 120 + 700 + 150 + 300 + 250;",
            T*3 + "const WIN_LINE_STAGGER = 200;",
            T*3 + "await new Promise((r) =>",
            T*4 + "setTimeout(",
            T*5 + "r,",
            T*5 + "WIN_LINE_DURATION + WIN_LINE_STAGGER * Math.max(0, bookEvent.wins.length - 1),",
            T*4 + "),",
            T*3 + ");",
        ]), "X1")


def main():
    report = []
    try:
        run_patches()
        for p, c in files.items():
            with open(p, "w") as f:
                f.write(c)
        report.append("===== JOURNAL DES PATCHS =====")
        report.extend(journal)
        patch_failed = False
    except AssertionError as e:
        report.append("===== JOURNAL DES PATCHS (ECHEC, RIEN N'A ETE ECRIT) =====")
        report.extend(journal)
        report.append(str(e))
        patch_failed = True

    report.append("")
    report.append("===== PREUVES =====")
    all_ok = not patch_failed
    with open(BOOK) as f:
        content = f.read()
    for needle, expected in [("WIN_LINE_STAGGER", 2), ("WIN_LINE_DURATION", 2)]:
        n = content.count(needle)
        report.append("%-22s : %d (attendu %d)" % (needle, n, expected))
        if n < expected:
            all_ok = False
    report.append("----- restes (attendu 0) -----")
    n = content.count("setTimeout(r, 120 + 700 + 150 + 300 + 250)")
    report.append("%-22s : %d" % ("ancienne attente fixe", n))
    if n != 0:
        all_ok = False

    # ---- Verification des vraies durees (pour ajuster si besoin) ----
    report.append("")
    report.append("===== VRAIES DUREES DE LA PRESENTATION DES LIGNES =====")
    for f_ in [
        "apps/louvo/src/components/WinLinesDisplay.svelte",
        "apps/louvo/src/components/WinLineReveal.svelte",
        "apps/louvo/src/components/Win.svelte",
    ]:
        report.append("### " + f_)
        if os.path.exists(f_):
            with open(f_) as fh:
                report.extend("%4d\t%s" % (i + 1, l.rstrip("\n")) for i, l in enumerate(fh.readlines()))
        else:
            report.append("INTROUVABLE")
        report.append("")

    report.append("===== git status --short =====")
    try:
        out = subprocess.run(["git", "status", "--short"], capture_output=True, text=True).stdout
        report.append(out.rstrip() or "(arbre propre)")
    except OSError as e:
        report.append("git indisponible : %s" % e)

    with open("rapport34.txt", "w") as f:
        f.write("\n".join(report) + "\n")

    report.append("")
    report.append("===== BUILD =====")
    if all_ok:
        timed_out = False
        with open("build_log.txt", "w") as logf:
            try:
                code = subprocess.run(["pnpm", "run", "build", "--filter=louvo"],
                                      stdout=logf, stderr=subprocess.STDOUT, timeout=300).returncode
            except subprocess.TimeoutExpired:
                timed_out = True
                code = -1
        with open("build_log.txt") as f:
            log = f.read()
        report.extend(log.rstrip().split("\n")[-6:])
        site_ecrit = ("Wrote site to" in log) or ("done" in log)
        if timed_out:
            report.append("(build coupe apres 5 min - comportement pnpm connu)")
        report.append("site ecrit : %s / exit : %s" % ("OUI" if site_ecrit else "NON", code))
        if not site_ecrit:
            all_ok = False
    else:
        report.append("BUILD SAUTE (patchs incomplets)")

    report.append("")
    report.append("VERDICT : %s" % ("TOUT OK - commit, deploiement, tests" if all_ok else "ECHEC - envoyez ce rapport a Claude"))

    with open("rapport34.txt", "w") as f:
        f.write("\n".join(report) + "\n")
    print("\n".join(report[-4:]))
    print("FINI - envoyez rapport34.txt")
    sys.exit(0 if all_ok else 1)


main()
