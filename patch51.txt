#   board_background.png              (coucher de soleil, SANS grille)
#   board_background_after_dark.png   (scene de nuit, SANS grille)
#   board_frame_overlay.png           (cadre LOUVO decoupe)
#   board_frame_overlay_after_dark.png(cadre AFTER DARK decoupe)
#
# CE QUI CHANGE :
#  - le FOND s'etire sur tout le canvas. Il ne contient plus de grille,
#    donc plus rien a aligner, plus de bande rose a rajouter, plus de
#    jointure. C'est ce que vous vouliez.
#  - le CADRE garde son ratio et sa boite (lot 49).
#  - les REPERES DE GRILLE sont mesures sur vos nouvelles images, en
#    detectant les lignes de separation dorees, et ils sont DIFFERENTS
#    pour chaque palier :
#         base       x 391->1275   y 163->857   cellule 177x139
#         after dark x 463->1197   y 130->671   cellule 147x108
#    (ancien decor : x 491->1166, y 130->657)
#    Le nouveau cadre de base est plus grand : LES SYMBOLES GROSSISSENT,
#    y compris sur bureau. C'est attendu, c'est votre nouvelle maquette.
#  - le plateau recoit deux echelles, X et Y, calculees sur la cellule du
#    cadre du palier courant. Les cellules d'After Dark n'ont pas la meme
#    forme que celles de base (1.36 contre 1.27) : c'est exactement ce que
#    compensaient vos decalages manuels par colonne, qui deviennent donc
#    inutiles. Je les mets a zero SANS les supprimer.
#
# SAUVEGARDES : chaque fichier modifie est copie en .avant-lot51
#
# Rapport ecrit apres chaque ligne. AUCUN BUILD.
#
# Usage :
#   1) deposer ce fichier a la RACINE de web-sdk
#   2) cd /workspaces/web-sdk && python3 patch51.txt
#   3) envoyer rapport58.txt

import os
import shutil
import sys
import traceback

ROOT = os.path.dirname(os.path.abspath(__file__))
os.chdir(ROOT)

RAPPORT = "rapport58.txt"
report = []
files = {}


def flush():
    try:
        with open(RAPPORT, "w") as f:
            f.write("\n".join(report) + "\n")
    except OSError:
        pass


def say(line=""):
    report.append(line)
    print(line)
    try:
        sys.stdout.flush()
    except Exception:
        pass
    flush()


def sauver(p):
    s = p + ".avant-lot51"
    if not os.path.exists(s):
        shutil.copyfile(p, s)


def load(p):
    if p not in files:
        sauver(p)
        with open(p, "r", encoding="utf-8") as f:
            files[p] = f.read()
    return files[p]


def rep(p, old, new, tag):
    txt = load(p)
    if new in txt:
        say("  %-46s deja applique" % tag)
        return
    n = txt.count(old)
    if n != 1:
        raise AssertionError("%s : motif trouve %d fois (attendu 1) dans %s" % (tag, n, p))
    files[p] = txt.replace(old, new, 1)
    say("  %-46s OK" % tag)


def ecrire(p, contenu, tag):
    if os.path.exists(p) and open(p, encoding="utf-8").read() == contenu:
        say("  %-46s deja en place" % tag)
        return
    sauver(p)
    with open(p, "w", encoding="utf-8") as f:
        f.write(contenu)
    say("  %-46s reecrit (%d lignes)" % (tag, len(contenu.split("\n"))))


BOARD_BOX = '''import { GRID_FRACS, SYMBOL_WIDTH, SYMBOL_HEIGHT } from './constants';
import { stateLayoutDerived } from './stateLayout';

// ------------------------------------------------------------------
// LA BOITE DE CADRE
//
// Depuis que le decor et le cadre sont deux images separees, cette boite
// est la surface ou l'on dessine LE CADRE seul, toujours au ratio de son
// image (1672x941). Le fond, lui, s'etire librement sur tout l'ecran : il
// ne contient plus de grille, donc il n'y a plus rien a aligner.
//
// La grille et les rangees se referent a cette boite. Chaque palier a son
// propre cadre, donc ses propres reperes : on passe le palier en argument
// plutot que de lire stateGame ici, ce qui eviterait un import circulaire.
// ------------------------------------------------------------------

const ART_RATIO = 1672 / 941;
const REFERENCE_WIDTH = 1422;

export type BoardTier = 'basegame' | 'speed_dating' | 'after_dark';

export const gridFracs = (tier: BoardTier) =>
	tier === 'after_dark' ? GRID_FRACS.afterDark : GRID_FRACS.base;

// ==================================================================
// REGLAGES PORTRAIT - c'est ici qu'on ajuste a l'oeil
// ==================================================================
// Part de la largeur d'ecran occupee par la grille.
const PORTRAIT_GRID_WIDTH_FRAC = 0.86;

// Hauteur a laquelle commence le cadre, en fraction de la hauteur d'ecran.
const PORTRAIT_BOX_TOP_FRAC = 0.02;

// Decalage horizontal fin, en fraction de la largeur d'ecran.
const PORTRAIT_GRID_X_OFFSET_FRAC = 0;
// ==================================================================

export type BoardBox = {
	x: number;
	y: number;
	width: number;
	height: number;
	scaleX: number;
	scaleY: number;
};

const avecEchelles = (
	x: number,
	y: number,
	width: number,
	height: number,
	tier: BoardTier,
): BoardBox => {
	const f = gridFracs(tier);
	// Une cellule du cadre, mesuree sur l'image, ramenee a la taille dessinee.
	const cellWidth = ((f.right - f.left) / 5) * width;
	const cellHeight = ((f.bottom - f.top) / 5) * height;
	return {
		x,
		y,
		width,
		height,
		// Les deux echelles different legerement en After Dark, dont les
		// cellules ne sont pas de la meme forme qu'en base (1.36 contre 1.27).
		// C'est ce qui rend inutiles les decalages manuels par colonne.
		scaleX: cellWidth / SYMBOL_WIDTH,
		scaleY: cellHeight / SYMBOL_HEIGHT,
	};
};

export const boardBox = (tier: BoardTier): BoardBox => {
	const main = stateLayoutDerived.mainLayout();
	const layoutType = stateLayoutDerived.layoutType();

	// Paysage : la boite EST la mise en page.
	if (layoutType === 'desktop' || layoutType === 'landscape') {
		return avecEchelles(0, 0, main.width, main.height, tier);
	}

	// Portrait / tablette : le cadre est dessine PLUS LARGE que l'ecran et
	// deborde sur les cotes, pour que la grille remplisse la largeur.
	const f = gridFracs(tier);
	const width = (PORTRAIT_GRID_WIDTH_FRAC * main.width) / (f.right - f.left);
	const height = width / ART_RATIO;
	const centreX = (f.left + f.right) / 2;

	return avecEchelles(
		// On centre la GRILLE sur l'ecran, pas la boite.
		main.width * (0.5 + PORTRAIT_GRID_X_OFFSET_FRAC) - width * centreX,
		PORTRAIT_BOX_TOP_FRAC * main.height,
		width,
		height,
		tier,
	);
};

// Echelle de reference, conservee pour d'eventuels usages externes.
export const boardReferenceScale = (width: number) => width / REFERENCE_WIDTH;
'''

BACKGROUND = '''<script lang="ts">
	import { Rectangle, Sprite } from 'pixi-svelte';
	import { FadeContainer } from 'components-pixi';
	import { SECOND } from 'constants-shared/time';
	import { getContext } from '../game/context';

	const context = getContext();

	const showBaseBackground = $derived(
		context.stateGame.gameType === 'basegame' || context.stateGame.tier === 'speed_dating',
	);
	const showFeatureBackground = $derived(context.stateGame.tier === 'after_dark');

	// ------------------------------------------------------------------
	// Le fond ne contient plus la grille ni l'enseigne : il n'y a donc plus
	// rien a aligner avec le cadre. Il s'etire simplement sur tout le canvas,
	// quelle que soit la forme de l'ecran. Plus de bande vide a combler,
	// plus de jointure a maquiller.
	// ------------------------------------------------------------------
	const canvas = $derived(context.stateLayoutDerived.canvasSizes());
	const etire = $derived({
		x: canvas.width * 0.5,
		y: canvas.height * 0.5,
		width: canvas.width,
		height: canvas.height,
	});
</script>

<Rectangle {...canvas} backgroundColor={0x000000} zIndex={-4} />

<FadeContainer show={showBaseBackground} duration={SECOND} zIndex={-3}>
	<Sprite key="boardBackground" {...etire} anchor={0.5} />
</FadeContainer>

<FadeContainer show={showFeatureBackground} duration={SECOND} zIndex={-2}>
	<Sprite key="boardBackgroundAfterDark" {...etire} anchor={0.5} />
</FadeContainer>
'''


ANCIEN_GRID = """const GRID_SOURCE_WIDTH = 1672;
const GRID_SOURCE_HEIGHT = 941;
export const GRID_LEFT_FRAC = 491 / GRID_SOURCE_WIDTH;
export const GRID_RIGHT_FRAC = 1166 / GRID_SOURCE_WIDTH;
export const GRID_TOP_FRAC = 130 / GRID_SOURCE_HEIGHT;
export const GRID_BOTTOM_FRAC = 657 / GRID_SOURCE_HEIGHT;"""

NOUVEAU_GRID = """const GRID_SOURCE_WIDTH = 1672;
const GRID_SOURCE_HEIGHT = 941;

// Mesures dans les IMAGES DE CADRE en detectant les lignes de separation
// dorees. Chaque palier a son propre cadre, donc ses propres reperes.
export const GRID_FRACS = {
\tbase: {
\t\tleft: 391 / GRID_SOURCE_WIDTH,
\t\tright: 1275 / GRID_SOURCE_WIDTH,
\t\ttop: 163 / GRID_SOURCE_HEIGHT,
\t\tbottom: 857 / GRID_SOURCE_HEIGHT,
\t},
\tafterDark: {
\t\tleft: 463 / GRID_SOURCE_WIDTH,
\t\tright: 1197 / GRID_SOURCE_WIDTH,
\t\ttop: 130 / GRID_SOURCE_HEIGHT,
\t\tbottom: 671 / GRID_SOURCE_HEIGHT,
\t},
} as const;

const centresDeRangee = (haut: number, bas: number) =>
\t[0, 1, 2, 3, 4].map((i) => (haut + ((bas - haut) / 5) * (i + 0.5)) / GRID_SOURCE_HEIGHT);

export const ROW_CENTERS_FRACS = {
\tbase: centresDeRangee(163, 857),
\tafterDark: centresDeRangee(130, 671),
};

// Conserves pour les composants qui les importent encore (presentoirs
// After Dark) : ce sont les valeurs du cadre de base.
export const GRID_LEFT_FRAC = GRID_FRACS.base.left;
export const GRID_RIGHT_FRAC = GRID_FRACS.base.right;
export const GRID_TOP_FRAC = GRID_FRACS.base.top;
export const GRID_BOTTOM_FRAC = GRID_FRACS.base.bottom;"""


def main():
    say("===== 1. constants.ts : reperes mesures sur vos cadres =====")
    c = "apps/louvo/src/game/constants.ts"
    load(c)
    rep(c, ANCIEN_GRID, NOUVEAU_GRID, "GRID_FRACS par palier")

    say("")
    say("===== 2. boardBox.ts : boite par palier, deux echelles =====")
    ecrire("apps/louvo/src/game/boardBox.ts", BOARD_BOX, "boardBox.ts")

    say("")
    say("===== 3. Background.svelte : fond etire sur tout l'ecran =====")
    ecrire("apps/louvo/src/components/Background.svelte", BACKGROUND, "Background.svelte")

    say("")
    say("===== 4. BoardFrame.svelte : cadre du palier courant =====")
    b = "apps/louvo/src/components/BoardFrame.svelte"
    txt = load(b)
    if "boardBox(context.stateGame.tier)" in txt:
        say("  %-46s deja applique" % "cadres par palier")
    else:
        # Remplacement global. On verifie d'abord que le fichier a bien la
        # forme attendue : les deux cadres, et au moins un appel par attribut.
        n = txt.count("boardBox()")
        for cle in ("boardFrameOverlay", "boardFrameOverlayAfterDark"):
            if cle not in txt:
                raise AssertionError("BoardFrame ne contient pas %s - rien modifie" % cle)
        if n < 8:
            raise AssertionError(
                "BoardFrame ne contient que %d 'boardBox()' - rien modifie" % n)
        nouveau = txt.replace("boardBox()", "boardBox(context.stateGame.tier)")
        if "boardBox()" in nouveau:
            raise AssertionError("il reste des boardBox() sans palier - rien modifie")
        files[b] = nouveau
        say("  %-46s OK (%d occurrences)" % ("cadres par palier", n))

    say("")
    say("===== 5. stateGame.svelte.ts : grille par palier =====")
    s = "apps/louvo/src/game/stateGame.svelte.ts"
    load(s)
    rep(s,
        "import { GRID_LEFT_FRAC, GRID_RIGHT_FRAC, GRID_TOP_FRAC, GRID_BOTTOM_FRAC } from './constants';",
        "import { GRID_FRACS } from './constants';",
        "import GRID_FRACS")
    rep(s,
        "import { boardBox } from './boardBox';",
        "import { boardBox, gridFracs } from './boardBox';",
        "import gridFracs")
    rep(s, "const AFTER_DARK_Y_ADJUST_FRAC = 0.005;",
        "const AFTER_DARK_Y_ADJUST_FRAC = 0;", "ajustement vertical remis a zero")
    rep(s,
        "\tconst box = boardBox();\n"
        "\treturn {\n"
        "\t\tx: box.x + box.width * ((GRID_LEFT_FRAC + GRID_RIGHT_FRAC) / 2),\n"
        "\t\ty: box.y + box.height * ((GRID_TOP_FRAC + GRID_BOTTOM_FRAC) / 2 + afterDarkYAdjust),\n"
        "\t\tscale: box.scale,",
        "\tconst box = boardBox(stateGame.tier);\n"
        "\tconst f = gridFracs(stateGame.tier);\n"
        "\treturn {\n"
        "\t\tx: box.x + box.width * ((f.left + f.right) / 2),\n"
        "\t\ty: box.y + box.height * ((f.top + f.bottom) / 2 + afterDarkYAdjust),\n"
        "\t\t// Deux echelles : les cellules d'After Dark n'ont pas la meme forme\n"
        "\t\t// que celles de base. Pixi accepte un objet pour scale.\n"
        "\t\tscale: { x: box.scaleX, y: box.scaleY },",
        "boardLayout par palier")

    say("")
    say("===== 6. utils.ts : rangees par palier =====")
    u = "apps/louvo/src/game/utils.ts"
    load(u)
    rep(u,
        "import { GRID_LEFT_FRAC, GRID_RIGHT_FRAC } from './constants';",
        "import { ROW_CENTERS_FRACS } from './constants';",
        "import ROW_CENTERS_FRACS")
    rep(u,
        "const AFTER_DARK_COLUMN_ADJUST_FRAC = [-0.035, -0.02, -0.005, 0.015, 0.03];",
        "// Remis a zero : les reperes propres a chaque palier rendent ces\n"
        "// decalages inutiles. Conserves au cas ou il faudrait re-affiner.\n"
        "const AFTER_DARK_COLUMN_ADJUST_FRAC = [0, 0, 0, 0, 0];",
        "decalages par colonne remis a zero")
    rep(u,
        "export const getSymbolY = (symbolIndexOfBoard: number) =>\n"
        "\tboardBox().y + boardBox().height * ROW_CENTERS_FRAC[symbolIndexOfBoard];",
        "export const getSymbolY = (symbolIndexOfBoard: number) => {\n"
        "\tconst box = boardBox(stateGame.tier);\n"
        "\tconst centres =\n"
        "\t\tstateGame.tier === 'after_dark' ? ROW_CENTERS_FRACS.afterDark : ROW_CENTERS_FRACS.base;\n"
        "\treturn box.y + box.height * centres[symbolIndexOfBoard];\n"
        "};",
        "getSymbolY par palier")

    for p, contenu in files.items():
        with open(p, "w", encoding="utf-8") as f:
            f.write(contenu)

    say("")
    say("===== 7. VERIFICATION =====")
    controles = (
        ("apps/louvo/src/game/constants.ts", "GRID_FRACS", True),
        ("apps/louvo/src/game/constants.ts", "ROW_CENTERS_FRACS", True),
        ("apps/louvo/src/game/boardBox.ts", "scaleX", True),
        ("apps/louvo/src/components/Background.svelte", "boardBox", False),
        ("apps/louvo/src/components/BoardFrame.svelte", "boardBox(context.stateGame.tier)", True),
        ("apps/louvo/src/game/stateGame.svelte.ts", "scale: { x: box.scaleX", True),
        ("apps/louvo/src/game/utils.ts", "ROW_CENTERS_FRACS.afterDark", True),
    )
    tout_ok = True
    for chemin, motif, attendu in controles:
        present = motif in open(chemin, encoding="utf-8").read()
        if present != attendu:
            tout_ok = False
        say("  %-62s %-8s %s" % (os.path.basename(chemin) + " : " + motif,
                                 "present" if present else "absent",
                                 "OK" if present == attendu else "*** ECHEC ***"))

    say("")
    say("===== 8. IMAGES PRESENTES =====")
    d = "apps/louvo/static/assets/sprites/screens"
    for fn in ("board_background.png", "board_background_after_dark.png",
               "board_frame_overlay.png", "board_frame_overlay_after_dark.png"):
        p = os.path.join(d, fn)
        if os.path.exists(p):
            say("  %-42s %8.2f Mo" % (fn, os.path.getsize(p) / 1048576.0))
        else:
            say("  %-42s *** ABSENT ***" % fn)
            tout_ok = False

    say("")
    say("===== 9. TAILLE DE SYMBOLE ATTENDUE =====")
    B = {"left": 391 / 1672.0, "right": 1275 / 1672.0, "top": 163 / 941.0, "bottom": 857 / 941.0}
    A = {"left": 463 / 1672.0, "right": 1197 / 1672.0, "top": 130 / 941.0, "bottom": 671 / 941.0}
    say("%-10s %-10s %14s %14s" % ("ecran", "type", "base (px)", "after dark (px)"))
    for nom, cw, ch, t, rw, rh in (
        ("Desktop", 1200, 675, "desktop", 1422, 800),
        ("Laptop", 1024, 576, "desktop", 1422, 800),
        ("Popout L", 800, 450, "landscape", 1600, 900),
        ("Popout S", 400, 225, "landscape", 1600, 900),
        ("Mobile L", 425, 812, "portrait", 800, 1422),
        ("Mobile M", 375, 667, "portrait", 800, 1422),
        ("Mobile S", 320, 568, "portrait", 800, 1422),
    ):
        s = min(cw / float(rw), ch / float(rh))
        cols = []
        for f in (B, A):
            if t == "portrait":
                w = 0.86 * rw / (f["right"] - f["left"])
                h = w / (1672 / 941.0)
            else:
                w, h = rw, rh
            cols.append("%.0f x %.0f" % (((f["right"] - f["left"]) / 5) * w * s,
                                         ((f["bottom"] - f["top"]) / 5) * h * s))
        say("%-10s %-10s %14s %14s" % (nom, t, cols[0], cols[1]))
    return tout_ok


try:
    ok = main()
except Exception:
    say("")
    say("===== ERREUR PYTHON =====")
    for l in traceback.format_exc().rstrip().split("\n"):
        say(l)
    ok = False

say("")
say("VERDICT : %s" % ("OK - build, puis regardez BUREAU d'abord"
                     if ok else "ECHEC - envoyez rapport58.txt"))
print("")
print("FINI - envoyez rapport58.txt")
