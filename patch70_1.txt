#!/usr/bin/env python3
# ============================================================
# Louvo - lot 73 : deux aplats noirs distincts, base et After Dark
#
# SEB AVAIT RAISON.
# L'aplat lisait gridFracs(tier), donc il etait bien "par palier" dans
# la forme - mais depuis le lot 52 les deux paliers contiennent les
# MEMES chiffres. En pratique il n'y avait qu'une seule grille, celle
# de la base, appliquee aussi a After Dark. D'ou le decor visible a
# gauche et a droite des wilds en After Dark.
#
# MESURE FAITE SUR LES DEUX IMAGES
# Chaque cadre contient 25 trous transparents, un par case. L'union de
# ces 25 trous donne la grille reelle :
#
#                    x                y           largeur
#   base           495 -> 1162     134 -> 653       667
#   after dark     471 -> 1190     136 -> 662       719
#
# After Dark est 52 px plus large, et pas centree pareil : 24 px de
# plus a gauche, 28 de plus a droite.
#
# CE QUE FAIT CE PATCH
# L'aplat n'utilise plus gridFracs. Il a ses propres mesures, une par
# palier, prises directement sur les images. On garde un petit debord
# de securite, recouvert par le cadre.
#
# CE QUE CE PATCH NE FAIT PAS
# Il ne touche pas a GRID_FRACS. Les symboles, les rangees, les lignes
# de gain et les cadres d'anticipation gardent exactement le placement
# que vous avez valide.
#
# Reecrit entierement components/BoardBackdrop.svelte.
# Sauvegarde : BoardBackdrop.svelte.avant-lot73
#
# Sort en erreur si ca echoue. AUCUN BUILD ICI.
#
# Usage :
#   cd /workspaces/web-sdk && python3 patch70.txt
# ============================================================
import os
import shutil
import sys
import traceback

ROOT = os.path.dirname(os.path.abspath(__file__))
os.chdir(ROOT)

RAPPORT = "rapport80.txt"
report = []


def flush():
    try:
        with open(RAPPORT, "w") as f:
            f.write("\n".join(report) + "\n")
    except OSError:
        pass


def say(line=""):
    report.append(line)
    print(line)
    try:
        sys.stdout.flush()
    except Exception:
        pass
    flush()


P = "apps/louvo/src/components/BoardBackdrop.svelte"

CONTENU = """<script lang="ts">
	import { Rectangle } from 'pixi-svelte';

	import { getContext } from '../game/context';
	import { boardBox } from '../game/boardBox';

	// ------------------------------------------------------------------
	// FOND NOIR DERRIERE LA GRILLE
	//
	// Depuis que le decor est une image pleine, les parties transparentes
	// des symboles laissent voir la scene au travers - visible surtout sur
	// les wilds, dont les bords sont decoupes. Avant, le fond contenait la
	// grille et faisait ecran.
	//
	// LES DEUX CADRES N'ONT PAS LA MEME GRILLE.
	// Mesure faite sur les images : chaque cadre contient 25 trous
	// transparents, un par case. L'union de ces trous donne la grille
	// reelle. After Dark est 52 px plus large que la base, et pas centree
	// pareil. C'est pour ca que cet aplat a ses PROPRES mesures au lieu de
	// reutiliser GRID_FRACS, qui contient les memes chiffres pour les deux
	// paliers depuis qu'on a aligne le placement des symboles.
	//
	// Le cadre etant dessine par-dessus (conteneur suivant), le debord de
	// securite ne se voit pas.
	// ------------------------------------------------------------------

	// ==================================================================
	// MESURES, en pixels des images 1672 x 941
	// board_frame_overlay.png et board_frame_overlay_after_dark.png
	// ==================================================================
	const TROU = {
		base: { left: 495, right: 1162, top: 134, bottom: 653 },
		afterDark: { left: 471, right: 1190, top: 136, bottom: 662 },
	} as const;

	// Debord de securite, en pixels d'image. Recouvert par le cadre.
	const DEBORD_PX = 30;

	// Opacite de l'aplat. 1 = noir franc.
	const OPACITE = 1;
	// ==================================================================

	const IMAGE_LARGEUR = 1672;
	const IMAGE_HAUTEUR = 941;

	const context = getContext();

	const box = $derived(boardBox(context.stateGame.tier));
	const trou = $derived(context.stateGame.tier === 'after_dark' ? TROU.afterDark : TROU.base);

	const zone = $derived({
		x: box.x + (box.width * (trou.left - DEBORD_PX)) / IMAGE_LARGEUR,
		y: box.y + (box.height * (trou.top - DEBORD_PX)) / IMAGE_HAUTEUR,
		width: (box.width * (trou.right - trou.left + DEBORD_PX * 2)) / IMAGE_LARGEUR,
		height: (box.height * (trou.bottom - trou.top + DEBORD_PX * 2)) / IMAGE_HAUTEUR,
	});
</script>

<Rectangle {...zone} backgroundColor={0x000000} backgroundAlpha={OPACITE} zIndex={-1} />
"""


def main():
    say("===== 1. %s =====" % P)
    if not os.path.exists(P):
        raise AssertionError("fichier absent - le lot 71 n'a pas ete lance. RIEN ENREGISTRE")
    if not os.path.exists(P + ".avant-lot73"):
        shutil.copyfile(P, P + ".avant-lot73")
    ancien = open(P, encoding="utf-8").read()
    say("  etat precedent : %s" % (
        "debord 60 px, une seule grille" if "DEBORD_X_PX" in ancien
        else "debord 4 unites, une seule grille" if "const DEBORD = 4" in ancien
        else "autre"))
    with open(P, "w", encoding="utf-8") as f:
        f.write(CONTENU)
    say("  reecrit avec deux jeux de mesures")

    say("")
    say("===== 2. VERIFICATION =====")
    apres = open(P, encoding="utf-8").read()
    tout_ok = True
    for motif, attendu in (
        ("base: { left: 495, right: 1162, top: 134, bottom: 653 }", True),
        ("afterDark: { left: 471, right: 1190, top: 136, bottom: 662 }", True),
        ("context.stateGame.tier === 'after_dark'", True),
        ("backgroundColor={0x000000}", True),
        ("zIndex={-1}", True),
        ("gridFracs", False),
    ):
        present = motif in apres
        if present != attendu:
            tout_ok = False
        say("  %-58s %-8s %s" % (motif[:58], "present" if present else "absent",
                                 "OK" if present == attendu else "*** ECHEC ***"))

    g = "apps/louvo/src/components/Game.svelte"
    branche = "<BoardBackdrop />" in open(g, encoding="utf-8").read()
    if not branche:
        tout_ok = False
    say("  %-58s %-8s %s" % ("Game.svelte : <BoardBackdrop />",
                             "present" if branche else "absent",
                             "OK" if branche else "*** ECHEC ***"))

    say("")
    say("===== 3. GRID_FRACS N'A PAS BOUGE =====")
    c = "apps/louvo/src/game/constants.ts"
    try:
        cst = open(c, encoding="utf-8").read()
        for l in cst.split("\n"):
            if "left:" in l and ("base" in l or "afterDark" in l):
                say("  %s" % l.strip()[:120])
    except OSError:
        say("  constants.ts illisible")
    say("")
    say("  Les symboles, rangees, lignes de gain et cadres d'anticipation")
    say("  gardent exactement le placement valide.")

    say("")
    say("===== 4. CE QUE COUVRE L'APLAT, EN PIXELS D'IMAGE =====")
    say("  %-12s %14s %14s %10s" % ("palier", "x", "y", "largeur"))
    say("  %-12s %14s %14s %10d" % ("base", "465 -> 1192", "104 -> 683", 1192 - 465))
    say("  %-12s %14s %14s %10d" % ("after dark", "441 -> 1220", "106 -> 692", 1220 - 441))
    say("")
    say("  Bordure opaque du cadre : 471 px au minimum a gauche (After")
    say("  Dark), 482 a droite, 136 en haut, 279 en bas. Le debord de 30")
    say("  reste dessous partout.")
    say("")
    say("  A noter, sans y toucher : les cases d'After Dark mesurent")
    say("  135 x 97 px contre 125 x 96 en base. Les symboles sont donc")
    say("  legerement plus etroits que leur case en After Dark. C'est")
    say("  reglable si vous le voulez, mais ca deplacerait tout ce que")
    say("  vous avez valide - je ne l'ai pas fait.")
    return tout_ok


try:
    ok = main()
except Exception:
    say("")
    say("===== ERREUR PYTHON =====")
    for l in traceback.format_exc().rstrip().split("\n"):
        say(l)
    ok = False

say("")
say("VERDICT : %s" % ("OK" if ok else "ECHEC"))
print("")
print("FINI - envoyez rapport80.txt")
sys.exit(0 if ok else 1)
