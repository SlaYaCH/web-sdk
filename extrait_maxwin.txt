===== WinLevelScreen =====
     1	<script lang="ts">
     2		// ============================================================
     3		// L'ecran de palier : l'image du palier et le montant qui monte.
     4		//
     5		// Tout est ici, y compris le montant : un seul fichier a ouvrir
     6		// pour regler cet ecran, et rien a chercher ailleurs.
     7		// ============================================================
     8		import { Rectangle, Sprite } from 'pixi-svelte';
     9		import { ResponsiveBitmapText } from 'components-pixi';
    10		import { stateBetDerived } from 'state-shared';
    11	
    12		import { SYMBOL_SIZE } from '../game/constants';
    13	
    14		type Props = { alias: string; montant: string };
    15		const props: Props = $props();
    16	
    17		// =====================================================
    18		//  BIG / SUPER / MEGA / EPIC WIN  -  vos quatre images
    19		// =====================================================
    20		const TAILLE = 1 / 3; // <<< LA taille de l'image. 1 = comme avant.
    21		const IMAGE_Y = -SYMBOL_SIZE * 0.55; // hauteur de l'image
    22		const MONTANT_Y = SYMBOL_SIZE * 0.95; // hauteur du montant
    23		const MONTANT_TAILLE = SYMBOL_SIZE * 0.75; // taille des chiffres
    24		const MONTANT_LARGEUR = SYMBOL_SIZE * 3.4; // au-dela, le texte retrecit
    25		const FOND_LARGEUR = SYMBOL_SIZE * 3.8; // le rectangle noir
    26		const FOND_HAUTEUR = SYMBOL_SIZE * 1.0;
    27		const FOND_OPACITE = 0.6; // 0 = pas de fond noir du tout
    28	
    29		// =====================================================
    30		//  MAX WIN  -  n'a jamais utilise le spine, ne bouge pas
    31		// =====================================================
    32		const MAX_IMAGE_Y = -SYMBOL_SIZE * 0.5;
    33		const MAX_MONTANT_Y = SYMBOL_SIZE * 2.15;
    34		const MAX_MONTANT_TAILLE = SYMBOL_SIZE * 1.05;
    35		const MAX_MONTANT_LARGEUR = SYMBOL_SIZE * 6.2;
    36	
    37		// --- l'animation d'entree ---
    38		const ENTREE_MS = 420; // duree du surgissement
    39		const ECHELLE_DEPART = 0.62; // taille de depart
    40		const RESPIRATION = 0.018; // souffle continu, 0 pour le couper
    41		const PERIODE_MS = 2600;
    42	
    43		// --- les proportions d'origine de l'ecran, avant reduction ---
    44		const LARGEUR_PLEINE = SYMBOL_SIZE * 6.6;
    45		const HAUTEUR_PLEINE = SYMBOL_SIZE * 4.4;
    46	
    47		const CLE_ECRAN = {
    48			big: 'bigwinScreen',
    49			superwin: 'superwinScreen',
    50			mega: 'megawinScreen',
    51			epic: 'epicwinScreen',
    52			max: 'maxwinScreen',
    53		} as const;
    54	
    55		const cle = $derived(CLE_ECRAN[props.alias as keyof typeof CLE_ECRAN]);
    56		const estMax = $derived(props.alias === 'max');
    57	
    58		const largeur = $derived(estMax ? LARGEUR_PLEINE : LARGEUR_PLEINE * TAILLE);
    59		const hauteur = $derived(estMax ? HAUTEUR_PLEINE : HAUTEUR_PLEINE * TAILLE);
    60		const imageY = $derived(estMax ? MAX_IMAGE_Y : IMAGE_Y);
    61		const montantY = $derived(estMax ? MAX_MONTANT_Y : MONTANT_Y);
    62		const montantTaille = $derived(estMax ? MAX_MONTANT_TAILLE : MONTANT_TAILLE);
    63		const montantLargeur = $derived(estMax ? MAX_MONTANT_LARGEUR : MONTANT_LARGEUR);
    64	
    65		let echelle = $state(ECHELLE_DEPART);
    66	
    67		$effect(() => {
    68			let arrete = false;
    69			let precedent = performance.now();
    70			let horloge = 0;
    71	
    72			const image = (maintenant: number) => {
    73				if (arrete) return;
    74	
    75				// Le turbo accelere la mise en scene : on suit la meme horloge
    76				// que le reste du jeu plutot que le temps reel.
    77				const vitesse = stateBetDerived.timeScale();
    78				horloge += (maintenant - precedent) * (vitesse > 0 ? vitesse : 1);
    79				precedent = maintenant;
    80	
    81				const t = Math.min(1, horloge / ENTREE_MS);
    82				const adouci = 1 - Math.pow(1 - t, 3);
    83				const souffle =
    84					t >= 1
    85						? Math.sin(((horloge - ENTREE_MS) / PERIODE_MS) * Math.PI * 2) * RESPIRATION
    86						: 0;
    87	
    88				echelle = ECHELLE_DEPART + (1 - ECHELLE_DEPART) * adouci + souffle;
    89	
    90				requestAnimationFrame(image);
    91			};
    92	
    93			requestAnimationFrame(image);
    94			return () => {
    95				arrete = true;
    96			};
    97		});
    98	</script>
    99	
   100	{#if cle}
   101		<Sprite
   102			anchor={0.5}
   103			key={cle}
   104			y={imageY}
   105			width={largeur * echelle}
   106			height={hauteur * echelle}
   107		/>
   108	
   109		{#if !estMax && FOND_OPACITE > 0}
   110			<!-- Le fond noir derriere le montant. MAX WIN n'en a jamais eu :
   111				son image occupe tout l'ecran, un rectangle par-dessus ferait
   112				une tache. -->
   113			<Rectangle
   114				anchor={0.5}
   115				y={MONTANT_Y}
   116				width={FOND_LARGEUR}
   117				height={FOND_HAUTEUR}
   118				backgroundColor={0x000000}
   119				alpha={FOND_OPACITE}
   120			/>
   121		{/if}
   122	
   123		<ResponsiveBitmapText
   124			anchor={0.5}
   125			y={montantY}
   126			maxWidth={montantLargeur}
   127			text={props.montant}
   128			style={{
   129				fontFamily: 'gold', fill: 0xff2d6a,
   130				fontSize: montantTaille,
   131				align: 'center',
   132				fontWeight: 'bold',
   133				letterSpacing: 0,
   134			}}
   135		/>
   136	{/if}

===== LouvoWinPanel =====
     1	<script lang="ts">
     2		import { onMount } from 'svelte';
     3		import { MainContainer } from 'components-layout';
     4		import { bookEventAmountToCurrencyString } from 'utils-shared/amount';
     5		import { BitmapText, Container, Sprite } from 'pixi-svelte';
     6	
     7		import { getContext } from '../game/context';
     8		import { SYMBOL_SIZE } from '../game/constants';
     9		import { counterAnchor } from '../game/boardBox';
    10	
    11		const context = getContext();
    12	
    13		// ============================================================
    14		// LE GAIN DU TOUR, SOUS LA GRILLE
    15		//
    16		// En dessous de BIG WIN, le montant s'affiche au centre de la
    17		// grille puis disparait quasi immediatement : Win.svelte appelle
    18		// finishCountUp() puis oncomplete() dans la foulee. En turbo, on
    19		// ne le voit pas. Ici il RESTE jusqu'au tour suivant.
    20		//
    21		// EN BASE GAME SEULEMENT. En bonus, cette bande porte deja le
    22		// compteur de free spins a gauche et le gain total a droite ;
    23		// entre les deux il reste 0,47 SYMBOL_SIZE et ce cadre en fait
    24		// 1,50. Le compteur TOTAL WIN y joue deja ce role.
    25		//
    26		// Le fondu ne joue QU'A LA DISPARITION : le montant surgit d'un
    27		// coup avec le gain, et s'efface doucement au tour suivant.
    28		// FadeContainer fait forcement les deux sens, d'ou le Container
    29		// simple et l'opacite pilotee ici - meme mecanique que les coeurs
    30		// de l'anticipation.
    31		// ============================================================
    32		const HAUTEUR = SYMBOL_SIZE * 1.0; // <<< LA taille du cadre
    33		const DECALAGE_Y = -SYMBOL_SIZE * 0.08; // <<< remonter (negatif) ou descendre
    34		const RATIO = 1536 / 1024; // proportions de votre image, a ne pas deformer
    35		const TEXTE_TAILLE = SYMBOL_SIZE * 0.26;
    36		const TEXTE_Y = 0; // le montant dans le cadre
    37		const SORTIE_MS = 320; // duree du fondu de disparition
    38	
    39		const largeur = HAUTEUR * RATIO;
    40		const ancre = $derived(counterAnchor(context.stateGame.tier));
    41	
    42		let montant = $state(0);
    43		let opacite = $state(0);
    44	
    45		let image: number | null = null;
    46		const arreter = () => {
    47			if (image !== null) {
    48				cancelAnimationFrame(image);
    49				image = null;
    50			}
    51		};
    52	
    53		// Apparition : franche, sans fondu.
    54		const apparaitre = () => {
    55			arreter();
    56			opacite = 1;
    57		};
    58	
    59		// Disparition : fondu, puis le composant se retire de lui-meme.
    60		const disparaitre = () => {
    61			arreter();
    62			if (opacite === 0) return;
    63			const depart = opacite;
    64			const debut = performance.now();
    65			const boucle = (maintenant: number) => {
    66				const t = (maintenant - debut) / SORTIE_MS;
    67				if (t >= 1) {
    68					opacite = 0;
    69					image = null;
    70					return;
    71				}
    72				opacite = depart * (1 - t);
    73				image = requestAnimationFrame(boucle);
    74			};
    75			image = requestAnimationFrame(boucle);
    76		};
    77	
    78		onMount(() => arreter);
    79	
    80		context.eventEmitter.subscribeOnMount({
    81			// Le tour suivant efface le montant du precedent.
    82			spinStart: () => disparaitre(),
    83			winUpdate: (emitterEvent) => {
    84				// Un gain tombe pendant un bonus ne doit pas etre memorise :
    85				// il reapparaitrait au retour en base game.
    86				if (context.stateGame.gameType !== 'basegame') return;
    87				if (emitterEvent.amount > 0) {
    88					montant = emitterEvent.amount;
    89					apparaitre();
    90				}
    91			},
    92		});
    93	
    94		const visible = $derived(opacite > 0 && context.stateGame.gameType === 'basegame');
    95	</script>
    96	
    97	<MainContainer>
    98		{#if visible}
    99			<Container
   100				x={ancre.centerX}
   101				y={ancre.y + HAUTEUR * 0.5 + DECALAGE_Y}
   102				alpha={opacite}
   103			>
   104				<Sprite anchor={0.5} key="totalWinFrame" width={largeur} height={HAUTEUR} />
   105				<BitmapText
   106					anchor={{ x: 0.5, y: 0.5 }}
   107					y={TEXTE_Y}
   108					text={bookEventAmountToCurrencyString(montant)}
   109					style={{
   110						fontFamily: 'gold', fill: 0xffffff,
   111						fontSize: TEXTE_TAILLE,
   112						wordWrap: false,
   113					}}
   114				/>
   115			</Container>
   116		{/if}
   117	</MainContainer>
