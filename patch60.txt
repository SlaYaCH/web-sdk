#!/usr/bin/env python3
# ============================================================
# Louvo - lot 60 : lignes de gain remontees d'une case, PARTOUT
#
# CORRECTION DU DIAGNOSTIC.
# Je croyais que le decalage n'existait qu'en portrait, et j'allais
# livrer une rustine qui ne s'activait qu'en portrait. Seb a verifie :
# les lignes sont une case trop bas dans TOUS les modes, bureau compris.
#
# Et ca, ce n'est plus un mystere de mise en page : un ecart constant
# d'exactement une case, identique partout, c'est une convention
# d'indice qui ne colle pas. Autrement dit props.positions[].row ne
# compte pas les rangees a partir du meme zero que les symboles.
# La formule geometrique, elle, est juste - c'est l'indice qui est
# decale de 1.
#
# On corrige donc a UN seul endroit, avec une constante nommee, sans
# condition de mise en page : plus simple et plus honnete que la
# rustine portrait que j'allais envoyer.
#
# SAUVEGARDE : WinLineReveal.svelte.avant-lot60
# Rapport ecrit apres chaque ligne. AUCUN BUILD.
#
# Usage :
#   1) deposer ce fichier a la RACINE de web-sdk
#   2) cd /workspaces/web-sdk && python3 patch60.txt
#   3) envoyer rapport67.txt
# ============================================================
import os
import re
import shutil
import sys
import traceback

ROOT = os.path.dirname(os.path.abspath(__file__))
os.chdir(ROOT)

RAPPORT = "rapport67.txt"
report = []


def flush():
    try:
        with open(RAPPORT, "w") as f:
            f.write("\n".join(report) + "\n")
    except OSError:
        pass


def say(line=""):
    report.append(line)
    print(line)
    try:
        sys.stdout.flush()
    except Exception:
        pass
    flush()


NOUVEAU = """	const WIN_LINE_Y_OFFSET = 0; // nudge fin, en unites de plateau

	// ------------------------------------------------------------------
	// DECALAGE DE RANGEE.
	// Les lignes de gain tombaient exactement une case trop bas, dans
	// tous les modes. Un ecart constant d'une case, identique partout,
	// n'est pas un probleme de mise en page : c'est que l'indice de
	// rangee fourni par props.positions ne part pas du meme zero que
	// celui des symboles. La formule ci-dessous reste la bonne, on
	// remet juste l'indice dans le bon repere.
	// Mettre 0 pour annuler. Unite : la case.
	// ------------------------------------------------------------------
	const WIN_LINE_ROW_SHIFT = -1;

	const points = props.positions.map((p) => ({
		x: getSymbolX(p.reel),
		y: SYMBOL_HEIGHT * (p.row + WIN_LINE_ROW_SHIFT + REEL_PADDING) + WIN_LINE_Y_OFFSET,
	}));"""

# Le bloc va de la constante de nudge jusqu'a la fin de la construction
# des points. Ecrit en regex pour tolerer les deux etats possibles du
# fichier (lot 56 seul, ou lot 56 + une rustine portrait deja posee).
MOTIF = re.compile(
    r"[ \t]*const WIN_LINE_Y_OFFSET\b.*?\}\)\);",
    re.DOTALL,
)


def main():
    p = "apps/louvo/src/components/WinLineReveal.svelte"
    s = p + ".avant-lot60"
    if not os.path.exists(s):
        shutil.copyfile(p, s)
    txt = open(p, encoding="utf-8").read()

    say("===== 1. ETAT DE DEPART =====")
    say("  fichier          : %s" % p)
    say("  taille           : %d octets" % len(txt))
    say("  rustine portrait : %s" % ("presente" if "PORTRAIT_ROW_SHIFT" in txt else "absente"))
    say("  deja corrige     : %s" % ("oui" if "WIN_LINE_ROW_SHIFT" in txt else "non"))

    say("")
    say("===== 2. BLOC REMPLACE =====")
    if "WIN_LINE_ROW_SHIFT" in txt:
        say("  deja applique, rien a faire")
    else:
        trouves = MOTIF.findall(txt)
        if len(trouves) != 1:
            raise AssertionError(
                "bloc des points trouve %d fois (attendu 1) - RIEN MODIFIE" % len(trouves))
        say("  --- avant ---")
        for l in trouves[0].split("\n"):
            say("  | " + l)
        txt = MOTIF.sub(lambda _m: NOUVEAU, txt, count=1)

        # Si une rustine portrait avait ete posee, son contexte devient
        # inutile ici ; on ne le retire pas de force, il est inoffensif,
        # mais on le signale.
        if "PORTRAIT_ROW_SHIFT" in txt:
            say("  ATTENTION : PORTRAIT_ROW_SHIFT survit ailleurs dans le fichier")

        with open(p, "w", encoding="utf-8") as f:
            f.write(txt)
        say("  --- apres ---")
        for l in NOUVEAU.split("\n"):
            say("  | " + l)

    say("")
    say("===== 3. VERIFICATION =====")
    apres = open(p, encoding="utf-8").read()
    tout_ok = True
    controles = (
        ("WIN_LINE_ROW_SHIFT = -1", True),
        ("p.row + WIN_LINE_ROW_SHIFT + REEL_PADDING", True),
        ("PORTRAIT_ROW_SHIFT", False),
        ("getSymbolX(p.reel)", True),
    )
    for motif, attendu in controles:
        present = motif in apres
        if present != attendu:
            tout_ok = False
        say("  %-46s %-8s %s" % (
            motif,
            "present" if present else "absent",
            "OK" if present == attendu else "*** ECHEC ***"))

    say("")
    say("===== 4. POSITION DES LIGNES (unites de plateau) =====")
    SH, RP = 91.0, 0.53
    say("  %-8s %14s %14s" % ("rangee", "avant", "apres"))
    for r in range(5):
        say("  %-8d %14.1f %14.1f" % (r + 1, SH * (r + RP), SH * (r - 1 + RP)))
    say("")
    say("  Tout remonte de 91 unites, soit une case pile, dans tous les modes.")
    return tout_ok


try:
    ok = main()
except Exception:
    say("")
    say("===== ERREUR PYTHON - AUCUN FICHIER ENREGISTRE =====")
    for l in traceback.format_exc().rstrip().split("\n"):
        say(l)
    ok = False

say("")
say("VERDICT : %s" % ("OK - vous pouvez lancer le build" if ok else "ECHEC - envoyez rapport67.txt"))
print("")
print("FINI - envoyez rapport67.txt")
