#!/usr/bin/env python3
# ============================================================
# Louvo - lot 58 : UNIQUEMENT le centrage lateral des bannieres
#
# Le lot 57 touchait aussi a la verticale. Ce n'etait pas demande : la
# hauteur et la taille des bannieres conviennent. Ce lot les REMET
# exactement comme avant et ne garde que la correction laterale.
#
# Il fonctionne QUEL QUE SOIT L'ETAT du fichier :
#   - si le lot 57 a ete applique, il annule sa partie verticale
#   - s'il ne l'a pas ete, il ne touche qu'a l'horizontale
#
# CE QUI CHANGE, ET RIEN D'AUTRE :
#   bannerX : le rattrapage manuel "- 2 px sauf sur la premiere colonne"
#   est remplace par le centre de colonne exact des symboles, plus un
#   reglage BANNER_X_OFFSET a 0 que vous pouvez ajuster vous-meme.
#   Deplacement : +2 unites pour les colonnes 2 a 5, rien pour la 1re.
#
# SAUVEGARDE : SpecialRevealOverlay.svelte.avant-lot58
# Rapport ecrit apres chaque ligne. AUCUN BUILD.
#
# Usage :
#   1) deposer ce fichier a la RACINE de web-sdk
#   2) cd /workspaces/web-sdk && python3 patch58.txt
#   3) envoyer rapport66.txt
# ============================================================
import os
import shutil
import sys
import traceback

ROOT = os.path.dirname(os.path.abspath(__file__))
os.chdir(ROOT)

RAPPORT = "rapport66.txt"
report = []


def flush():
    try:
        with open(RAPPORT, "w") as f:
            f.write("\n".join(report) + "\n")
    except OSError:
        pass


def say(line=""):
    report.append(line)
    print(line)
    try:
        sys.stdout.flush()
    except Exception:
        pass
    flush()


BLOC_BANNER_Y = """
	// La banniere couvre toute la colonne : son centre est celui de la
	// rangee du milieu, exprime dans la MEME convention locale que les
	// symboles, les coeurs Super Like et les lignes de gain :
	//     centre de rangee = SYMBOL_HEIGHT * (rangee + REEL_PADDING)
	// La moitie de la hauteur du plateau valait 227.5 alors que le vrai
	// centre vaut 230.2 : 2.7 unites trop haut, parce que REEL_PADDING
	// vaut 0.53 et non 0.5.
	const BANNER_Y = SYMBOL_HEIGHT * ((BOARD_DIMENSIONS.y - 1) / 2 + REEL_PADDING);"""

REGLAGE = """
	// Reglage lateral des bannieres, en unites de plateau.
	// 0 = exactement le centre de colonne utilise par les symboles.
	// Negatif = vers la gauche, positif = vers la droite.
	const BANNER_X_OFFSET = 0;"""


def main():
    p = "apps/louvo/src/components/SpecialRevealOverlay.svelte"
    s = p + ".avant-lot58"
    if not os.path.exists(s):
        shutil.copyfile(p, s)
    txt = open(p, encoding="utf-8").read()
    modifie = False

    # ---------- 1. Annuler la verticale si le lot 57 est passe ----------
    say("===== 1. VERTICALE : remise a l'identique =====")
    if "BANNER_Y" not in txt:
        say("  la verticale n'a jamais ete touchee, rien a faire")
    else:
        for ancien, nouveau, tag in (
            (BLOC_BANNER_Y, "", "constante BANNER_Y retiree"),
            ("\timport { SYMBOL_HEIGHT, REEL_PADDING, BOARD_DIMENSIONS } from '../game/constants';",
             "\timport { BOARD_SIZES } from '../game/constants';", "import remis"),
            ("y={BANNER_Y}", "y={BOARD_SIZES.height / 2}", "hauteur remise"),
        ):
            n = txt.count(ancien)
            if n != 1:
                raise AssertionError("%s : motif trouve %d fois (attendu 1) - rien modifie"
                                     % (tag, n))
            txt = txt.replace(ancien, nouveau, 1)
            say("  %-40s OK" % tag)
        modifie = True

    # ---------- 2. La correction laterale, seule ----------
    say("")
    say("===== 2. LATERAL : le seul changement voulu =====")
    fudge = ("\t\t\t\t\t// Correction fine : bannieres des rouleaux 2-5 ~1mm trop a droite (valeur ajustable).\n"
             "\t\t\t\t\tbannerX: getSymbolX(emitterEvent.reelIndex) - (emitterEvent.reelIndex > 0 ? 2 : 0),")
    propre = "\t\t\t\t\tbannerX: getSymbolX(emitterEvent.reelIndex),"
    voulu = ("\t\t\t\t\t// Centre de colonne exact, celui des symboles.\n"
             "\t\t\t\t\tbannerX: getSymbolX(emitterEvent.reelIndex) + BANNER_X_OFFSET,")

    if "BANNER_X_OFFSET" in txt:
        say("  deja applique")
    else:
        if txt.count(fudge) == 1:
            txt = txt.replace(fudge, voulu, 1)
            say("  rattrapage manuel remplace par le centre exact")
        elif txt.count(propre) == 1:
            txt = txt.replace(propre, voulu, 1)
            say("  reglage ajoute (le rattrapage avait deja ete retire)")
        else:
            raise AssertionError("ligne bannerX non reconnue - rien modifie")

        ancre = "\tconst context = getContext();"
        if txt.count(ancre) != 1:
            raise AssertionError("ancre getContext introuvable - rien modifie")
        txt = txt.replace(ancre, ancre + "\n" + REGLAGE, 1)
        say("  constante BANNER_X_OFFSET ajoutee")
        modifie = True

    if modifie:
        with open(p, "w", encoding="utf-8") as f:
            f.write(txt)

    # ---------- 3. Verification ----------
    say("")
    say("===== 3. VERIFICATION =====")
    apres = open(p, encoding="utf-8").read()
    tout_ok = True
    for motif, attendu in (
        ("y={BOARD_SIZES.height / 2}", True),
        ("BANNER_Y", False),
        ("BANNER_X_OFFSET", True),
        ("reelIndex > 0 ? 2 : 0", False),
    ):
        present = motif in apres
        if present != attendu:
            tout_ok = False
        say("  %-46s %-8s %s" % (motif, "present" if present else "absent",
                                 "OK" if present == attendu else "*** ECHEC ***"))

    say("")
    say("===== 4. CE QUI BOUGE, ET C'EST TOUT =====")
    SW, RP = 116.0, 0.53
    say("  hauteur : INCHANGEE")
    say("  taille  : INCHANGEE")
    say("")
    say("  %-10s %10s %10s %8s" % ("colonne", "avant", "apres", "ecart"))
    for i in range(5):
        avant = SW * (i + RP) - (2 if i > 0 else 0)
        ap = SW * (i + RP)
        say("  %-10d %10.1f %10.1f %8.1f" % (i + 1, avant, ap, ap - avant))
    say("")
    say("  Si ce n'est toujours pas parfait, ajustez BANNER_X_OFFSET en tete")
    say("  de SpecialRevealOverlay.svelte : negatif = vers la gauche.")
    return tout_ok


try:
    ok = main()
except Exception:
    say("")
    say("===== ERREUR PYTHON - AUCUN FICHIER ENREGISTRE =====")
    for l in traceback.format_exc().rstrip().split("\n"):
        say(l)
    ok = False

say("")
say("VERDICT : %s" % ("OK - build" if ok else "ECHEC - envoyez rapport66.txt"))
print("")
print("FINI - envoyez rapport66.txt")
