#!/usr/bin/env python3
# ============================================================
# Louvo - lot 35 : REPRISE du nettoyage des assets de la demo
#
# Difference capitale avec le lot 34 :
#   - le rapport est ecrit sur le disque APRES CHAQUE LIGNE,
#     donc meme si le terminal coupe, rapport41.txt existe
#     et contient tout ce qui a ete fait jusque la.
#   - AUCUN BUILD n'est lance ici (c'est le build qui a tue
#     le lot 34 avant qu'il n'ecrive son rapport).
#     Vous lancerez le build vous-meme apres, comme d'habitude.
#
# Le script est REPRENABLE : si le lot 34 a deja deplace les
# dossiers, il le detecte et n'y touche pas une seconde fois.
#
# VERROUS EN DUR - jamais touches :
#   - assets/fonts/**                 (polices chargees dynamiquement)
#   - assets/audio/**                 (sons de sounds.json)
#   - assets/spines/anticipation/**   (les rochers, votre demande)
#
# Usage :
#   1) deposer ce fichier a la RACINE de web-sdk
#   2) cd /workspaces/web-sdk && python3 patch35.txt
#   3) envoyer rapport41.txt
# ============================================================
import os
import re
import shutil
import subprocess
import sys
import traceback

ROOT = os.path.dirname(os.path.abspath(__file__))
os.chdir(ROOT)

BUILD = "apps/louvo/build"
STATIC = "apps/louvo/static/assets"
BACKUP = "_demo_assets_backup"
RAPPORT = "rapport41.txt"

PROTEGES = ("fonts", "audio")
PROTEGES_DIRS = ("spines/anticipation",)

report = []


def flush():
    try:
        with open(RAPPORT, "w") as f:
            f.write("\n".join(report) + "\n")
    except OSError:
        pass


def say(line=""):
    report.append(line)
    print(line)
    try:
        sys.stdout.flush()
    except Exception:
        pass
    flush()


def human(n):
    if n < 1024:
        return "%d o" % n
    if n < 1024 * 1024:
        return "%.0f Ko" % (n / 1024.0)
    return "%.2f Mo" % (n / 1024.0 / 1024.0)


def dir_size(path):
    total = 0
    for dirpath, _d, filenames in os.walk(path):
        for fn in filenames:
            try:
                total += os.path.getsize(os.path.join(dirpath, fn))
            except OSError:
                pass
    return total


def main():
    # ---------- 0. ETAT ACTUEL (avant toute action) ----------
    say("===== 0. ETAT AVANT ACTION =====")
    if os.path.isdir(BACKUP):
        say("_demo_assets_backup/ : PRESENT  (%s)" % human(dir_size(BACKUP)))
        base = os.path.join(BACKUP, "assets")
        if os.path.isdir(base):
            for top in sorted(os.listdir(base)):
                tp = os.path.join(base, top)
                if os.path.isdir(tp):
                    for sub in sorted(os.listdir(tp)):
                        sp = os.path.join(tp, sub)
                        if os.path.isdir(sp):
                            say("   deja mis a l'ecart : %s/%s  (%s)" % (top, sub, human(dir_size(sp))))
    else:
        say("_demo_assets_backup/ : ABSENT (le lot 34 n'a rien deplace)")

    say("")
    if not os.path.isdir(STATIC):
        say("ERREUR : %s introuvable" % STATIC)
        return False
    say("poids actuel de static/assets : %s" % human(dir_size(STATIC)))
    for top in sorted(os.listdir(STATIC)):
        tp = os.path.join(STATIC, top)
        if not os.path.isdir(tp):
            continue
        subs = sorted([d for d in os.listdir(tp) if os.path.isdir(os.path.join(tp, d))])
        if subs:
            say("  %-10s : %s" % (top, ", ".join(subs)))
        else:
            say("  %-10s : (%d fichiers)" % (top, len(os.listdir(tp))))

    say("")
    try:
        out = subprocess.run(["git", "status", "--short"], capture_output=True,
                             text=True, timeout=60).stdout
        lignes = [l for l in out.rstrip().split("\n") if l.strip()]
        say("git status : %d entrees" % len(lignes))
        for l in lignes[:10]:
            say("   " + l)
        if len(lignes) > 10:
            say("   ... (%d de plus)" % (len(lignes) - 10))
    except Exception as e:
        say("git indisponible : %s" % e)

    # ---------- 1. Ce que le bundle deploye demande vraiment ----------
    say("")
    say("===== 1. REFERENCES TROUVEES DANS LE BUNDLE =====")
    if not os.path.isdir(BUILD):
        say("ERREUR : dossier build introuvable - lancez d'abord un build")
        return False

    referenced = set()
    for dirpath, _d, filenames in os.walk(BUILD):
        if os.path.sep + "assets" + os.path.sep in dirpath + os.path.sep:
            continue
        for fn in filenames:
            if not fn.endswith((".js", ".html", ".json", ".css")):
                continue
            try:
                with open(os.path.join(dirpath, fn), "r", errors="ignore") as f:
                    content = f.read()
            except OSError:
                continue
            for h in re.findall(r"assets/[A-Za-z0-9_./-]+", content):
                referenced.add(os.path.splitext(h)[0])

    say("chemins d'assets distincts demandes par le jeu : %d" % len(referenced))
    if len(referenced) < 30:
        say("ARRET DE SECURITE : trop peu de references, le bundle semble incomplet.")
        say("Relancez un build puis relancez ce script. Rien n'a ete touche.")
        return False

    def est_utilise(path):
        rel = os.path.relpath(path, os.path.dirname(STATIC)).replace(os.sep, "/")
        return os.path.splitext(rel)[0] in referenced

    # ---------- 2. Dossiers jamais demandes ----------
    say("")
    say("===== 2. DOSSIERS MIS A L'ECART =====")
    candidats = []
    for top in sorted(os.listdir(STATIC)):
        if top in PROTEGES:
            continue
        top_path = os.path.join(STATIC, top)
        if not os.path.isdir(top_path):
            continue
        for sub in sorted(os.listdir(top_path)):
            sub_path = os.path.join(top_path, sub)
            if not os.path.isdir(sub_path):
                continue
            label = "%s/%s" % (top, sub)
            if label in PROTEGES_DIRS:
                continue
            nb = 0
            used = 0
            for dirpath, _d, filenames in os.walk(sub_path):
                for fn in filenames:
                    nb += 1
                    if est_utilise(os.path.join(dirpath, fn)):
                        used += 1
            if nb > 0 and used == 0:
                candidats.append((label, sub_path, dir_size(sub_path)))

    total = 0
    for label, path, size in sorted(candidats, key=lambda x: -x[2]):
        dest = os.path.join(BACKUP, "assets", label)
        os.makedirs(os.path.dirname(dest), exist_ok=True)
        if os.path.exists(dest):
            shutil.rmtree(dest)
        shutil.move(path, dest)
        total += size
        say("%-46s %10s  -> %s" % (label, human(size), dest))
    if not candidats:
        say("(aucun dossier a deplacer - c'etait deja fait)")
    say("")
    say("RETIRE DU BUILD PAR CETTE EXECUTION : %s" % human(total))

    # ---------- 3. index.ts perdus dans static ----------
    say("")
    say("===== 3. index.ts (code source jamais execute) =====")
    moved_ts = 0
    for dirpath, _d, filenames in os.walk(STATIC):
        rel_dir = os.path.relpath(dirpath, STATIC).replace(os.sep, "/")
        if rel_dir.split("/")[0] in PROTEGES or rel_dir in PROTEGES_DIRS:
            continue
        for fn in filenames:
            if fn != "index.ts":
                continue
            src = os.path.join(dirpath, fn)
            dest = os.path.join(BACKUP, "assets", rel_dir, fn)
            os.makedirs(os.path.dirname(dest), exist_ok=True)
            if os.path.exists(dest):
                os.remove(dest)
            shutil.move(src, dest)
            moved_ts += 1
    say("deplaces cette fois : %d (ceux de fonts/ et anticipation/ restent en place)" % moved_ts)

    # ---------- 4. Fichiers non demandes DANS des dossiers utilises ----------
    say("")
    say("===== 4. FICHIERS NON DEMANDES restant dans des dossiers utilises =====")
    say("(RIEN n'est touche ici - on decidera ensemble)")
    restants = []
    for dirpath, _d, filenames in os.walk(STATIC):
        rel_dir = os.path.relpath(dirpath, STATIC).replace(os.sep, "/")
        if rel_dir.split("/")[0] in PROTEGES:
            continue
        for fn in filenames:
            fp = os.path.join(dirpath, fn)
            if not est_utilise(fp):
                try:
                    restants.append((rel_dir + "/" + fn, os.path.getsize(fp)))
                except OSError:
                    pass
    for name, size in sorted(restants, key=lambda x: -x[1])[:25]:
        say("%-58s %10s" % (name, human(size)))
    if not restants:
        say("(aucun)")

    # ---------- 5. AUDIT AUDIO (lecture seule) ----------
    say("")
    say("===== 5. AUDIT DU DOSSIER AUDIO (aucune suppression) =====")
    audio_dir = os.path.join(STATIC, "audio")
    sounds_json = os.path.join(audio_dir, "sounds.json")
    if os.path.exists(sounds_json):
        with open(sounds_json, "r", errors="ignore") as f:
            sj = f.read()
        audio_refs = set(re.findall(r"[A-Za-z0-9_./-]+\.(?:mp3|ogg|m4a|webm|wav|aac|flac)", sj))
        say("fichiers cites dans sounds.json : %s"
            % (", ".join(sorted(os.path.basename(a) for a in audio_refs)) or "(aucun)"))
        say("entrees 'sprite' declarees      : %d" % len(re.findall(r'"[^"]+"\s*:\s*\[\s*\d', sj)))
        say("")
        say("%-46s %10s   %s" % ("fichier present", "poids", "cite ?"))
        total_audio = 0
        total_orphelin = 0
        for fn in sorted(os.listdir(audio_dir)):
            fp = os.path.join(audio_dir, fn)
            if not os.path.isfile(fp):
                continue
            size = os.path.getsize(fp)
            total_audio += size
            cite = fn == "sounds.json" or any(os.path.basename(a) == fn for a in audio_refs)
            if not cite:
                total_orphelin += size
            say("%-46s %10s   %s" % (fn, human(size), "oui" if cite else "*** NON ***"))
        say("")
        say("poids total audio   : %s" % human(total_audio))
        say("poids des non-cites : %s" % human(total_orphelin))
    else:
        say("sounds.json introuvable - audit impossible")

    # ---------- 6. .gitignore ----------
    say("")
    say("===== 6. .gitignore =====")
    try:
        gi = ""
        if os.path.exists(".gitignore"):
            with open(".gitignore") as f:
                gi = f.read()
        if BACKUP not in gi:
            with open(".gitignore", "a") as f:
                f.write("\n# Assets de la demo d'origine, mis a l'ecart (recuperables ici ou via git)\n%s/\n" % BACKUP)
            say("%s/ ajoute au .gitignore" % BACKUP)
        else:
            say("%s/ deja present dans .gitignore" % BACKUP)
    except OSError as e:
        say("gitignore : %s" % e)

    # ---------- 7. Bilan ----------
    say("")
    say("===== 7. BILAN =====")
    say("poids de static/assets apres nettoyage : %s" % human(dir_size(STATIC)))
    if os.path.isdir(BACKUP):
        say("poids mis a l'ecart au total          : %s" % human(dir_size(BACKUP)))
    say("")
    say("AUCUN BUILD N'A ETE LANCE PAR CE SCRIPT.")
    say("Etape suivante, a taper vous-meme :")
    say("   pnpm run build --filter=louvo")
    say("puis testez le jeu avant tout commit.")
    return True


try:
    ok = main()
except Exception:
    say("")
    say("===== ERREUR PYTHON =====")
    for l in traceback.format_exc().rstrip().split("\n"):
        say(l)
    ok = False

say("")
say("VERDICT : %s" % ("OK - envoyez rapport41.txt" if ok else "ARRET - envoyez rapport41.txt"))
print("")
print("FINI - envoyez rapport41.txt")
