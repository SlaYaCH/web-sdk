import os
import sys
import traceback

ROOT = os.path.dirname(os.path.abspath(__file__))
os.chdir(ROOT)

RAPPORT = "rapport69.txt"
report = []

FICHIERS = [
    "apps/louvo/src/components/FreeSpinCounter.svelte",
    "apps/louvo/src/components/BonusWinCounter.svelte",
    "apps/louvo/src/components/AfterDarkStreakDisplay.svelte",
    "apps/louvo/src/game/boardBox.ts",
]


def flush():
    try:
        with open(RAPPORT, "w") as f:
            f.write("\n".join(report) + "\n")
    except OSError:
        pass


def say(line=""):
    report.append(line)
    flush()


def main():
    for p in FICHIERS:
        print("lecture : %s" % p)
        try:
            sys.stdout.flush()
        except Exception:
            pass
        say("")
        say("==============================================================")
        try:
            with open(p, "r", errors="ignore") as f:
                txt = f.read()
        except OSError:
            say(">>> INTROUVABLE : %s" % p)
            say("==============================================================")
            continue
        say(">>> %s" % p)
        say("==============================================================")
        for i, l in enumerate(txt.split("\n"), 1):
            say("%4d| %s" % (i, l))

    # le bloc exact de Game.svelte qu'on va modifier
    say("")
    say("==============================================================")
    say(">>> Game.svelte, lignes 240 a 262 (le {#if} a modifier)")
    say("==============================================================")
    try:
        with open("apps/louvo/src/components/Game.svelte", "r", errors="ignore") as f:
            lignes = f.read().split("\n")
        for i in range(239, min(262, len(lignes))):
            say("%4d| %s" % (i + 1, lignes[i]))
    except OSError:
        say("  INTROUVABLE")
    return True


try:
    ok = main()
except Exception:
    say("")
    say("===== ERREUR PYTHON =====")
    for l in traceback.format_exc().rstrip().split("\n"):
        say(l)
    ok = False

say("")
say("VERDICT : %s (aucun fichier modifie)" % ("OK" if ok else "ARRET"))
print("")
print("FINI - aucun fichier touche. Envoyez rapport69.txt")
