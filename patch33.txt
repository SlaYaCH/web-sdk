#!/usr/bin/env python3
# ============================================================
# Louvo - lot 33 : les lignes de gains s'affichent du PLUS PETIT
# au PLUS GROS gain, a chaque tour (montee en puissance, on finit
# sur la plus belle ligne). Le meme ordre est applique a la
# surbrillance des symboles pour que les deux restent synchronises.
#
# Purement visuel : les montants et le total ne changent pas, le
# tableau du book n'est pas modifie (copie triee).
#
# Usage :
#   1) deposer ce fichier a la RACINE de web-sdk
#   2) cd /workspaces/web-sdk && python3 patch33.txt
#   3) envoyer rapport36.txt
# Build coupe automatiquement (5 min max), pas de Ctrl+C.
# ============================================================
import os
import subprocess
import sys

T = "\t"
ROOT = os.path.dirname(os.path.abspath(__file__))
os.chdir(ROOT)

BOOK = "apps/louvo/src/game/bookEventHandlerMap.ts"

files = {}
journal = []


def load(p):
    if p not in files:
        with open(p) as f:
            files[p] = f.read()


def rep(p, old, new, tag):
    load(p)
    if new and new not in old:
        applied = new in files[p]
    else:
        applied = old not in files[p]
    if applied:
        journal.append("%s : deja applique, saute" % tag)
        return
    n = files[p].count(old)
    if n != 1:
        raise AssertionError("%s ECHEC : %d occurrence(s) du texte attendu dans %s" % (tag, n, p))
    files[p] = files[p].replace(old, new, 1)
    journal.append("%s : OK" % tag)


def run_patches():
    rep(BOOK,
        "\n".join([
            T*2 + "// Instant de depart de la presentation des lignes : sert plus bas a ne",
            T*2 + "// completer que le temps qui manque VRAIMENT a la fin.",
            T*2 + "const winLinesStartedAt = performance.now();",
            T*2 + "if (bookEvent.wins.length > 0) {",
            T*3 + "// Affichee tout de suite (pas apres la surbrillance des symboles),",
            T*3 + "// directement quand le 5eme rouleau vient de s'arreter.",
            T*3 + "eventEmitter.broadcast({ type: 'winLinesShow', wins: bookEvent.wins });",
            T*2 + "}",
            T*2 + "await sequence(bookEvent.wins, async (win) => {",
            T*3 + "await animateSymbols({ positions: win.positions });",
            T*2 + "});",
        ]),
        "\n".join([
            T*2 + "// Presentation du PLUS PETIT au PLUS GROS gain : on finit sur la plus",
            T*2 + "// belle ligne du tour. Copie triee - le tableau du book n'est pas touche,",
            T*2 + "// et les montants / le total sont evidemment inchanges. Le meme ordre",
            T*2 + "// sert aussi a la surbrillance des symboles, pour rester synchronise.",
            T*2 + "const orderedWins = [...bookEvent.wins].sort((a, b) => a.win - b.win);",
            T*2 + "// Instant de depart de la presentation des lignes : sert plus bas a ne",
            T*2 + "// completer que le temps qui manque VRAIMENT a la fin.",
            T*2 + "const winLinesStartedAt = performance.now();",
            T*2 + "if (orderedWins.length > 0) {",
            T*3 + "// Affichee tout de suite (pas apres la surbrillance des symboles),",
            T*3 + "// directement quand le 5eme rouleau vient de s'arreter.",
            T*3 + "eventEmitter.broadcast({ type: 'winLinesShow', wins: orderedWins });",
            T*2 + "}",
            T*2 + "await sequence(orderedWins, async (win) => {",
            T*3 + "await animateSymbols({ positions: win.positions });",
            T*2 + "});",
        ]), "Z1")


def main():
    report = []
    try:
        run_patches()
        for p, c in files.items():
            with open(p, "w") as f:
                f.write(c)
        report.append("===== JOURNAL DES PATCHS =====")
        report.extend(journal)
        patch_failed = False
    except AssertionError as e:
        report.append("===== JOURNAL DES PATCHS (ECHEC, RIEN N'A ETE ECRIT) =====")
        report.extend(journal)
        report.append(str(e))
        patch_failed = True

    report.append("")
    report.append("===== PREUVES =====")
    all_ok = not patch_failed
    with open(BOOK) as f:
        content = f.read()
    for needle, expected in [
        ("orderedWins", 4),
        ("sort((a, b) => a.win - b.win)", 1),
    ]:
        n = content.count(needle)
        report.append("%-32s : %d (attendu %d)" % (needle, n, expected))
        if n < expected:
            all_ok = False
    report.append("----- restes (attendu 0) -----")
    for needle in [
        "winLinesShow', wins: bookEvent.wins",
        "sequence(bookEvent.wins,",
    ]:
        n = content.count(needle)
        report.append("%-32s : %d" % (needle, n))
        if n != 0:
            all_ok = False

    report.append("")
    report.append("===== winInfo final =====")
    with open(BOOK) as f:
        lines = f.readlines()
    for i, l in enumerate(lines):
        if "winInfo: async" in l:
            report.extend("%4d\t%s" % (j + 1, lines[j].rstrip("\n")) for j in range(i, min(i + 30, len(lines))))
            break

    report.append("")
    report.append("===== git status --short =====")
    try:
        out = subprocess.run(["git", "status", "--short"], capture_output=True, text=True).stdout
        report.append(out.rstrip() or "(arbre propre)")
    except OSError as e:
        report.append("git indisponible : %s" % e)

    with open("rapport36.txt", "w") as f:
        f.write("\n".join(report) + "\n")

    report.append("")
    report.append("===== BUILD =====")
    if all_ok:
        timed_out = False
        with open("build_log.txt", "w") as logf:
            try:
                code = subprocess.run(["pnpm", "run", "build", "--filter=louvo"],
                                      stdout=logf, stderr=subprocess.STDOUT, timeout=300).returncode
            except subprocess.TimeoutExpired:
                timed_out = True
                code = -1
        with open("build_log.txt") as f:
            log = f.read()
        report.extend(log.rstrip().split("\n")[-6:])
        site_ecrit = ("Wrote site to" in log) or ("done" in log)
        if timed_out:
            report.append("(build coupe apres 5 min - comportement pnpm connu)")
        report.append("site ecrit : %s / exit : %s" % ("OUI" if site_ecrit else "NON", code))
        if not site_ecrit:
            all_ok = False
    else:
        report.append("BUILD SAUTE (patchs incomplets)")

    report.append("")
    report.append("VERDICT : %s" % ("TOUT OK - commit, deploiement, tests" if all_ok else "ECHEC - envoyez ce rapport a Claude"))

    with open("rapport36.txt", "w") as f:
        f.write("\n".join(report) + "\n")
    print("\n".join(report[-4:]))
    print("FINI - envoyez rapport36.txt")
    sys.exit(0 if all_ok else 1)


main()
