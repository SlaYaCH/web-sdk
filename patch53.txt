#!/usr/bin/env python3
# ============================================================
# Louvo - lot 53 : bouton BONUS et presentoirs After Dark
#
# 1) Le bouton BONUS passe AU-DESSUS du bouton 3 traits, au lieu d'etre
#    colle a gauche de la barre ou il debordait sur petit ecran.
#
# 2) En PORTRAIT, les presentoirs et cartes After Dark ne tiennent pas a
#    cote de la grille : ils sont reduits de 28% et descendus dans
#    l'espace libre sous le cadre, sur deux rangees de deux.
#    EN PAYSAGE RIEN NE CHANGE : la branche portrait n'est prise que si
#    layoutType vaut portrait ou tablet.
#
# Tous les reglages sont en haut de AfterDarkStreakDisplay.svelte,
# commentes, pour que vous puissiez ajuster a l'oeil sans moi.
#
# SAUVEGARDES : chaque fichier modifie est copie en .avant-lot53
#
# Rapport ecrit apres chaque ligne. AUCUN BUILD.
#
# Usage :
#   1) deposer ce fichier a la RACINE de web-sdk
#   2) cd /workspaces/web-sdk && python3 patch53.txt
#   3) envoyer rapport60.txt
# ============================================================
import os
import shutil
import sys
import traceback

ROOT = os.path.dirname(os.path.abspath(__file__))
os.chdir(ROOT)

RAPPORT = "rapport60.txt"
report = []
files = {}


def flush():
    try:
        with open(RAPPORT, "w") as f:
            f.write("\n".join(report) + "\n")
    except OSError:
        pass


def say(line=""):
    report.append(line)
    print(line)
    try:
        sys.stdout.flush()
    except Exception:
        pass
    flush()


def load(p):
    if p not in files:
        s = p + ".avant-lot53"
        if not os.path.exists(s):
            shutil.copyfile(p, s)
        with open(p, "r", encoding="utf-8") as f:
            files[p] = f.read()
    return files[p]


def rep(p, old, new, tag):
    txt = load(p)
    if new in txt:
        say("  %-46s deja applique" % tag)
        return
    n = txt.count(old)
    if n != 1:
        raise AssertionError("%s : motif trouve %d fois (attendu 1) dans %s" % (tag, n, p))
    files[p] = txt.replace(old, new, 1)
    say("  %-46s OK" % tag)


def main():
    # ---------- 1. Le bouton BONUS ----------
    say("===== 1. LouvoBottomBar.svelte : BONUS au-dessus du menu =====")
    b = "apps/louvo/src/components/LouvoBottomBar.svelte"
    load(b)
    rep(b,
        "\tconst BONUS_X = -BAR_WIDTH / 2 - 45; // juste a gauche de la barre, reste dans le cadre visible",
        "\t// Le BONUS est pose AU-DESSUS du bouton 3 traits : meme colonne que\n"
        "\t// MENU_X, remonte au-dessus de la barre. Il debordait a gauche avant,\n"
        "\t// et sortait de l'ecran sur petit format.\n"
        "\tconst BONUS_X = BAR_WIDTH * 0.116 - BAR_WIDTH / 2;\n"
        "\tconst BONUS_Y_OFFSET = -118;",
        "position du bouton BONUS")
    rep(b,
        "\t\tx={BONUS_X}\n\t\ty={ROW_Y}",
        "\t\tx={BONUS_X}\n\t\ty={ROW_Y + BONUS_Y_OFFSET}",
        "remontee du bouton BONUS")

    # ---------- 2. Les presentoirs After Dark ----------
    say("")
    say("===== 2. AfterDarkStreakDisplay.svelte : portrait =====")
    a = "apps/louvo/src/components/AfterDarkStreakDisplay.svelte"
    load(a)

    rep(a,
        "\timport { getContext } from '../game/context';",
        "\timport { getContext } from '../game/context';\n"
        "\timport { boardBox } from '../game/boardBox';",
        "import boardBox")

    rep(a,
        "\tconst groupY = (i: number) =>\n"
        "\t\tmainH() * (GRID_TOP_FRAC + (GRID_BOTTOM_FRAC - GRID_TOP_FRAC) * GROUP_Y_FRACS[i]);",
        "\tconst groupY = (i: number) =>\n"
        "\t\tmainH() * (GRID_TOP_FRAC + (GRID_BOTTOM_FRAC - GRID_TOP_FRAC) * GROUP_Y_FRACS[i]);\n"
        "\n"
        "\t// ============================================================\n"
        "\t// PORTRAIT : les presentoirs ne tiennent pas a cote de la grille,\n"
        "\t// ils la chevauchent et sortent de l'ecran. On les reduit et on les\n"
        "\t// descend dans l'espace libre sous le cadre, en deux rangees de deux.\n"
        "\t// REGLAGES A AJUSTER A L'OEIL :\n"
        "\t// ============================================================\n"
        "\tconst PORTRAIT_SCALE = 0.72; // 28% plus petit\n"
        "\tconst PORTRAIT_X_FRAC = 0.26; // ecart des colonnes, en largeur d'ecran\n"
        "\tconst PORTRAIT_MARGE_HAUT = 40; // espace entre le bas du cadre et la 1re rangee\n"
        "\tconst PORTRAIT_ECART_RANGEES = 150; // ecart vertical entre les deux rangees\n"
        "\t// ============================================================\n"
        "\n"
        "\tconst estPortrait = $derived(\n"
        "\t\t['portrait', 'tablet'].includes(context.stateLayoutDerived.layoutType()),\n"
        "\t);\n"
        "\tconst box = $derived(boardBox(context.stateGame.tier));\n"
        "\n"
        "\tconst posX = (droite: boolean) =>\n"
        "\t\testPortrait\n"
        "\t\t\t? mainW() * (0.5 + (droite ? PORTRAIT_X_FRAC : -PORTRAIT_X_FRAC))\n"
        "\t\t\t: droite\n"
        "\t\t\t\t? rightX()\n"
        "\t\t\t\t: leftX();\n"
        "\tconst posY = (i: number) =>\n"
        "\t\testPortrait\n"
        "\t\t\t? box.y +\n"
        "\t\t\t\tbox.height * GRID_BOTTOM_FRAC +\n"
        "\t\t\t\tPORTRAIT_MARGE_HAUT +\n"
        "\t\t\t\ti * PORTRAIT_ECART_RANGEES\n"
        "\t\t\t: groupY(i);\n"
        "\tconst echelle = $derived(estPortrait ? PORTRAIT_SCALE : 1);",
        "placement portrait")

    for ancien, nouveau, tag in (
        ("<Container x={leftX()} y={groupY(0)}>",
         "<Container x={posX(false)} y={posY(0)} scale={echelle}>", "groupe palier 2"),
        ("<Container x={leftX()} y={groupY(1)}>",
         "<Container x={posX(false)} y={posY(1)} scale={echelle}>", "groupe palier 3"),
        ("<Container x={rightX()} y={groupY(0)}>",
         "<Container x={posX(true)} y={posY(0)} scale={echelle}>", "groupe palier 4"),
        ("<Container x={rightX()} y={groupY(1)}>",
         "<Container x={posX(true)} y={posY(1)} scale={echelle}>", "groupe palier 5x"),
    ):
        rep(a, ancien, nouveau, tag)

    for p, contenu in files.items():
        with open(p, "w", encoding="utf-8") as f:
            f.write(contenu)

    # ---------- 3. Verification ----------
    say("")
    say("===== 3. VERIFICATION =====")
    controles = (
        ("apps/louvo/src/components/LouvoBottomBar.svelte", "BONUS_Y_OFFSET", True),
        ("apps/louvo/src/components/LouvoBottomBar.svelte", "-BAR_WIDTH / 2 - 45", False),
        ("apps/louvo/src/components/AfterDarkStreakDisplay.svelte", "PORTRAIT_SCALE", True),
        ("apps/louvo/src/components/AfterDarkStreakDisplay.svelte", "posX(true)", True),
        ("apps/louvo/src/components/AfterDarkStreakDisplay.svelte", "x={leftX()}", False),
    )
    tout_ok = True
    for chemin, motif, attendu in controles:
        present = motif in open(chemin, encoding="utf-8").read()
        if present != attendu:
            tout_ok = False
        say("  %-62s %-8s %s" % (os.path.basename(chemin) + " : " + motif,
                                 "present" if present else "absent",
                                 "OK" if present == attendu else "*** ECHEC ***"))

    # ---------- 4. Fichiers pour les cadres de suspense ----------
    say("")
    say("===== 4. LES CADRES DE SUSPENSE (lecture seule) =====")
    for chemin in ("apps/louvo/src/components/Anticipation.svelte",
                   "apps/louvo/src/components/Anticipations.svelte"):
        say("")
        say("--------------------------------------------------------------")
        if not os.path.exists(chemin):
            say(">>> INTROUVABLE : %s" % chemin)
            continue
        say(">>> %s" % chemin)
        say("--------------------------------------------------------------")
        for i, l in enumerate(open(chemin, encoding="utf-8").read().split("\n"), 1):
            say("%4d| %s" % (i, l))
    return tout_ok


try:
    ok = main()
except Exception:
    say("")
    say("===== ERREUR PYTHON - AUCUN FICHIER ENREGISTRE =====")
    for l in traceback.format_exc().rstrip().split("\n"):
        say(l)
    ok = False

say("")
say("VERDICT : %s" % ("OK - build" if ok else "ECHEC - envoyez rapport60.txt"))
print("")
print("FINI - envoyez rapport60.txt")
