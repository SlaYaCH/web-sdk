#!/usr/bin/env python3
# ============================================================
# Louvo - lot 43 : RESPONSIVE des elements qui echappent au SDK
#
# createLayout() fait deja le travail : innerWidth/innerHeight (jamais
# screen.*), scale = min(canvasW/refW, canvasH/refH), applique a tout le
# conteneur par MainContainer. Les references de Louvo tombent juste aux
# 5 tailles testees par Stake.
#
# MAIS trois elements sont dessines en pixels de canvas BRUTS, sans
# passer par MainContainer, donc sans aucune mise a l'echelle :
#   - LouvoSettingsMenu   380 x 650
#   - LouvoHistoryPanel   980 x 640   <- deborde de 160% sur un 375 px
#   - LouvoBonusMenu      (dimensions inconnues, signale seulement)
#
# Et la barre du bas fait 1300 de large alors que la reference standard
# en portrait n'en fait que 1080 : elle est coupee des deux cotes.
#
# On applique donc EXACTEMENT le meme principe : un seul facteur
# d'echelle sur tout le conteneur, plafonne a 1 pour ne rien changer
# sur grand ecran.
#
# SUR DESKTOP CE PATCH NE CHANGE RIEN : tous les facteurs valent 1.
#
# Rapport ecrit apres chaque ligne. AUCUN BUILD.
#
# Usage :
#   1) deposer ce fichier a la RACINE de web-sdk
#   2) cd /workspaces/web-sdk && python3 patch43.txt
#   3) envoyer rapport49.txt
# ============================================================
import os
import sys
import traceback

ROOT = os.path.dirname(os.path.abspath(__file__))
os.chdir(ROOT)

RAPPORT = "rapport49.txt"
report = []
files = {}


def flush():
    try:
        with open(RAPPORT, "w") as f:
            f.write("\n".join(report) + "\n")
    except OSError:
        pass


def say(line=""):
    report.append(line)
    print(line)
    try:
        sys.stdout.flush()
    except Exception:
        pass
    flush()


def load(p):
    if p not in files:
        with open(p, "r", encoding="utf-8") as f:
            files[p] = f.read()
    return files[p]


def save_all():
    for p, content in files.items():
        with open(p, "w", encoding="utf-8") as f:
            f.write(content)


def rep(p, old, new, tag):
    txt = load(p)
    if new in txt:
        say("  %-46s deja applique" % tag)
        return
    n = txt.count(old)
    if n != 1:
        raise AssertionError("%s : motif trouve %d fois (attendu 1) dans %s" % (tag, n, p))
    files[p] = txt.replace(old, new, 1)
    say("  %-46s OK" % tag)


def main():
    say("===== 1. Game.svelte : facteur d'echelle des panneaux =====")
    g = "apps/louvo/src/components/Game.svelte"
    load(g)

    rep(g,
        "\tconst isReplayMode = $derived(stateUi.config.mode === 'replay');",
        "\tconst isReplayMode = $derived(stateUi.config.mode === 'replay');\n\n"
        "\t// Les panneaux ci-dessous sont poses en pixels de canvas bruts, sans\n"
        "\t// passer par MainContainer : ils ne beneficient donc pas du scale du\n"
        "\t// SDK. Sur un telephone de 375 px de large, un panneau de 980 px\n"
        "\t// deborde largement. On applique le meme principe qu'ailleurs : un\n"
        "\t// seul facteur d'echelle sur tout le conteneur, plafonne a 1 pour ne\n"
        "\t// rien changer sur grand ecran.\n"
        "\tconst overlayScale = (panelWidth: number, panelHeight: number) => {\n"
        "\t\tconst canvas = context.stateLayoutDerived.canvasSizes();\n"
        "\t\treturn Math.min(\n"
        "\t\t\t1,\n"
        "\t\t\t(canvas.width - 24) / panelWidth,\n"
        "\t\t\t(canvas.height - 24) / panelHeight,\n"
        "\t\t);\n"
        "\t};",
        "fonction overlayScale")

    rep(g,
        "\t\t\t>\n\t\t\t\t<LouvoSettingsMenu />",
        "\t\t\t\tscale={overlayScale(380, 650)}\n\t\t\t>\n\t\t\t\t<LouvoSettingsMenu />",
        "echelle du menu de reglages")

    rep(g,
        "\t\t\t>\n\t\t\t\t<LouvoHistoryPanel onclose={() => (historyOpen = false)} />",
        "\t\t\t\tscale={overlayScale(980, 640)}\n"
        "\t\t\t>\n\t\t\t\t<LouvoHistoryPanel onclose={() => (historyOpen = false)} />",
        "echelle du panneau History")

    say("")
    say("===== 2. LouvoBottomBar.svelte : barre du bas en portrait =====")
    b = "apps/louvo/src/components/LouvoBottomBar.svelte"
    load(b)
    rep(b,
        "\tconst ROW_Y = BAR_HEIGHT * 0.511 - BAR_HEIGHT / 2;",
        "\tconst ROW_Y = BAR_HEIGHT * 0.511 - BAR_HEIGHT / 2;\n\n"
        "\t// En portrait la mise en page standard ne fait que 1080 de large :\n"
        "\t// une barre de 1300 serait coupee des deux cotes. Plafonne a 1, donc\n"
        "\t// aucun changement sur desktop (1920 de large).\n"
        "\tconst barScale = $derived(\n"
        "\t\tMath.min(1, (context.stateLayoutDerived.mainLayoutStandard().width - 40) / BAR_WIDTH),\n"
        "\t);",
        "facteur barScale")
    rep(b,
        "<Container x={0} y={-BAR_HEIGHT * 0.5}>",
        "<Container x={0} y={-BAR_HEIGHT * 0.5 * barScale} scale={barScale}>",
        "application a la barre")

    say("")
    say("===== 3. LouvoReplayBar.svelte : meme traitement =====")
    r = "apps/louvo/src/components/LouvoReplayBar.svelte"
    load(r)
    rep(r,
        "\tconst BAR_HEIGHT = 150;",
        "\tconst BAR_HEIGHT = 150;\n\n"
        "\t// Meme raison que dans LouvoBottomBar : 1300 de large ne rentre pas\n"
        "\t// dans les 1080 de la mise en page standard en portrait.\n"
        "\tconst barScale = $derived(\n"
        "\t\tMath.min(1, (context.stateLayoutDerived.mainLayoutStandard().width - 40) / BAR_WIDTH),\n"
        "\t);",
        "facteur barScale")
    rep(r,
        "<Container x={0} y={-BAR_HEIGHT * 0.5}>",
        "<Container x={0} y={-BAR_HEIGHT * 0.5 * barScale} scale={barScale}>",
        "application a la barre")

    save_all()

    say("")
    say("===== 4. VERIFICATION =====")
    controles = (
        ("apps/louvo/src/components/Game.svelte", "overlayScale(380, 650)"),
        ("apps/louvo/src/components/Game.svelte", "overlayScale(980, 640)"),
        ("apps/louvo/src/components/LouvoBottomBar.svelte", "scale={barScale}"),
        ("apps/louvo/src/components/LouvoReplayBar.svelte", "scale={barScale}"),
    )
    tout_ok = True
    for chemin, motif in controles:
        present = motif in open(chemin, encoding="utf-8").read()
        if not present:
            tout_ok = False
        say("  %-58s %s" % (os.path.basename(chemin) + " : " + motif,
                            "OK" if present else "*** ABSENT ***"))

    say("")
    say("===== 5. LE MENU BONUS RESTE A REGLER =====")
    say("LouvoBonusMenu est pose lui aussi en pixels de canvas bruts, mais je")
    say("n'ai pas ses dimensions. Voici son code pour que Claude les releve :")
    p = "apps/louvo/src/components/LouvoBonusMenu.svelte"
    say("")
    say("--------------------------------------------------------------")
    if os.path.exists(p):
        say(">>> %s" % p)
        say("--------------------------------------------------------------")
        for i, l in enumerate(open(p, encoding="utf-8").read().split("\n"), 1):
            say("%4d| %s" % (i, l))
    else:
        say(">>> INTROUVABLE : %s" % p)
    return tout_ok


try:
    ok = main()
except Exception:
    say("")
    say("===== ERREUR PYTHON - AUCUN FICHIER ENREGISTRE =====")
    for l in traceback.format_exc().rstrip().split("\n"):
        say(l)
    ok = False

say("")
say("VERDICT : %s" % ("OK - lancez le build" if ok else "ECHEC - envoyez rapport49.txt"))
print("")
print("FINI - envoyez rapport49.txt")
