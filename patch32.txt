#!/usr/bin/env python3
# ============================================================
# Louvo - lot 32 (affine le lot 31) : n'ajouter a la fin que le
# temps REELLEMENT manquant.
#
# Les lignes demarrent AVANT la surbrillance des symboles
# (winLinesShow est diffuse en premier) : en vitesse normale la
# surbrillance absorbe deja tout le decalage entre les lignes.
# Le lot 31 rajoutait ce decalage une deuxieme fois -> jeu ralenti
# sans raison en vitesse normale. Ici on mesure le temps ecoule
# depuis le depart des lignes et on ne complete que le manque.
#
# Verifie sur le code reel :
#   WinLinesDisplay.svelte : STAGGER_MS = 200
#   WinLineReveal.svelte   : 120 + HOLD 700 + FADE 150
#                            + EXTRA_HOLD 300 + AMOUNT_FADE 250 = 1520
#
# Usage :
#   1) deposer ce fichier a la RACINE de web-sdk
#   2) cd /workspaces/web-sdk && python3 patch32.txt
#   3) envoyer rapport35.txt
# Build coupe automatiquement (5 min max), pas de Ctrl+C.
# ============================================================
import os
import subprocess
import sys

T = "\t"
ROOT = os.path.dirname(os.path.abspath(__file__))
os.chdir(ROOT)

BOOK = "apps/louvo/src/game/bookEventHandlerMap.ts"

files = {}
journal = []


def load(p):
    if p not in files:
        with open(p) as f:
            files[p] = f.read()


def rep(p, old, new, tag):
    load(p)
    if new and new not in old:
        applied = new in files[p]
    else:
        applied = old not in files[p]
    if applied:
        journal.append("%s : deja applique, saute" % tag)
        return
    n = files[p].count(old)
    if n != 1:
        raise AssertionError("%s ECHEC : %d occurrence(s) du texte attendu dans %s" % (tag, n, p))
    files[p] = files[p].replace(old, new, 1)
    journal.append("%s : OK" % tag)


def run_patches():
    # Y1. horodatage du depart de la presentation des lignes
    rep(BOOK,
        "\n".join([
            T*2 + "if (bookEvent.wins.length > 0) {",
            T*3 + "// Affichee tout de suite (pas apres la surbrillance des symboles),",
            T*3 + "// directement quand le 5eme rouleau vient de s'arreter.",
            T*3 + "eventEmitter.broadcast({ type: 'winLinesShow', wins: bookEvent.wins });",
            T*2 + "}",
        ]),
        "\n".join([
            T*2 + "// Instant de depart de la presentation des lignes : sert plus bas a ne",
            T*2 + "// completer que le temps qui manque VRAIMENT a la fin.",
            T*2 + "const winLinesStartedAt = performance.now();",
            T*2 + "if (bookEvent.wins.length > 0) {",
            T*3 + "// Affichee tout de suite (pas apres la surbrillance des symboles),",
            T*3 + "// directement quand le 5eme rouleau vient de s'arreter.",
            T*3 + "eventEmitter.broadcast({ type: 'winLinesShow', wins: bookEvent.wins });",
            T*2 + "}",
        ]), "Y1")

    # Y2. attente finale : pause d'origine + seulement le decalage non absorbe
    rep(BOOK,
        "\n".join([
            T*3 + "const WIN_LINE_DURATION = 120 + 700 + 150 + 300 + 250;",
            T*3 + "const WIN_LINE_STAGGER = 200;",
            T*3 + "await new Promise((r) =>",
            T*4 + "setTimeout(",
            T*5 + "r,",
            T*5 + "WIN_LINE_DURATION + WIN_LINE_STAGGER * Math.max(0, bookEvent.wins.length - 1),",
            T*4 + "),",
            T*3 + ");",
        ]),
        "\n".join([
            T*3 + "// Valeurs verifiees dans le code : WinLineReveal = 120 + 700 + 150 + 300",
            T*3 + "// + 250 par ligne, WinLinesDisplay = 200 ms entre deux lignes. La derniere",
            T*3 + "// ligne se termine donc WIN_LINE_STAGGER x (nb lignes - 1) apres la",
            T*3 + "// premiere. Les lignes ayant demarre AVANT la surbrillance des symboles,",
            T*3 + "// celle-ci absorbe deja une partie de ce decalage : on garde la pause",
            T*3 + "// d'origine et on ne rajoute QUE le manque. Vitesse normale : attente",
            T*3 + "// inchangee. Turbo/super turbo (surbrillance quasi instantanee) : allongee",
            T*3 + "// juste assez pour que la derniere ligne finisse avant le tour suivant.",
            T*3 + "const WIN_LINE_DURATION = 120 + 700 + 150 + 300 + 250;",
            T*3 + "const WIN_LINE_STAGGER = 200;",
            T*3 + "const winLinesElapsed = performance.now() - winLinesStartedAt;",
            T*3 + "const staggerLeft = Math.max(",
            T*4 + "0,",
            T*4 + "WIN_LINE_STAGGER * (bookEvent.wins.length - 1) - winLinesElapsed,",
            T*3 + ");",
            T*3 + "await new Promise((r) => setTimeout(r, WIN_LINE_DURATION + staggerLeft));",
        ]), "Y2")


def main():
    report = []
    try:
        run_patches()
        for p, c in files.items():
            with open(p, "w") as f:
                f.write(c)
        report.append("===== JOURNAL DES PATCHS =====")
        report.extend(journal)
        patch_failed = False
    except AssertionError as e:
        report.append("===== JOURNAL DES PATCHS (ECHEC, RIEN N'A ETE ECRIT) =====")
        report.extend(journal)
        report.append(str(e))
        patch_failed = True

    report.append("")
    report.append("===== PREUVES =====")
    all_ok = not patch_failed
    with open(BOOK) as f:
        content = f.read()
    for needle, expected in [
        ("winLinesStartedAt", 2),
        ("winLinesElapsed", 2),
        ("staggerLeft", 2),
    ]:
        n = content.count(needle)
        report.append("%-22s : %d (attendu %d)" % (needle, n, expected))
        if n < expected:
            all_ok = False
    report.append("----- restes (attendu 0) -----")
    n = content.count("WIN_LINE_STAGGER * Math.max(0, bookEvent.wins.length - 1)")
    report.append("%-22s : %d" % ("attente lot 31", n))
    if n != 0:
        all_ok = False

    report.append("")
    report.append("===== winInfo final =====")
    with open(BOOK) as f:
        lines = f.readlines()
    for i, l in enumerate(lines):
        if "winInfo: async" in l:
            report.extend("%4d\t%s" % (j + 1, lines[j].rstrip("\n")) for j in range(i, min(i + 40, len(lines))))
            break

    report.append("")
    report.append("===== git status --short =====")
    try:
        out = subprocess.run(["git", "status", "--short"], capture_output=True, text=True).stdout
        report.append(out.rstrip() or "(arbre propre)")
    except OSError as e:
        report.append("git indisponible : %s" % e)

    with open("rapport35.txt", "w") as f:
        f.write("\n".join(report) + "\n")

    report.append("")
    report.append("===== BUILD =====")
    if all_ok:
        timed_out = False
        with open("build_log.txt", "w") as logf:
            try:
                code = subprocess.run(["pnpm", "run", "build", "--filter=louvo"],
                                      stdout=logf, stderr=subprocess.STDOUT, timeout=300).returncode
            except subprocess.TimeoutExpired:
                timed_out = True
                code = -1
        with open("build_log.txt") as f:
            log = f.read()
        report.extend(log.rstrip().split("\n")[-6:])
        site_ecrit = ("Wrote site to" in log) or ("done" in log)
        if timed_out:
            report.append("(build coupe apres 5 min - comportement pnpm connu)")
        report.append("site ecrit : %s / exit : %s" % ("OUI" if site_ecrit else "NON", code))
        if not site_ecrit:
            all_ok = False
    else:
        report.append("BUILD SAUTE (patchs incomplets)")

    report.append("")
    report.append("VERDICT : %s" % ("TOUT OK - commit, deploiement, tests" if all_ok else "ECHEC - envoyez ce rapport a Claude"))

    with open("rapport35.txt", "w") as f:
        f.write("\n".join(report) + "\n")
    print("\n".join(report[-4:]))
    print("FINI - envoyez rapport35.txt")
    sys.exit(0 if all_ok else 1)


main()
