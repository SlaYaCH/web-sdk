# ============================================================
#   1. Stake Engine Loader          (interdit dans le jeu final)
#   2. Barre d'espace = bouton mise

#   4. Confirmation avant achat d'un mode couteux
#   5. Le panneau de regles : RTP, max win, paytable, modes,
#      free spins, retrigger, disclaimer, guide d'interaction
#   6. app.html : double-tap zoom, viewport
#   7. Paliers de mise : lus depuis authenticate ou codes en dur
#   8. end-round sur un tour sans gain
#   9. Monnaies non standard (SC / GC de Stake.US)

#   cd /workspaces/web-sdk && python3 audit_soumission.txt

import os
import re
import sys
import traceback

ROOT = os.path.dirname(os.path.abspath(__file__))
os.chdir(ROOT)

RAPPORT = "rapport_soumission.txt"
report = []


def flush():
    try:
        with open(RAPPORT, "w") as f:
            f.write("\n".join(report) + "\n")
    except OSError:
        pass


def say(line=""):
    report.append(line)
    print(line)
    try:
        sys.stdout.flush()
    except Exception:
        pass
    flush()


def lire(p):
    try:
        with open(p, "r", encoding="utf-8", errors="ignore") as f:
            return f.read()
    except OSError:
        return None


def fichiers(racines, exts=(".ts", ".svelte", ".html")):
    out = []
    for r in racines:
        if not os.path.isdir(r):
            continue
        for dossier, sousdossiers, noms in os.walk(r):
            sousdossiers[:] = [d for d in sousdossiers
                               if d not in ("node_modules", ".svelte-kit", "dist",
                                            "build", ".turbo")]
            for n in noms:
                if n.endswith(exts):
                    out.append(os.path.join(dossier, n))
    return out


LOUVO = ["apps/louvo/src"]
PARTAGE = ["packages/components-shared/src", "packages/state-shared/src",
           "packages/utils-shared", "packages/pixi-svelte/src"]


def chercher(motifs, racines, maxi=8, ignorer_commentaires=True):
    """Renvoie [(chemin, no_ligne, texte)] pour les lignes qui contiennent
    l'un des motifs. Les commentaires sont ecartes par defaut : compter un
    mot present uniquement dans un commentaire est l'erreur que j'ai deja
    commise six fois sur ce projet."""
    trouves = []
    for f in fichiers(racines):
        txt = lire(f)
        if txt is None:
            continue
        for i, ligne in enumerate(txt.split("\n"), 1):
            nue = ligne.strip()
            if ignorer_commentaires and (nue.startswith("//")
                                         or nue.startswith("*")
                                         or nue.startswith("/*")
                                         or nue.startswith("<!--")):
                continue
            for m in motifs:
                if m in ligne:
                    trouves.append((f, i, nue[:110]))
                    break
            if len(trouves) >= maxi:
                return trouves
    return trouves


def bloc(titre, critere):
    say("")
    say("=" * 66)
    say("  %s" % titre)
    say("  critere Stake : %s" % critere)
    say("=" * 66)


def resultat(trouves, si_vide, si_trouve):
    if trouves:
        say("  >>> %s" % si_trouve)
        for f, i, t in trouves:
            say("      %s:%d" % (f.replace("apps/louvo/src/", "louvo/"), i))
            say("          %s" % t)
    else:
        say("  >>> %s" % si_vide)
    return bool(trouves)


def main():
    say("===== AUDIT DE SOUMISSION LOUVO =====")
    say("")
    say("  Lecture seule. Aucun fichier modifie, aucun build.")
    say("  Les commentaires sont ecartes des recherches.")

    manquants = []
    for d in LOUVO + PARTAGE:
        if not os.path.isdir(d):
            manquants.append(d)
    if manquants:
        say("")
        say("  ATTENTION - dossiers introuvables : %s" % ", ".join(manquants))
        say("  Lancez le script depuis /workspaces/web-sdk.")
        if "apps/louvo/src" in manquants:
            return False

    # ------------------------------------------------------------------
    bloc("1. STAKE ENGINE LOADER",
         "Game should not contain the Stake Engine Loader")
    t = chercher(["StakeEngineLoader", "stake-engine-loader", "EngineLoader",
                  "stakeEngineLoader"], LOUVO + PARTAGE)
    resultat(t,
             "aucune trace du loader Stake Engine dans le code du jeu",
             "TRACES TROUVEES - a examiner :")

    # ------------------------------------------------------------------
    bloc("2. BARRE D'ESPACE",
         "Space bar should be bound to the bet button")
    t = chercher(["'Space'", '"Space"', "keyCode === 32", "code === 'Space'",
                  "key === ' '", "onkeydown", "addEventListener('keydown'",
                  'addEventListener("keydown"'], LOUVO + PARTAGE, maxi=12)
    resultat(t,
             "AUCUN gestionnaire clavier trouve - la barre d'espace n'est",
             "gestionnaires clavier trouves :")
    if not t:
        say("      probablement pas liee au bouton de mise.")
        say("      Louvo a sa propre LouvoBottomBar : si le SDK liait")
        say("      l'espace dans son UI standard, Louvo ne l'herite pas.")

    # ------------------------------------------------------------------
    bloc("3. CONFIRMATION AVANT AUTOPLAY",
         "Auto-bet requires a confirmation step before starting")
    t = chercher(["autoplay", "autoSpin", "autoBet", "stateAutoBet",
                  "AutoBet", "Autoplay"], LOUVO, maxi=14)
    resultat(t, "aucun code d'autoplay trouve dans Louvo",
             "code d'autoplay :")
    t2 = chercher(["confirm", "Confirm", "AreYouSure", "areYouSure"],
                  LOUVO, maxi=10)
    say("")
    resultat(t2,
             "AUCUNE confirmation trouvee nulle part dans Louvo",
             "elements de confirmation existants :")

    # ------------------------------------------------------------------
    bloc("4. CONFIRMATION AVANT ACHAT D'UN MODE COUTEUX",
         "High cost bet modes require confirmation before activation")
    t = chercher(["BONUS_AFTER_DARK", "bonus_after_dark", "buyBonus",
                  "onBuy", "BuyBonus"], LOUVO, maxi=12)
    resultat(t, "aucun code d'achat trouve", "code d'achat des modes :")

    # ------------------------------------------------------------------
    bloc("5. LE PANNEAU DE REGLES",
         "RTP, max win, paytable, modes, free spins, retrigger, "
         "disclaimer, guide")
    g = lire("apps/louvo/src/components/Game.svelte")
    if g is None:
        say("  Game.svelte introuvable")
    else:
        attendus = [
            ("RTP annonce", ["96.5", "96,5", "RTP"]),
            ("max win annonce", ["15000", "15,000", "15 000"]),
            ("paytable / symboles", ["assets.H1", "PAYTABLE", "Paytable",
                                     "paytable"]),
            ("cout des modes", ["150x", "80x", "60x", "3x"]),
            ("free spins", ["FREE SPIN", "Free Spin", "free spin",
                            "SPEED DATING", "AFTER DARK"]),
            ("retrigger", ["RETRIGGER", "Retrigger", "retrigger",
                           "+5", "additional"]),
            ("wilds coupes", ["WILD-ONLY", "wild-only", "WILD ONLY"]),
            ("disclaimer general", ["MALFUNCTION", "Malfunction",
                                    "malfunction", "DISCLAIMER",
                                    "Disclaimer", "voids all pays",
                                    "void all pays"]),
            ("guide d'interaction", ["HOW TO PLAY", "How to play",
                                     "HOW TO", "INTERACTION", "CONTROLS",
                                     "Controls", "SPACE"]),
        ]
        for nom, motifs in attendus:
            present = [m for m in motifs if m in g]
            say("  %-24s %s%s"
                % (nom,
                   "PRESENT" if present else "*** ABSENT ***",
                   ("   (%s)" % ", ".join(present[:3])) if present else ""))

    # ------------------------------------------------------------------
    bloc("6. APP.HTML : ZOOM ET VIEWPORT",
         "Double tap to zoom is disabled on mobile")
    trouve_html = False
    for chemin in ("apps/louvo/src/app.html", "apps/louvo/app.html",
                   "apps/louvo/static/app.html"):
        h = lire(chemin)
        if h is None:
            continue
        trouve_html = True
        say("  %s :" % chemin)
        for cle, libelle in (("user-scalable", "user-scalable"),
                             ("maximum-scale", "maximum-scale"),
                             ("touch-action", "touch-action"),
                             ("viewport", "balise viewport")):
            say("      %-16s %s" % (libelle,
                                    "present" if cle in h else "*** ABSENT ***"))
        for ligne in h.split("\n"):
            if "viewport" in ligne:
                say("      -> %s" % ligne.strip()[:120])
        break
    if not trouve_html:
        say("  app.html introuvable aux emplacements habituels")
    t = chercher(["touch-action", "touchAction"], LOUVO)
    say("")
    resultat(t, "aucun touch-action dans le code du jeu",
             "touch-action trouve dans le code :")

    # ------------------------------------------------------------------
    bloc("7. PALIERS DE MISE",
         "Game dynamically uses all betting parameters from the "
         "authenticate response")
    t = chercher(["betLevels", "betModes", "minBet", "maxBet",
                  "stateBet.betAmount =", "betAmounts"],
                 LOUVO + PARTAGE, maxi=12)
    resultat(t, "aucune reference aux paliers de mise trouvee",
             "references aux paliers de mise :")
    if g:
        say("")
        for motif in ("0.01", "0,01", "1000", "0.10", "2000"):
            if motif in g:
                say("      Game.svelte contient le nombre %s "
                    "(a verifier : le panneau de regles annonce-t-il une"
                    % motif)
                say("      plage de mise en dur ? elle varie selon la monnaie)")
                break

    # ------------------------------------------------------------------
    bloc("8. END-ROUND SUR UN TOUR SANS GAIN",
         "Zero-win bets do not send an end-round request to the RGS")
    t = chercher(["end-round", "endRound", "requestEndRound"],
                 LOUVO + PARTAGE + ["packages/rgs-requests/src"], maxi=10)
    resultat(t, "aucun appel end-round trouve", "appels end-round :")

    # ------------------------------------------------------------------
    bloc("9. MONNAIES NON STANDARD (Stake.US : SC / GC)",
         "Game supports SC and GC currencies & values do not display "
         "a \"$\" prefix")
    a = lire("packages/utils-shared/amount.ts")
    if a is None:
        say("  packages/utils-shared/amount.ts introuvable")
    else:
        say("  amount.ts :")
        say("      try/catch sur Intl        %s"
            % ("present" if "catch" in a else "*** ABSENT ***"))
        say("      mention SC/GC             %s"
            % ("presente" if ("'SC'" in a or '"SC"' in a) else "absente"))
    t = chercher(["'$'", '"$"', "US$", "$ +", "+ '$'"], LOUVO, maxi=8)
    say("")
    resultat(t, "aucun prefixe $ code en dur dans Louvo",
             "PREFIXES $ CODES EN DUR - a examiner :")

    say("")
    say("=" * 66)
    say("  FIN DE L'AUDIT")
    say("=" * 66)
    say("")
    say("  Ce rapport ne conclut rien tout seul : envoyez-le-moi et je")
    say("  le croise avec la Submission Checklist officielle, ligne par")
    say("  ligne.")
    return True


try:
    ok = main()
except Exception:
    say("")
    say("===== ERREUR PYTHON =====")
    for l in traceback.format_exc().rstrip().split("\n"):
        say(l)
    ok = False

say("")
print("")
print("FINI - envoyez rapport_soumission.txt")
sys.exit(0 if ok else 1)
