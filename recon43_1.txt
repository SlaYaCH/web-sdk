import os
import re
import sys
import traceback

ROOT = os.path.dirname(os.path.abspath(__file__))
os.chdir(ROOT)

RAPPORT = "rapport82.txt"
report = []

IGNORE = {"node_modules", "dist", ".turbo", "build", ".svelte-kit", ".git",
          "_demo_assets_backup", "_nouveaux_cadres", "static", "stories"}

FICHIERS = [
    "apps/louvo/src/components/WinLinesDisplay.svelte",
    "apps/louvo/src/components/WinLineReveal.svelte",
    "apps/louvo/src/components/Anticipations.svelte",
    "apps/louvo/src/components/Anticipation.svelte",
    "apps/louvo/src/components/SpecialRevealOverlay.svelte",
    "apps/louvo/src/components/Win.svelte",
    "apps/louvo/src/game/constants.ts",
    "apps/louvo/src/game/stateGame.svelte.ts",
    "apps/louvo/src/game/bookEventHandlerMap.ts",
]


def flush():
    try:
        with open(RAPPORT, "w") as f:
            f.write("\n".join(report) + "\n")
    except OSError:
        pass


def say(line=""):
    report.append(line)
    flush()


def marche(*a):
    print(*a)
    try:
        sys.stdout.flush()
    except Exception:
        pass


def lire(p):
    try:
        with open(p, "r", errors="ignore") as f:
            return f.read()
    except OSError:
        return None


def dump(p, max_lignes=500):
    txt = lire(p)
    say("")
    say("--------------------------------------------------------------")
    if txt is None:
        say(">>> INTROUVABLE : %s" % p)
        return
    say(">>> %s" % p)
    say("--------------------------------------------------------------")
    lignes = txt.split("\n")
    for i, l in enumerate(lignes[:max_lignes], 1):
        say("%4d| %s" % (i, l))
    if len(lignes) > max_lignes:
        say("... (%d lignes de plus, tronque)" % (len(lignes) - max_lignes))


def parcours(bases):
    for base in bases:
        if not os.path.isdir(base):
            continue
        for dirpath, dirnames, filenames in os.walk(base):
            dirnames[:] = [d for d in dirnames if d not in IGNORE]
            for fn in sorted(filenames):
                if fn.endswith((".ts", ".svelte")) and not fn.endswith(".d.ts"):
                    yield os.path.join(dirpath, fn)


BASES = ["apps/louvo/src", "packages/state-shared/src", "packages/components-shared/src",
         "packages/utils-book/src", "packages/constants-shared/src"]


def cherche(titre, motif, bases=BASES, limite=200):
    say("")
    say("--- %s ---" % titre)
    rx = re.compile(motif)
    n = 0
    for p in parcours(bases):
        txt = lire(p) or ""
        for i, l in enumerate(txt.split("\n"), 1):
            if rx.search(l):
                say("%s:%d: %s" % (p, i, l.strip()[:150]))
                n += 1
                if n >= limite:
                    say("   ... (arrete a %d)" % limite)
                    return
    if n == 0:
        say("   (rien)")


def main():
    # ---------- 1. fichiers ----------
    marche("1/3 fichiers...")
    say("===== 1. FICHIERS OUVERTS EN ENTIER =====")
    for p in FICHIERS:
        dump(p)

    # ---------- 2. turbo et durees ----------
    marche("2/3 turbo et durees...")
    say("")
    say("===== 2. TURBO ET DUREES =====")
    cherche("le turbo, partout ou il est lu ou ecrit", r"\bturbo\b|isTurbo|Turbo")
    cherche("durees des lignes de gain", r"winLine|WIN_LINE|winline", limite=120)
    cherche("constantes de duree", r"SECOND\b|duration|DURATION|delay|DELAY|timeout|setTimeout")

    # ---------- 3. niveau de gain et anticipations ----------
    marche("3/3 niveau de gain et anticipations...")
    say("")
    say("===== 3. NIVEAU DE GAIN =====")
    cherche("niveaux de gain (big win, super win...)",
            r"winLevel|WIN_LEVEL|bigWin|BIG_WIN|superWin|SUPER_WIN|megaWin|maxWin", limite=120)

    say("")
    say("===== 4. ANTICIPATIONS ET BANNIERES =====")
    cherche("anticipation", r"anticipat|Anticipat|ANTICIPAT", limite=120)
    cherche("bannieres et reveals speciaux",
            r"banner|Banner|BANNER|specialReveal|SpecialReveal|reelIndex", limite=120)
    return True


try:
    ok = main()
except Exception:
    say("")
    say("===== ERREUR PYTHON =====")
    for l in traceback.format_exc().rstrip().split("\n"):
        say(l)
    ok = False

say("")
say("VERDICT : %s (aucun fichier modifie)" % ("OK" if ok else "ARRET"))
print("")
print("FINI - envoyez rapport82.txt")
sys.exit(0 if ok else 1)
