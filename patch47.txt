#!/usr/bin/env python3
# ============================================================
# Louvo - lot 47 : nom du jeu et logo a l'echelle
#
# LE PROBLEME (visible sur la capture Popout S 400x225) :
#   Dans Game.svelte, <UiGameName /> et le logo louvoLogo sont poses en
#   pixels de canvas BRUTS, hors MainContainer. Un logo de 100 px fait 8%
#   de la largeur sur un ecran de 1200, mais 25% sur un ecran de 400 :
#   il devore l'ecran et le nom du jeu chevauche l'enseigne LOUVO.
#
# LA CORRECTION :
#   Meme principe que les panneaux (lots 43-44) : un facteur d'echelle
#   proportionnel a la largeur du canvas, plafonne a 1.
#     1200 px -> 1.00 (aucun changement)
#      800 px -> 0.89
#      400 px -> 0.44
#
# SUR DESKTOP CE PATCH NE CHANGE RIEN.
#
# Rapport ecrit apres chaque ligne. AUCUN BUILD.
#
# Usage :
#   1) deposer ce fichier a la RACINE de web-sdk
#   2) cd /workspaces/web-sdk && python3 patch47.txt
#   3) envoyer rapport53.txt
# ============================================================
import os
import sys
import traceback

ROOT = os.path.dirname(os.path.abspath(__file__))
os.chdir(ROOT)

RAPPORT = "rapport53.txt"
report = []
files = {}


def flush():
    try:
        with open(RAPPORT, "w") as f:
            f.write("\n".join(report) + "\n")
    except OSError:
        pass


def say(line=""):
    report.append(line)
    print(line)
    try:
        sys.stdout.flush()
    except Exception:
        pass
    flush()


def load(p):
    if p not in files:
        with open(p, "r", encoding="utf-8") as f:
            files[p] = f.read()
    return files[p]


def rep(p, old, new, tag):
    txt = load(p)
    if new in txt:
        say("  %-46s deja applique" % tag)
        return
    n = txt.count(old)
    if n != 1:
        raise AssertionError("%s : motif trouve %d fois (attendu 1) dans %s" % (tag, n, p))
    files[p] = txt.replace(old, new, 1)
    say("  %-46s OK" % tag)


def main():
    g = "apps/louvo/src/components/Game.svelte"
    load(g)

    say("===== 1. Game.svelte : facteur hudScale =====")
    rep(g,
        "\t\t\t(canvas.height - 24) / panelHeight,\n\t\t);\n\t};",
        "\t\t\t(canvas.height - 24) / panelHeight,\n\t\t);\n\t};\n\n"
        "\t// Le nom du jeu et le logo sont eux aussi poses en pixels de canvas\n"
        "\t// bruts, hors MainContainer. Sur un petit ecran ils prennent une place\n"
        "\t// enorme et chevauchent l'enseigne LOUVO. Plafonne a 1 : aucun\n"
        "\t// changement au-dela de 900 px de large.\n"
        "\tconst hudScale = $derived(\n"
        "\t\tMath.min(1, context.stateLayoutDerived.canvasSizes().width / 900),\n"
        "\t);",
        "definition de hudScale")

    say("")
    say("===== 2. Application au nom du jeu et au logo =====")
    rep(g,
        "\t\t<Container x={20}>\n\t\t\t<UiGameName name=\"LOUVO\" />",
        "\t\t<Container x={20} scale={hudScale}>\n\t\t\t<UiGameName name=\"LOUVO\" />",
        "nom du jeu")
    rep(g,
        "\t\t<Container x={context.stateLayoutDerived.canvasSizes().width - 20}>\n"
        "\t\t\t<Sprite key=\"louvoLogo\"",
        "\t\t<Container x={context.stateLayoutDerived.canvasSizes().width - 20} scale={hudScale}>\n"
        "\t\t\t<Sprite key=\"louvoLogo\"",
        "logo")

    for p, content in files.items():
        with open(p, "w", encoding="utf-8") as f:
            f.write(content)

    say("")
    say("===== 3. VERIFICATION =====")
    apres = open(g, encoding="utf-8").read()
    tout_ok = True
    for motif in ("const hudScale", "x={20} scale={hudScale}", "- 20} scale={hudScale}"):
        present = motif in apres
        if not present:
            tout_ok = False
        say("  %-40s %s" % (motif, "OK" if present else "*** ABSENT ***"))

    say("")
    say("===== 4. POUR LA DECISION PORTRAIT =====")
    p = "apps/louvo/src/game/stateLayout.ts"
    say("")
    say(">>> %s" % p)
    for i, l in enumerate(open(p, encoding="utf-8").read().split("\n"), 1):
        say("%4d| %s" % (i, l))

    say("")
    say("--- ce que donnent les references actuelles ---")
    refs = {"desktop": (1422, 800), "tablet": (1000, 1000),
            "landscape": (1600, 900), "portrait": (800, 1422)}
    ecrans = [("Desktop", 1200, 675), ("Laptop", 1024, 576), ("Popout S", 400, 225),
              ("Popout L", 800, 450), ("Mobile L", 425, 812), ("Mobile M", 375, 667),
              ("Mobile S", 320, 568)]
    say("%-12s %-11s %-11s %8s   %s" % ("ecran", "ratio", "type", "echelle", "rendu"))
    for nom, w, h in ecrans:
        ratio = w / float(h)
        if 0.8 < ratio < 1.3:
            t = "tablet"
        elif ratio <= 0.8:
            t = "portrait"
        elif min(w, h) <= 480:
            t = "landscape"
        else:
            t = "desktop"
        rw, rh = refs[t]
        s = min(w / float(rw), h / float(rh))
        say("%-12s %-11s %-11s %8.3f   %d x %d" % (nom, "%.3f" % ratio, t, s, rw * s, rh * s))
    say("")
    say("Le decor et le cadre sont dessines en 1672x941 (ratio 1.777).")
    say("La reference portrait 800x1422 a un ratio de 0.563 : le cadre y est")
    say("etire, et les GRID_*_FRAC mesures sur l'image paysage n'y tombent plus")
    say("au bon endroit. C'est la cause des rangees vides sur les captures.")
    return tout_ok


try:
    ok = main()
except Exception:
    say("")
    say("===== ERREUR PYTHON - AUCUN FICHIER ENREGISTRE =====")
    for l in traceback.format_exc().rstrip().split("\n"):
        say(l)
    ok = False

say("")
say("VERDICT : %s" % ("OK - build" if ok else "ECHEC - envoyez rapport53.txt"))
print("")
print("FINI - envoyez rapport53.txt")
