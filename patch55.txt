#!/usr/bin/env python3
# ============================================================
# Louvo - lot 55 : deux reglages + lecture des lignes de gain
#
# 1) Cadre de suspense allonge : 1.00 -> 1.06 de la hauteur de grille.
# 2) Presentoirs After Dark en portrait descendus : la marge sous le
#    cadre passe de 40 a 130 unites, ce qui les centre dans l'espace
#    libre au lieu de les coller au bas de la grille.
# 3) Ouvre WinLinesDisplay.svelte et WinLineReveal.svelte : le decalage
#    de 0.8 case des lignes de gain se lit forcement la-dedans.
#
# SAUVEGARDES en .avant-lot55
# Rapport ecrit apres chaque ligne. AUCUN BUILD.
#
# Usage :
#   1) deposer ce fichier a la RACINE de web-sdk
#   2) cd /workspaces/web-sdk && python3 patch55.txt
#   3) envoyer rapport62.txt
# ============================================================
import os
import shutil
import sys
import traceback

ROOT = os.path.dirname(os.path.abspath(__file__))
os.chdir(ROOT)

RAPPORT = "rapport62.txt"
report = []
files = {}


def flush():
    try:
        with open(RAPPORT, "w") as f:
            f.write("\n".join(report) + "\n")
    except OSError:
        pass


def say(line=""):
    report.append(line)
    print(line)
    try:
        sys.stdout.flush()
    except Exception:
        pass
    flush()


def load(p):
    if p not in files:
        s = p + ".avant-lot55"
        if not os.path.exists(s):
            shutil.copyfile(p, s)
        with open(p, "r", encoding="utf-8") as f:
            files[p] = f.read()
    return files[p]


def rep(p, old, new, tag):
    txt = load(p)
    if new in txt:
        say("  %-46s deja applique" % tag)
        return
    n = txt.count(old)
    if n != 1:
        raise AssertionError("%s : motif trouve %d fois (attendu 1) dans %s" % (tag, n, p))
    files[p] = txt.replace(old, new, 1)
    say("  %-46s OK" % tag)


def main():
    say("===== 1. Cadre de suspense allonge =====")
    rep("apps/louvo/src/components/Anticipation.svelte",
        "\tconst ANTICIPATION_HAUTEUR = 1.0;",
        "\tconst ANTICIPATION_HAUTEUR = 1.06;",
        "hauteur 1.00 -> 1.06")

    say("")
    say("===== 2. Presentoirs After Dark descendus (portrait) =====")
    rep("apps/louvo/src/components/AfterDarkStreakDisplay.svelte",
        "\tconst PORTRAIT_MARGE_HAUT = 40; // espace entre le bas du cadre et la 1re rangee",
        "\tconst PORTRAIT_MARGE_HAUT = 130; // espace entre le bas du cadre et la 1re rangee",
        "marge 40 -> 130")

    for chemin, contenu in files.items():
        with open(chemin, "w", encoding="utf-8") as f:
            f.write(contenu)

    say("")
    say("===== 3. VERIFICATION =====")
    tout_ok = True
    for chemin, motif in (
        ("apps/louvo/src/components/Anticipation.svelte", "ANTICIPATION_HAUTEUR = 1.06"),
        ("apps/louvo/src/components/AfterDarkStreakDisplay.svelte", "PORTRAIT_MARGE_HAUT = 130"),
    ):
        present = motif in open(chemin, encoding="utf-8").read()
        if not present:
            tout_ok = False
        say("  %-62s %s" % (os.path.basename(chemin) + " : " + motif,
                            "OK" if present else "*** ABSENT ***"))

    say("")
    say("===== 4. LES LIGNES DE GAIN (lecture seule) =====")
    for chemin in ("apps/louvo/src/components/WinLinesDisplay.svelte",
                   "apps/louvo/src/components/WinLineReveal.svelte",
                   "apps/louvo/src/components/SymbolWrap.svelte",
                   "apps/louvo/src/components/ReelSymbol.svelte"):
        say("")
        say("--------------------------------------------------------------")
        if not os.path.exists(chemin):
            say(">>> INTROUVABLE : %s" % chemin)
            continue
        say(">>> %s" % chemin)
        say("--------------------------------------------------------------")
        for i, l in enumerate(open(chemin, encoding="utf-8").read().split("\n"), 1):
            say("%4d| %s" % (i, l))

    say("")
    say("--- ou getSymbolX / getSymbolY sont utilises ---")
    import re
    for dirpath, dirnames, filenames in os.walk("apps/louvo/src"):
        dirnames[:] = [d for d in dirnames if d not in ("node_modules", "build", ".svelte-kit")]
        for fn in sorted(filenames):
            if not fn.endswith((".svelte", ".ts")):
                continue
            fp = os.path.join(dirpath, fn)
            try:
                contenu = open(fp, encoding="utf-8", errors="ignore").read()
            except OSError:
                continue
            for i, l in enumerate(contenu.split("\n"), 1):
                if re.search(r"getSymbolX|getSymbolY|SYMBOL_HEIGHT|boardLayout\(\)", l):
                    say("%s:%d: %s" % (fp, i, l.strip()[:150]))
    return tout_ok


try:
    ok = main()
except Exception:
    say("")
    say("===== ERREUR PYTHON - AUCUN FICHIER ENREGISTRE =====")
    for l in traceback.format_exc().rstrip().split("\n"):
        say(l)
    ok = False

say("")
say("VERDICT : %s" % ("OK - build" if ok else "ECHEC - envoyez rapport62.txt"))
print("")
print("FINI - envoyez rapport62.txt")
