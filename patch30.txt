#!/usr/bin/env python3
# ============================================================
# Louvo - lot 30 : turbo et super turbo DESACTIVES pendant les
# bonus (Speed Dating / After Dark). L'entree de bonus les
# coupait deja (freeSpinTrigger) ; desormais CHAQUE tour de
# free spins les force a OFF - un reglage reactive via le menu
# en cours de bonus est coupe au tour suivant. Le menu reglages
# n'est pas touche.
#
# Usage :
#   1) deposer ce fichier a la RACINE de web-sdk
#   2) cd /workspaces/web-sdk && python3 patch30.txt
#   3) envoyer rapport33.txt
# Build coupe automatiquement (5 min max), pas de Ctrl+C.
# ============================================================
import os
import subprocess
import sys

T = "\t"
ROOT = os.path.dirname(os.path.abspath(__file__))
os.chdir(ROOT)

BOOK = "apps/louvo/src/game/bookEventHandlerMap.ts"

files = {}
journal = []


def load(p):
    if p not in files:
        with open(p) as f:
            files[p] = f.read()


def rep(p, old, new, tag):
    load(p)
    if new and new not in old:
        applied = new in files[p]
    else:
        applied = old not in files[p]
    if applied:
        journal.append("%s : deja applique, saute" % tag)
        return
    n = files[p].count(old)
    if n != 1:
        raise AssertionError("%s ECHEC : %d occurrence(s) du texte attendu dans %s" % (tag, n, p))
    files[p] = files[p].replace(old, new, 1)
    journal.append("%s : OK" % tag)


def run_patches():
    rep(BOOK,
        "\n".join([
            T*2 + "stateGame.gameType = bookEvent.gameType;",
            T*2 + "await stateGameDerived.enhancedBoard.spin({",
        ]),
        "\n".join([
            T*2 + "// Turbo et super turbo DESACTIVES pendant les bonus : aucun interet en",
            T*2 + "// free spins (le joueur ne clique pas) et source de bugs visuels. Un",
            T*2 + "// reglage reactive via le menu en cours de bonus est coupe ici, a",
            T*2 + "// CHAQUE tour (l'entree de bonus le coupait deja dans freeSpinTrigger).",
            T*2 + "if (bookEvent.gameType === 'freegame') {",
            T*3 + "stateBetDerived.updateIsTurbo(false, { persistent: true });",
            T*3 + "stateBet.isSuperTurbo = false;",
            T*2 + "}",
            T*2 + "stateGame.gameType = bookEvent.gameType;",
            T*2 + "await stateGameDerived.enhancedBoard.spin({",
        ]), "W1")


def main():
    report = []
    try:
        run_patches()
        for p, c in files.items():
            with open(p, "w") as f:
                f.write(c)
        report.append("===== JOURNAL DES PATCHS =====")
        report.extend(journal)
        patch_failed = False
    except AssertionError as e:
        report.append("===== JOURNAL DES PATCHS (ECHEC, RIEN N'A ETE ECRIT) =====")
        report.extend(journal)
        report.append(str(e))
        patch_failed = True

    report.append("")
    report.append("===== PREUVES =====")
    all_ok = not patch_failed
    with open(BOOK) as f:
        content = f.read()
    n = content.count("Turbo et super turbo DESACTIVES pendant les bonus")
    report.append("garde turbo par tour de bonus : %d (attendu 1)" % n)
    if n != 1:
        all_ok = False
    n = content.count("updateIsTurbo(false")
    report.append("appels updateIsTurbo(false : %d (attendu 2 : freeSpinTrigger + reveal)" % n)
    if n < 2:
        all_ok = False

    report.append("")
    report.append("===== git status --short =====")
    try:
        out = subprocess.run(["git", "status", "--short"], capture_output=True, text=True).stdout
        report.append(out.rstrip() or "(arbre propre)")
    except OSError as e:
        report.append("git indisponible : %s" % e)

    with open("rapport33.txt", "w") as f:
        f.write("\n".join(report) + "\n")

    report.append("")
    report.append("===== BUILD =====")
    if all_ok:
        timed_out = False
        with open("build_log.txt", "w") as logf:
            try:
                code = subprocess.run(["pnpm", "run", "build", "--filter=louvo"],
                                      stdout=logf, stderr=subprocess.STDOUT, timeout=300).returncode
            except subprocess.TimeoutExpired:
                timed_out = True
                code = -1
        with open("build_log.txt") as f:
            log = f.read()
        report.extend(log.rstrip().split("\n")[-6:])
        site_ecrit = ("Wrote site to" in log) or ("done" in log)
        if timed_out:
            report.append("(build coupe apres 5 min - comportement pnpm connu)")
        report.append("site ecrit : %s / exit : %s" % ("OUI" if site_ecrit else "NON", code))
        if not site_ecrit:
            all_ok = False
    else:
        report.append("BUILD SAUTE (patchs incomplets)")

    report.append("")
    report.append("VERDICT : %s" % ("TOUT OK - commit, deploiement, tests" if all_ok else "ECHEC - envoyez ce rapport a Claude"))

    with open("rapport33.txt", "w") as f:
        f.write("\n".join(report) + "\n")
    print("\n".join(report[-4:]))
    print("FINI - envoyez rapport33.txt")
    sys.exit(0 if all_ok else 1)


main()
