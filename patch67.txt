#!/usr/bin/env python3
# ============================================================
# Louvo - lot 70 : zone cliquable des boutons OK, SANS toucher au visuel
#
# CORRIGE LE LOT 69, qui avait grossi le bouton a l'oeil alors que la
# demande etait explicitement : la zone cliquable uniquement.
#
# CE QUE FAIT CE PATCH
#   1) remet BaseIcon a height="3rem" (le graphisme retrouve sa taille
#      d'origine ; sans effet si le lot 69 n'a pas ete lance)
#   2) entoure le bouton OK d'un <div class="ok-hit"> et ajoute
#      1.5rem de rembourrage transparent au-dessus et au-dessous du
#      bouton lui-meme
#
# Resultat : le graphisme reste a 3rem, la zone sensible passe a 6rem.
# Rien ne bouge visuellement.
#
# POURQUOI DU !important : le rembourrage doit gagner sur la regle de
# Button.svelte, un composant partage du SDK que je prefere ne pas
# modifier pour deux modales. C'est un choix assume, pas un reflexe.
#
# PORTEE : la regle est enfermee dans la classe .ok-hit, propre a
# chaque modale. Le bouton d'autoplay, les cartes de bonus et les +/-
# de mise ne sont pas concernes.
#
# Sauvegardes en .avant-lot70. Sort en erreur si ca echoue.
# AUCUN BUILD ICI.
#
# Usage :
#   cd /workspaces/web-sdk && python3 patch67.txt
# ============================================================
import os
import re
import shutil
import sys
import traceback

ROOT = os.path.dirname(os.path.abspath(__file__))
os.chdir(ROOT)

RAPPORT = "rapport77.txt"
report = []


def flush():
    try:
        with open(RAPPORT, "w") as f:
            f.write("\n".join(report) + "\n")
    except OSError:
        pass


def say(line=""):
    report.append(line)
    print(line)
    try:
        sys.stdout.flush()
    except Exception:
        pass
    flush()


BASE = "packages/components-ui-html/src/components"
CIBLES = [("ModalBuyBonusConfirm.svelte", True), ("ModalBetMenu.svelte", False)]

BOUTON = re.compile(r"([ \t]*)(<Button\b(?:.|\n)*?</Button>)")
ICONE_6 = re.compile(r'(<BaseIcon\b[^>]*height=")6rem("[^>]*/>)')

STYLE = """
<style lang="scss">
	/* -------------------------------------------------------------
	   Zone cliquable du bouton OK.
	   Le graphisme reste a 3rem : on n'y touche pas. On ajoute
	   seulement 1.5rem de rembourrage transparent au-dessus et
	   au-dessous du bouton, ce qui double la surface sensible sans
	   rien changer a l'aspect.
	   Le !important passe devant la regle de Button.svelte, composant
	   partage du SDK qu'on prefere ne pas modifier pour deux modales.
	   ------------------------------------------------------------- */
	.ok-hit {
		width: 100%;
		display: flex;
		justify-content: center;
	}

	.ok-hit > :global(*) {
		width: 100%;
		box-sizing: content-box !important;
		padding-top: 1.5rem !important;
		padding-bottom: 1.5rem !important;
	}
</style>
"""


def indente(bloc, tab="\t"):
    return "\n".join((tab + l) if l.strip() else l for l in bloc.split("\n"))


def traite(txt):
    """Renvoie (txt, nb_icones_remises, nb_boutons_entoures)."""
    n_icone = 0
    n_bouton = 0
    morceaux = []
    fin = 0
    for m in BOUTON.finditer(txt):
        bloc = m.group(2)
        if "i18nDerived.confirm()" not in bloc:
            continue
        indent = m.group(1)
        bloc2, k = ICONE_6.subn(lambda mm: mm.group(1) + "3rem" + mm.group(2), bloc)
        n_icone += k
        nouveau = (indent + '<div class="ok-hit">\n'
                   + indent + indente(bloc2) + "\n"
                   + indent + "</div>")
        morceaux.append(txt[fin:m.start()])
        morceaux.append(nouveau)
        fin = m.end()
        n_bouton += 1
    morceaux.append(txt[fin:])
    return "".join(morceaux), n_icone, n_bouton


def pose_style(txt):
    if ".ok-hit" in txt:
        return txt, False
    m = re.search(r"<style[^>]*>", txt)
    if m:
        # on insere nos regles dans le bloc de style existant
        regles = STYLE.split("<style lang=\"scss\">", 1)[1].rsplit("</style>", 1)[0]
        return txt[:m.end()] + regles + txt[m.end():], True
    return txt.rstrip("\n") + "\n" + STYLE, True


def main():
    tout_ok = True

    say("===== 1. MODIFICATIONS =====")
    for nom, obligatoire in CIBLES:
        p = os.path.join(BASE, nom)
        if not os.path.exists(p):
            say("  %-32s INTROUVABLE%s" % (nom, "" if obligatoire else " (optionnel)"))
            if obligatoire:
                tout_ok = False
            continue

        txt = open(p, encoding="utf-8").read()
        if 'class="ok-hit"' in txt:
            say("  %-32s deja fait" % nom)
            continue

        txt2, n_icone, n_bouton = traite(txt)
        if n_bouton == 0:
            say("  %-32s aucun bouton confirm() trouve - NON MODIFIE%s"
                % (nom, "" if obligatoire else " (optionnel, sans gravite)"))
            if obligatoire:
                tout_ok = False
            continue

        txt2, style_pose = pose_style(txt2)

        if not os.path.exists(p + ".avant-lot70"):
            shutil.copyfile(p, p + ".avant-lot70")
        with open(p, "w", encoding="utf-8") as f:
            f.write(txt2)
        say("  %-32s %d bouton(s) entoure(s), %d icone(s) remise(s) a 3rem, style %s"
            % (nom, n_bouton, n_icone, "ajoute" if style_pose else "deja la"))

    # ---------- 2. verification ----------
    say("")
    say("===== 2. VERIFICATION =====")
    for nom, obligatoire in CIBLES:
        p = os.path.join(BASE, nom)
        if not os.path.exists(p):
            continue
        txt = open(p, encoding="utf-8").read()
        for motif, attendu in (('class="ok-hit"', True),
                               ('height="3rem"', True),
                               ('height="6rem"', False),
                               ('padding-top: 1.5rem !important', True)):
            present = motif in txt
            if present != attendu and obligatoire:
                tout_ok = False
            say("  %-30s %-34s %-8s %s" % (
                nom, motif, "present" if present else "absent",
                "OK" if present == attendu else
                ("*** ECHEC ***" if obligatoire else "(optionnel)")))

    # ---------- 3. rien d'autre n'a bouge ----------
    say("")
    say("===== 3. LES AUTRES BOUTONS SONT INTACTS =====")
    for fn in sorted(os.listdir(BASE)):
        if not fn.endswith(".svelte"):
            continue
        txt = open(os.path.join(BASE, fn), encoding="utf-8", errors="ignore").read()
        for i, l in enumerate(txt.split("\n"), 1):
            if "BaseIcon" in l and "height=" in l:
                say("  %-34s:%-4d %s" % (fn, i, l.strip()[:90]))

    # ---------- 4. le fichier obtenu ----------
    say("")
    say("===== 4. LE BOUTON OK APRES MODIFICATION =====")
    p = os.path.join(BASE, "ModalBuyBonusConfirm.svelte")
    if os.path.exists(p):
        lignes = open(p, encoding="utf-8").read().split("\n")
        for i, l in enumerate(lignes, 1):
            if 'ok-hit' in l or 'BaseIcon' in l or '<Button' in l or '</Button>' in l \
               or 'padding-top' in l or 'box-sizing' in l:
                say("%4d| %s" % (i, l))

    say("")
    say("===== 5. TAILLE =====")
    say("  graphisme du bouton : 3rem, INCHANGE")
    say("  zone cliquable      : 3rem -> 6rem (1.5rem de chaque cote)")
    say("  soit, en pixels     : 48 -> 96 au-dessus de 500 px de large")
    say("                        24 -> 48 en dessous")
    say("")
    say("  Pour ajuster : les deux valeurs 1.5rem dans le bloc <style>")
    say("  de chaque modale.")

    # ---------- 6. de quoi enchainer ----------
    say("")
    say("===== 6. LECTURE : Button.svelte et BaseIcon.svelte =====")
    for chemin in ("packages/components-shared/src/components/Button.svelte",
                   os.path.join(BASE, "BaseIcon.svelte")):
        say("")
        say("--------------------------------------------------------------")
        try:
            src = open(chemin, encoding="utf-8", errors="ignore").read()
        except OSError:
            say(">>> INTROUVABLE : %s" % chemin)
            continue
        say(">>> %s" % chemin)
        say("--------------------------------------------------------------")
        for i, l in enumerate(src.split("\n")[:200], 1):
            say("%4d| %s" % (i, l))
    return tout_ok


try:
    ok = main()
except Exception:
    say("")
    say("===== ERREUR PYTHON =====")
    for l in traceback.format_exc().rstrip().split("\n"):
        say(l)
    ok = False

say("")
say("VERDICT : %s" % ("OK" if ok else "ECHEC"))
print("")
print("FINI - envoyez rapport77.txt")
sys.exit(0 if ok else 1)
