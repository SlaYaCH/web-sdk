#!/usr/bin/env python3
# ============================================================
# Louvo - lot 50 : reglages de mise en page (retours de test)
#
# 1) DOUBLE ENSEIGNE LOUVO en portrait
#    Le fond contient LUI-MEME l'enseigne et le contour de la grille.
#    Depuis le lot 48 il couvrait l'ecran independamment du cadre : on
#    voyait donc l'enseigne deux fois. Le fond suit desormais EXACTEMENT
#    la meme boite que le cadre, et le sol est prolonge d'un aplat uni
#    jusqu'en bas de l'ecran (votre idee).
#
# 2) BARRE DE CHARGEMENT qui disparait sur du noir
#    Elle etait masquee des que le chargement finissait. Elle reste
#    maintenant affichee, pleine, jusqu'au TAP TO CONTINUE.
#
# 3) GRILLE 10% TROP PETITE en Popout L et Popout S
#    Defaut ANCIEN, revele par les tests : la reference paysage fait 1600
#    de large alors que le plateau garde sa taille fixe de 580, dans une
#    zone de grille de 646. Sur bureau (reference 1422) il tombe juste.
#    L'echelle vaut desormais main.width/1422 : exactement 1 sur bureau
#    (donc RIEN NE CHANGE la), 1.125 en paysage.
#
# 4) GRILLE DECENTREE en portrait
#    Dans l'image d'origine la grille n'est pas au milieu : son centre
#    tombe a 49.55%. On centre desormais la GRILLE, pas la boite.
#    Et la grille est legerement reduite (0.90 -> 0.86 de la largeur).
#
# APRES CE LOT, l'ecart entre le plateau et la zone de grille du cadre
# vaut +1% a TOUTES les resolutions - le meme qu'aujourd'hui sur bureau.
#
# SAUVEGARDES : chaque fichier modifie est copie en .avant-lot50
#
# Rapport ecrit apres chaque ligne. AUCUN BUILD.
#
# Usage :
#   1) deposer ce fichier a la RACINE de web-sdk
#   2) cd /workspaces/web-sdk && python3 patch50.txt
#   3) envoyer rapport57.txt
# ============================================================
import os
import shutil
import sys
import traceback

ROOT = os.path.dirname(os.path.abspath(__file__))
os.chdir(ROOT)

RAPPORT = "rapport57.txt"
report = []


def flush():
    try:
        with open(RAPPORT, "w") as f:
            f.write("\n".join(report) + "\n")
    except OSError:
        pass


def say(line=""):
    report.append(line)
    print(line)
    try:
        sys.stdout.flush()
    except Exception:
        pass
    flush()


def sauver(p):
    s = p + ".avant-lot50"
    if not os.path.exists(s):
        shutil.copyfile(p, s)


def ecrire(p, contenu, tag):
    if os.path.exists(p) and open(p, encoding="utf-8").read() == contenu:
        say("  %-46s deja en place" % tag)
        return
    sauver(p)
    with open(p, "w", encoding="utf-8") as f:
        f.write(contenu)
    say("  %-46s reecrit (%d lignes)" % (tag, len(contenu.split("\n"))))


BOARD_BOX = '''import { GRID_LEFT_FRAC, GRID_RIGHT_FRAC, GRID_TOP_FRAC, GRID_BOTTOM_FRAC } from './constants';
import { stateLayoutDerived } from './stateLayout';

// ------------------------------------------------------------------
// LA BOITE DE CADRE
//
// Tout le decor de Louvo (fond ET cadre) est dessine en 1672x941, ratio
// 1.777, et les reperes GRID_*_FRAC de constants.ts sont mesures dessus.
// Cette boite est la surface ou l'on dessine ce decor, TOUJOURS a ce ratio.
// Le fond, le cadre, la grille et les rangees s'y referent tous.
// ------------------------------------------------------------------

const ART_RATIO = 1672 / 941;

// Largeur de reference : la taille en unites de mise en page pour laquelle
// le jeu a ete regle (bureau 1422x800). Le plateau y est a l'echelle 1.
const REFERENCE_WIDTH = 1422;

// Centre horizontal de la grille DANS l'image : 0.4955, pas 0.5. La grille
// n'est pas exactement au milieu du decor d'origine.
const GRID_CENTER_X_FRAC = (GRID_LEFT_FRAC + GRID_RIGHT_FRAC) / 2;
const GRID_WIDTH_FRAC = GRID_RIGHT_FRAC - GRID_LEFT_FRAC;

// ==================================================================
// REGLAGES PORTRAIT - c'est ici qu'on ajuste a l'oeil
// ==================================================================
// Part de la largeur d'ecran occupee par la grille.
// Monter = grille plus grosse et decor plus recadre sur les cotes.
const PORTRAIT_GRID_WIDTH_FRAC = 0.86;

// Hauteur a laquelle commence le decor, en fraction de la hauteur d'ecran.
// 0 = le decor touche le haut, aucune bande vide au-dessus.
const PORTRAIT_BOX_TOP_FRAC = 0;

// Decalage horizontal fin, en fraction de la largeur d'ecran.
// Positif = vers la droite. 0.01 vaut environ 3 px sur un ecran de 320.
const PORTRAIT_GRID_X_OFFSET_FRAC = 0;
// ==================================================================

export type BoardBox = {
	x: number;
	y: number;
	width: number;
	height: number;
	scale: number;
};

export const boardBox = (): BoardBox => {
	const main = stateLayoutDerived.mainLayout();
	const layoutType = stateLayoutDerived.layoutType();

	// Paysage : la boite EST la mise en page.
	// L'echelle vaut main.width / 1422, soit exactement 1 sur bureau (1422)
	// et 1.125 en paysage (1600) - ce qui corrige la grille 10% trop petite
	// des fenetres Popout, ou le plateau gardait sa taille fixe de 580 dans
	// une zone de grille de 646.
	if (layoutType === 'desktop' || layoutType === 'landscape') {
		return {
			x: 0,
			y: 0,
			width: main.width,
			height: main.height,
			scale: main.width / REFERENCE_WIDTH,
		};
	}

	// Portrait / tablette : le decor est dessine PLUS LARGE que l'ecran et
	// deborde sur les cotes, pour que la grille remplisse la largeur.
	const width = (PORTRAIT_GRID_WIDTH_FRAC * main.width) / GRID_WIDTH_FRAC;
	const height = width / ART_RATIO;

	return {
		// On centre la GRILLE sur l'ecran, pas la boite : sans ca la grille
		// tombe 0.45% trop a gauche, ce qui se voit.
		x: main.width * (0.5 + PORTRAIT_GRID_X_OFFSET_FRAC) - width * GRID_CENTER_X_FRAC,
		y: PORTRAIT_BOX_TOP_FRAC * main.height,
		width,
		height,
		scale: width / REFERENCE_WIDTH,
	};
};
'''

BACKGROUND = '''<script lang="ts">
	import { Rectangle, Sprite } from 'pixi-svelte';
	import { FadeContainer } from 'components-pixi';
	import { MainContainer } from 'components-layout';
	import { SECOND } from 'constants-shared/time';
	import { getContext } from '../game/context';
	import { boardBox } from '../game/boardBox';

	const context = getContext();

	const showBaseBackground = $derived(
		context.stateGame.gameType === 'basegame' || context.stateGame.tier === 'speed_dating',
	);
	const showFeatureBackground = $derived(context.stateGame.tier === 'after_dark');

	// ------------------------------------------------------------------
	// Le fond suit EXACTEMENT la meme boite que le cadre.
	// Il ne peut pas en etre detache : l'image de fond contient elle-meme
	// l'enseigne LOUVO et le contour de la grille. Si on la posait ailleurs
	// que le cadre, on verrait l'enseigne en double.
	//
	// L'ecran etant plus haut que l'image en portrait, on prolonge le sol
	// d'un aplat uni jusqu'en bas.
	// ------------------------------------------------------------------

	// Couleur du sol, prolongee sous le decor. A ajuster si la jointure se voit.
	const FLOOR_COLOR = 0xc0509a;
	const FLOOR_COLOR_AFTER_DARK = 0x3d1030;

	const box = $derived(boardBox());
	const main = $derived(context.stateLayoutDerived.mainLayout());
	const floorColor = $derived(showFeatureBackground ? FLOOR_COLOR_AFTER_DARK : FLOOR_COLOR);

	const sprite = $derived({
		x: box.x + box.width * 0.5,
		y: box.y + box.height * 0.5,
		width: box.width,
		height: box.height,
	});
</script>

<!-- Aplat de secours sur tout le canvas : evite toute bande noire quand la
     mise en page ne remplit pas l'ecran (ecrans tres allonges). -->
<Rectangle
	{...context.stateLayoutDerived.canvasSizes()}
	backgroundColor={floorColor}
	zIndex={-4}
/>

<MainContainer>
	<!-- Prolongement du sol sous le decor, largement deborde : le canvas
	     recadre ce qui depasse. -->
	<Rectangle
		x={-main.width}
		y={box.y + box.height}
		width={main.width * 3}
		height={main.height * 2}
		backgroundColor={floorColor}
		zIndex={-3}
	/>

	<FadeContainer show={showBaseBackground} duration={SECOND} zIndex={-2}>
		<Sprite key="boardBackground" {...sprite} anchor={0.5} />
	</FadeContainer>

	<FadeContainer show={showFeatureBackground} duration={SECOND} zIndex={-1}>
		<Sprite key="boardBackgroundAfterDark" {...sprite} anchor={0.5} />
	</FadeContainer>
</MainContainer>
'''


def main():
    say("===== 1. boardBox.ts : echelle paysage et centrage portrait =====")
    ecrire("apps/louvo/src/game/boardBox.ts", BOARD_BOX, "boardBox.ts")

    say("")
    say("===== 2. Background.svelte : le fond suit le cadre =====")
    ecrire("apps/louvo/src/components/Background.svelte", BACKGROUND, "Background.svelte")

    say("")
    say("===== 3. LoadingScreen.svelte : barre gardee pleine =====")
    p = "apps/louvo/src/components/LoadingScreen.svelte"
    txt = open(p, encoding="utf-8").read()
    if "{#if !context.stateApp.loaded}" not in txt:
        say("  deja applique")
    else:
        n_if = txt.count("{#if")
        n_fi = txt.count("{/if}")
        if n_if != 1 or n_fi != 1:
            raise AssertionError(
                "LoadingScreen contient %d '{#if' et %d '{/if}' (attendu 1 et 1) - "
                "rien n'a ete modifie" % (n_if, n_fi))
        sauver(p)
        nouveau = txt
        for ligne in ("        {#if !context.stateApp.loaded}\n", "        {/if}\n"):
            if nouveau.count(ligne) != 1:
                raise AssertionError("ligne introuvable ou multiple : %r" % ligne)
            nouveau = nouveau.replace(ligne, "", 1)
        with open(p, "w", encoding="utf-8") as f:
            f.write(nouveau)
        say("  la barre reste affichee jusqu'au TAP TO CONTINUE")

    say("")
    say("===== 4. VERIFICATION =====")
    controles = (
        ("apps/louvo/src/game/boardBox.ts", "PORTRAIT_GRID_X_OFFSET_FRAC", True),
        ("apps/louvo/src/game/boardBox.ts", "main.width / REFERENCE_WIDTH", True),
        ("apps/louvo/src/components/Background.svelte", "boardBox", True),
        ("apps/louvo/src/components/Background.svelte", "FLOOR_COLOR", True),
        ("apps/louvo/src/components/LoadingScreen.svelte", "stateApp.loaded", False),
    )
    tout_ok = True
    for chemin, motif, attendu in controles:
        present = motif in open(chemin, encoding="utf-8").read()
        if present != attendu:
            tout_ok = False
        say("  %-58s %-8s %s" % (os.path.basename(chemin) + " : " + motif,
                                 "present" if present else "absent",
                                 "OK" if present == attendu else "*** ECHEC ***"))

    say("")
    say("===== 5. GEOMETRIE ATTENDUE =====")
    GL, GR = 491 / 1672.0, 1166 / 1672.0
    GW, GCX = GR - GL, (GL + GR) / 2
    ART, REF = 1672 / 941.0, 1422.0
    W_FRAC = 0.86
    say("%-10s %-10s %8s %11s %10s" % ("ecran", "type", "echelle", "symbole px", "ecart cadre"))
    for nom, cw, ch, t, rw, rh in (
        ("Desktop", 1200, 675, "desktop", 1422, 800),
        ("Laptop", 1024, 576, "desktop", 1422, 800),
        ("Popout L", 800, 450, "landscape", 1600, 900),
        ("Popout S", 400, 225, "landscape", 1600, 900),
        ("Mobile L", 425, 812, "portrait", 800, 1422),
        ("Mobile M", 375, 667, "portrait", 800, 1422),
        ("Mobile S", 320, 568, "portrait", 800, 1422),
    ):
        s = min(cw / float(rw), ch / float(rh))
        if t == "portrait":
            w = W_FRAC * rw / GW
            echelle = w / REF
            zone = GW * w * s
        else:
            echelle = rw / REF
            zone = GW * rw * s
        plateau = 580 * echelle * s
        say("%-10s %-10s %8.3f %11.1f %9.0f%%" % (nom, t, echelle, plateau / 5,
                                                  100 * (plateau / zone - 1)))
    return tout_ok


try:
    ok = main()
except Exception:
    say("")
    say("===== ERREUR PYTHON =====")
    for l in traceback.format_exc().rstrip().split("\n"):
        say(l)
    ok = False

say("")
say("VERDICT : %s" % ("OK - build, puis verifiez BUREAU (inchange), Popout et portrait"
                     if ok else "ECHEC - envoyez rapport57.txt"))
print("")
print("FINI - envoyez rapport57.txt")
