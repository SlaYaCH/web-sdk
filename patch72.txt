#!/usr/bin/env python3
# ============================================================
# Louvo - lot 76 : lignes de gain plus rapides en turbo,
#                  et pas de suspense sur une colonne a banniere
#
# ================== SUJET 1 : LE TURBO ==================
# Aujourd'hui la presentation d'une ligne dure 1520 ms (120 apparition
# + 700 maintien + 150 disparition + 300 attente + 250 disparition du
# montant), plus 200 ms entre deux lignes - la meme chose en turbo
# qu'en vitesse normale.
#
# ATTENTION, LE PIEGE : il ne suffit PAS de raccourcir l'attente finale
# du handler. Cette attente existe justement pour que le tour suivant ne
# parte pas par-dessus une animation inachevee. Si on la coupe seule, on
# retombe sur le bug d'origine. Il faut accelerer les TROIS ensemble :
# l'animation de la ligne, l'ecart entre deux lignes, et l'attente.
#
# D'ou un facteur unique, dans un seul fichier, lu par les trois :
#   game/winLineSpeed.svelte.ts
#     turbo       -> 2.5x plus rapide  (1520 ms -> 608 ms)
#     super turbo -> 4x   plus rapide  (1520 ms -> 380 ms)
#     au moins BIG WIN -> 1x, jamais raccourci
#
# Le niveau de gain n'est pas porte par winInfo : il arrive sur le
# setWin qui suit. On le lit donc en regardant en avant dans le book.
# Si on ne le trouve pas, on considere que c'est un gros gain et on ne
# raccourcit rien - l'echec va vers l'ancien comportement.
#
# ============ SUJET 2 : SUSPENSE ET BANNIERES ============
# Le garde-fou existait deja mais il regardait les bannieres DEJA
# AFFICHEES (activeBannerReelIndexes). Or le suspense se joue pendant
# que les rouleaux tournent, avant que la banniere n'existe : il ne
# pouvait donc rien voir.
#
# Le book, lui, sait deja. Les evenements de banniere de CE tour se
# trouvent entre ce reveal et le suivant. On neutralise l'anticipation
# sur ces colonnes-la. Rien n'est change cote math : on ne touche qu'au
# tableau d'anticipation deja recopie localement, exactement comme le
# fait la regle des 2 DATE visibles juste au-dessus.
#
# FICHIERS
#   NOUVEAU   game/winLineSpeed.svelte.ts
#   MODIFIE   game/bookEventHandlerMap.ts
#   MODIFIE   components/WinLineReveal.svelte
#   MODIFIE   components/WinLinesDisplay.svelte
# Sauvegardes en .avant-lot76.
#
# Sort en erreur si ca echoue. AUCUN BUILD ICI.
#
# Usage :
#   cd /workspaces/web-sdk && python3 patch72.txt
# ============================================================
import os
import shutil
import sys
import traceback

ROOT = os.path.dirname(os.path.abspath(__file__))
os.chdir(ROOT)

RAPPORT = "rapport83.txt"
report = []


def flush():
    try:
        with open(RAPPORT, "w") as f:
            f.write("\n".join(report) + "\n")
    except OSError:
        pass


def say(line=""):
    report.append(line)
    print(line)
    try:
        sys.stdout.flush()
    except Exception:
        pass
    flush()


def lire(p):
    with open(p, encoding="utf-8") as f:
        return f.read()


def ecrire(p, txt):
    with open(p, "w", encoding="utf-8") as f:
        f.write(txt)


def sauver(p):
    if not os.path.exists(p + ".avant-lot76"):
        shutil.copyfile(p, p + ".avant-lot76")


def unique(txt, ancien, nouveau, ou):
    n = txt.count(ancien)
    if n != 1:
        raise AssertionError("%s : motif trouve %d fois (attendu 1) - RIEN ENREGISTRE" % (ou, n))
    return txt.replace(ancien, nouveau, 1)

VITESSE = """import { stateBet } from 'state-shared';

// ------------------------------------------------------------------
// VITESSE DE PRESENTATION DES LIGNES DE GAIN
//
// Trois endroits doivent rester d'accord, sinon le tour suivant part
// par-dessus une animation inachevee :
//   - WinLineReveal    : l'animation d'une ligne
//   - WinLinesDisplay  : l'ecart entre deux lignes
//   - bookEventHandlerMap (winInfo) : l'attente avant la suite
// Ils lisent tous les trois ce meme facteur.
//
// On ne raccourcit JAMAIS un gain qui atteint au moins BIG WIN : c'est
// le moment que le joueur veut voir.
// ------------------------------------------------------------------

// ==================================================================
// REGLAGES
// ==================================================================
// 1 = vitesse normale. 2.5 = deux fois et demie plus rapide.
const FACTEUR_TURBO = 2.5;
const FACTEUR_SUPER_TURBO = 4;
// ==================================================================

// Pose par le handler winInfo juste avant d'afficher les lignes.
export const stateWinLineSpeed = $state({ atteintBigWin: false });

export const winLineSpeed = () => {
	if (stateWinLineSpeed.atteintBigWin) return 1;
	if (stateBet.isSuperTurbo) return FACTEUR_SUPER_TURBO;
	if (stateBet.isTurbo) return FACTEUR_TURBO;
	return 1;
};
"""

ANTICIPATION_ANCIEN = """		{
			let cumulVisibles = 0;
			bookEvent.anticipation = bookEvent.anticipation.map((value, i) => {
				const keep = cumulVisibles >= 2 ? value : 0;
				cumulVisibles += bookEvent.board[i].slice(1, -1).filter((s) => s.name === 'S').length;
				return keep;
			});
		}"""

ANTICIPATION_NOUVEAU = ANTICIPATION_ANCIEN + """
		// Une colonne qui va porter une banniere (Match Duel ou Super Like)
		// n'a pas a faire monter le suspense : son resultat est deja annonce.
		// Le garde-fou d'Anticipations.svelte ne voit que les bannieres DEJA
		// affichees, or le suspense se joue pendant que les rouleaux tournent,
		// avant que la banniere n'existe. Le book, lui, sait deja : les
		// evenements de banniere de CE tour sont entre ce reveal et le suivant.
		{
			const index = (bookEvents as unknown[]).indexOf(bookEvent as unknown);
			const suite = index >= 0 ? bookEvents.slice(index + 1) : [];
			const finDuTour = suite.findIndex((e) => e.type === 'reveal');
			const duTour = finDuTour >= 0 ? suite.slice(0, finDuTour) : suite;
			const colonnesBanniere = new Set(
				duTour
					.filter((e) => e.type === 'matchDuelReveal' || e.type === 'superlikeReveal')
					.map((e) => (e as unknown as { reelIndex: number }).reelIndex),
			);
			if (colonnesBanniere.size > 0) {
				bookEvent.anticipation = bookEvent.anticipation.map((value, i) =>
					colonnesBanniere.has(i) ? 0 : value,
				);
			}
		}"""

ORDERED_NOUVEAU = ORDERED_ANCIEN + """
		// Le niveau de gain n'est pas porte par winInfo : il arrive sur le
		// setWin qui suit. On le lit en avant dans le book pour savoir si on
		// a droit de raccourcir la presentation en turbo. Introuvable = on
		// considere que c'est un gros gain et on ne raccourcit rien.
		{
			const index = (bookEvents as unknown[]).indexOf(bookEvent as unknown);
			const suivant =
				index >= 0
					? bookEvents.slice(index + 1).find((e) => e.type === 'setWin')

			stateWinLineSpeed.atteintBigWin = index < 0 || niveau?.type === 'big';
