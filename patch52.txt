#!/usr/bin/env python3
# ============================================================
# Louvo - lot 52 : retour aux anciens cadres
#
# ON GARDE : vos nouveaux fonds sans grille (board_background.png et
#            board_background_after_dark.png), et le fond etire.
# ON REPREND : les anciens cadres, recuperes depuis git. Ils etaient DEJA
#            sur fond transparent - sinon ils auraient masque le decor
#            derriere eux. Il n'y a donc rien a decouper.
# ON REVIENT : aux anciennes mesures de grille (491->1166, 130->657),
#            aux anciens centres de rangee, a l'ajustement vertical After
#            Dark et aux decalages par colonne.
#
# CE QU'ON GARDE DU LOT 51 : la boite de cadre, les deux echelles X et Y,
# et le fond independant. Avec les anciennes mesures, l'echelle retombe a
# 0.99 sur bureau (donc comme avant) et 1.11 en Popout, ce qui corrige la
# grille 10% trop petite sans rien changer ailleurs.
#
# VOS NOUVEAUX CADRES SONT SAUVEGARDES dans _nouveaux_cadres/ a la racine.
#
# Rapport ecrit apres chaque ligne. AUCUN BUILD.
#
# Usage :
#   1) deposer ce fichier a la RACINE de web-sdk
#   2) cd /workspaces/web-sdk && python3 patch52.txt
#   3) envoyer rapport59.txt
# ============================================================
import os
import shutil
import struct
import subprocess
import sys
import traceback

ROOT = os.path.dirname(os.path.abspath(__file__))
os.chdir(ROOT)

RAPPORT = "rapport59.txt"
SCREENS = "apps/louvo/static/assets/sprites/screens"
CADRES = ("board_frame_overlay.png", "board_frame_overlay_after_dark.png")
report = []
files = {}


def flush():
    try:
        with open(RAPPORT, "w") as f:
            f.write("\n".join(report) + "\n")
    except OSError:
        pass


def say(line=""):
    report.append(line)
    print(line)
    try:
        sys.stdout.flush()
    except Exception:
        pass
    flush()


def load(p):
    if p not in files:
        s = p + ".avant-lot52"
        if not os.path.exists(s):
            shutil.copyfile(p, s)
        with open(p, "r", encoding="utf-8") as f:
            files[p] = f.read()
    return files[p]


def rep(p, old, new, tag):
    txt = load(p)
    if new in txt:
        say("  %-46s deja applique" % tag)
        return
    n = txt.count(old)
    if n != 1:
        raise AssertionError("%s : motif trouve %d fois (attendu 1) dans %s" % (tag, n, p))
    files[p] = txt.replace(old, new, 1)
    say("  %-46s OK" % tag)


def infos_png(p):
    """Dimensions et type de couleur, sans dependance externe."""
    try:
        with open(p, "rb") as f:
            t = f.read(26)
    except OSError:
        return None
    if t[:8] != b"\x89PNG\r\n\x1a\n" or t[12:16] != b"IHDR":
        return None
    w, h = struct.unpack(">II", t[16:24])
    types = {0: "gris", 2: "RVB (PAS d'alpha)", 3: "palette", 4: "gris+alpha",
             6: "RVBA (transparent)"}
    return w, h, types.get(t[25], "type %d" % t[25])


NOUVEAU = """export const GRID_FRACS = {
\tbase: {
\t\tleft: 391 / GRID_SOURCE_WIDTH,
\t\tright: 1275 / GRID_SOURCE_WIDTH,
\t\ttop: 163 / GRID_SOURCE_HEIGHT,
\t\tbottom: 857 / GRID_SOURCE_HEIGHT,
\t},
\tafterDark: {
\t\tleft: 463 / GRID_SOURCE_WIDTH,
\t\tright: 1197 / GRID_SOURCE_WIDTH,
\t\ttop: 130 / GRID_SOURCE_HEIGHT,
\t\tbottom: 671 / GRID_SOURCE_HEIGHT,
\t},
} as const;

const centresDeRangee = (haut: number, bas: number) =>
\t[0, 1, 2, 3, 4].map((i) => (haut + ((bas - haut) / 5) * (i + 0.5)) / GRID_SOURCE_HEIGHT);

export const ROW_CENTERS_FRACS = {
\tbase: centresDeRangee(163, 857),
\tafterDark: centresDeRangee(130, 671),
};"""

ANCIEN = """// Anciennes mesures, celles des cadres d'origine. Les deux paliers
// partagent les memes reperes : c'est l'ajustement vertical et les
// decalages par colonne qui rattrapent la difference de cadre en
// After Dark, comme avant le lot 51.
export const GRID_FRACS = {
\tbase: {
\t\tleft: 491 / GRID_SOURCE_WIDTH,
\t\tright: 1166 / GRID_SOURCE_WIDTH,
\t\ttop: 130 / GRID_SOURCE_HEIGHT,
\t\tbottom: 657 / GRID_SOURCE_HEIGHT,
\t},
\tafterDark: {
\t\tleft: 491 / GRID_SOURCE_WIDTH,
\t\tright: 1166 / GRID_SOURCE_WIDTH,
\t\ttop: 130 / GRID_SOURCE_HEIGHT,
\t\tbottom: 657 / GRID_SOURCE_HEIGHT,
\t},
} as const;

// Centres exacts de chaque rangee, mesures dans l'image d'origine.
const CENTRES_ORIGINE = [
\t(130 + 233) / 2 / GRID_SOURCE_HEIGHT,
\t(237 + 340) / 2 / GRID_SOURCE_HEIGHT,
\t(343 + 446) / 2 / GRID_SOURCE_HEIGHT,
\t(450 + 554) / 2 / GRID_SOURCE_HEIGHT,
\t(557 + 657) / 2 / GRID_SOURCE_HEIGHT,
];

export const ROW_CENTERS_FRACS = {

\tafterDark: CENTRES_ORIGINE,
};"""


def main():
    # ---------- 1. Recuperation des anciens cadres ----------
    say("===== 1. ANCIENS CADRES RECUPERES DEPUIS GIT =====")
    os.makedirs("_nouveaux_cadres", exist_ok=True)
    for fn in CADRES:
        p = os.path.join(SCREENS, fn)
        garde = os.path.join("_nouveaux_cadres", fn)
        if not os.path.exists(garde) and os.path.exists(p):
            shutil.copyfile(p, garde)
            say("  votre nouveau %s sauvegarde dans _nouveaux_cadres/" % fn)
        r = subprocess.run(["git", "checkout", "HEAD", "--", p],
                           capture_output=True, text=True)
        if r.returncode != 0:
            raise AssertionError("git n'a pas pu restaurer %s : %s" % (fn, r.stderr.strip()))
        say("  %s restaure depuis git" % fn)

    say("")
    say("===== 2. CE QUE SONT LES FICHIERS RESTAURES =====")
    for fn in CADRES:
        p = os.path.join(SCREENS, fn)
        i = infos_png(p)
        taille = os.path.getsize(p) / 1048576.0
        if i:
            say("  %-38s %d x %d   %-20s %5.2f Mo" % (fn, i[0], i[1], i[2], taille))
            if "PAS d'alpha" in i[2]:
                say("     *** ATTENTION : pas de canal alpha, ce cadre n'est pas decoupe ***")
        else:
            say("  %-38s (illisible)" % fn)

    say("")
    say("  --- les fonds, eux, restent les votres ---")
    for fn in ("board_background.png", "board_background_after_dark.png"):
        p = os.path.join(SCREENS, fn)
        i = infos_png(p)
        if i:
            say("  %-38s %d x %d   %-20s %5.2f Mo"
                % (fn, i[0], i[1], i[2], os.path.getsize(p) / 1048576.0))

    # ---------- 3. Retour aux anciennes mesures ----------
    say("")
    say("===== 3. RETOUR AUX ANCIENNES MESURES =====")
    c = "apps/louvo/src/game/constants.ts"
    load(c)
    rep(c, NOUVEAU, ANCIEN, "GRID_FRACS et centres de rangee")

    s = "apps/louvo/src/game/stateGame.svelte.ts"
    load(s)
    rep(s, "const AFTER_DARK_Y_ADJUST_FRAC = 0;",
        "const AFTER_DARK_Y_ADJUST_FRAC = 0.005;", "ajustement vertical After Dark")

    u = "apps/louvo/src/game/utils.ts"
    load(u)
    rep(u,
        "// Remis a zero : les reperes propres a chaque palier rendent ces\n"
        "// decalages inutiles. Conserves au cas ou il faudrait re-affiner.\n"
        "const AFTER_DARK_COLUMN_ADJUST_FRAC = [0, 0, 0, 0, 0];",
        "const AFTER_DARK_COLUMN_ADJUST_FRAC = [-0.035, -0.02, -0.005, 0.015, 0.03];",
        "decalages par colonne restaures")

    for p, contenu in files.items():
        with open(p, "w", encoding="utf-8") as f:
            f.write(contenu)

    # ---------- 4. Verification ----------
    say("")
    say("===== 4. VERIFICATION =====")
    controles = (
        ("apps/louvo/src/game/constants.ts", "491 / GRID_SOURCE_WIDTH", True),
        ("apps/louvo/src/game/constants.ts", "391 / GRID_SOURCE_WIDTH", False),
        ("apps/louvo/src/game/stateGame.svelte.ts", "AFTER_DARK_Y_ADJUST_FRAC = 0.005", True),
        ("apps/louvo/src/game/utils.ts", "[-0.035, -0.02, -0.005, 0.015, 0.03]", True),
        ("apps/louvo/src/components/Background.svelte", "boardBox", False),
    )
    tout_ok = True
    for chemin, motif, attendu in controles:
        present = motif in open(chemin, encoding="utf-8").read()
        if present != attendu:
            tout_ok = False
        say("  %-58s %-8s %s" % (os.path.basename(chemin) + " : " + motif,
                                 "present" if present else "absent",
                                 "OK" if present == attendu else "*** ECHEC ***"))

    say("")
    say("===== 5. TAILLE DE SYMBOLE ATTENDUE =====")
    L, R = 491 / 1672.0, 1166 / 1672.0
    T, B = 130 / 941.0, 657 / 941.0
    say("%-10s %-10s %8s %8s %13s" % ("ecran", "type", "ech. X", "ech. Y", "symbole px"))
    for nom, cw, ch, t, rw, rh in (
        ("Desktop", 1200, 675, "desktop", 1422, 800),
        ("Laptop", 1024, 576, "desktop", 1422, 800),
        ("Popout L", 800, 450, "landscape", 1600, 900),
        ("Popout S", 400, 225, "landscape", 1600, 900),
        ("Mobile L", 425, 812, "portrait", 800, 1422),
        ("Mobile M", 375, 667, "portrait", 800, 1422),
        ("Mobile S", 320, 568, "portrait", 800, 1422),
    ):
        e = min(cw / float(rw), ch / float(rh))
        if t == "portrait":
            w = 0.86 * rw / (R - L)
            h = w / (1672 / 941.0)
        else:
            w, h = rw, rh
        cw_cell = ((R - L) / 5) * w
        ch_cell = ((B - T) / 5) * h
        say("%-10s %-10s %8.3f %8.3f %13s"
            % (nom, t, cw_cell / 116.0, ch_cell / 91.0,
               "%.0f x %.0f" % (cw_cell * e, ch_cell * e)))
    return tout_ok


try:
    ok = main()
except Exception:
    say("")
    say("===== ERREUR PYTHON =====")
    for l in traceback.format_exc().rstrip().split("\n"):
        say(l)
    ok = False

say("")
say("VERDICT : %s" % ("OK - build" if ok else "ECHEC - envoyez rapport59.txt"))
print("")
print("FINI - envoyez rapport59.txt")
