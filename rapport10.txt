     4	
     5		import { getContext } from '../game/context';

    18		const onTurbo = () => {

    20			const next = !stateBet.isTurbo;

    22			if (!next) stateBet.isSuperTurbo = false;
    23		};
    24		const onSuperTurbo = () => {
    25			context.eventEmitter.broadcast({ type: 'soundPressGeneral' });
    26			const next2 = !stateBet.isSuperTurbo;
    27			stateBet.isSuperTurbo = next2;
    28			if (next2) stateBetDerived.updateIsTurbo(true, { persistent: true });
    29		};
    30		const onStopOnWin = () => {
    31			context.eventEmitter.broadcast({ type: 'soundPressGeneral' });
    32			stateBet.stopOnWin = !stateBet.stopOnWin;
    33		};
    34	
    35		const onInfo = () => {
    36			context.eventEmitter.broadcast({ type: 'soundPressGeneral' });
    37			stateModal.modal = { name: 'gameRules' };
    38		};
    39	
    40		// Accueil : pas de mecanisme de sortie/lobby existant - ferme le panneau
    41		const onHome = () => {
    42			context.eventEmitter.broadcast({ type: 'soundPressGeneral' });
    43			stateUi.menuOpen = false;
    44		};
    45	
    46		const turboLabel = $derived(stateBet.isTurbo ? 'TURBO: ON' : 'TURBO: OFF');
    47		const superTurboLabel = $derived(stateBet.isSuperTurbo ? 'SUPER TURBO: ON' : 'SUPER TURBO: OFF');
    48		const stopOnWinLabel = $derived(stateBet.stopOnWin ? 'STOP ON WIN: ON' : 'STOP ON WIN: OFF');
    49	
    50		const BUTTON_ROWS = $derived([
    51			{ label: turboLabel, onpress: onTurbo },
    52			{ label: superTurboLabel, onpress: onSuperTurbo },
    53			{ label: stopOnWinLabel, onpress: onStopOnWin },
    54			{ label: 'INFO / RULES', onpress: onInfo },
    55			{ label: 'CLOSE', onpress: onHome },
    56		]);
    57	</script>
    58	
    59	<Container>
    60		<Rectangle
