#!/usr/bin/env python3
# ============================================================
# Louvo - lot 49 : la boite de cadre (decouplage du cadre et de la grille)
#
# SUITE DU LOT 48. Le fond est deja independant ; c'est au tour du cadre.
#
# LE PROBLEME : le cadre etait etire sur mainLayout(), et la grille placee
# par des fractions de mainLayout(). En portrait mainLayout fait 800x1422
# (ratio 0.563) alors que le decor est dessine en 1672x941 (ratio 1.777) :
# le cadre s'ecrase et la grille ne tombe plus dedans. D'ou les rangees
# vides sur vos captures.
#
# LA CORRECTION : un nouveau module boardBox.ts definit LA boite ou le
# cadre est dessine, toujours au ratio de l'image. Le cadre, la grille et
# les rangees s'y referent tous.
#
# PROPRIETE DE SECURITE : en paysage, boardBox() renvoie exactement
# {x:0, y:0, width: mainLayout.width, height: mainLayout.height, scale:1}.
# Toutes les formules modifiees se reduisent alors a celles d'avant :
#   x = 0 + width * frac   ==   width * frac
# LE PAYSAGE EST DONC INCHANGE PAR CONSTRUCTION, pas par chance.
# Tout risque est confine au portrait, qui est deja casse aujourd'hui.
#
# En portrait le cadre est dessine PLUS LARGE que l'ecran et deborde sur
# les cotes : la grille passe de 30 px par symbole a environ 67 px.
# Les deux reglages (largeur de grille, hauteur du centre) sont en haut de
# boardBox.ts, commentes, faciles a ajuster a l'oeil.
#
# SAUVEGARDES : chaque fichier modifie est copie en .avant-lot49
#
# Rapport ecrit apres chaque ligne. AUCUN BUILD.
#
# Usage :
#   1) deposer ce fichier a la RACINE de web-sdk
#   2) cd /workspaces/web-sdk && python3 patch49.txt
#   3) envoyer rapport56.txt
# ============================================================
import os
import shutil
import sys
import traceback

ROOT = os.path.dirname(os.path.abspath(__file__))
os.chdir(ROOT)

RAPPORT = "rapport56.txt"
report = []
files = {}


def flush():
    try:
        with open(RAPPORT, "w") as f:
            f.write("\n".join(report) + "\n")
    except OSError:
        pass


def say(line=""):
    report.append(line)
    print(line)
    try:
        sys.stdout.flush()
    except Exception:
        pass
    flush()


def load(p):
    if p not in files:
        sauvegarde = p + ".avant-lot49"
        if not os.path.exists(sauvegarde):
            shutil.copyfile(p, sauvegarde)
        with open(p, "r", encoding="utf-8") as f:
            files[p] = f.read()
    return files[p]


def rep(p, old, new, tag):
    txt = load(p)
    if new in txt:
        say("  %-46s deja applique" % tag)
        return
    n = txt.count(old)
    if n != 1:
        raise AssertionError("%s : motif trouve %d fois (attendu 1) dans %s" % (tag, n, p))
    files[p] = txt.replace(old, new, 1)
    say("  %-46s OK" % tag)


BOARD_BOX = '''import { GRID_LEFT_FRAC, GRID_RIGHT_FRAC, GRID_TOP_FRAC, GRID_BOTTOM_FRAC } from './constants';
import { stateLayoutDerived } from './stateLayout';

// ------------------------------------------------------------------
// LA BOITE DE CADRE
//
// Tout le decor de Louvo (fond ET cadre) est dessine en 1672x941, ratio
// 1.777. Les reperes GRID_*_FRAC de constants.ts sont mesures sur cette
// image. Tant que le cadre etait etire sur mainLayout(), dont le ratio
// vaut 0.563 en portrait, la grille ne tombait plus dedans.
//
// Cette boite est donc la surface ou l'on dessine le cadre, TOUJOURS au
// ratio de l'image. La grille, les rangees et le cadre s'y referent tous,
// au lieu de se referer a mainLayout().
//
// PROPRIETE IMPORTANTE : en paysage, la boite vaut exactement la mise en
// page (x=0, y=0, scale=1). Toutes les formules qui l'utilisent se
// reduisent alors a celles d'avant. Le paysage est donc inchange par
// construction, pas par chance.
// ------------------------------------------------------------------

// Ratio natif des images de decor (board_background.png / board_frame_overlay.png)
const ART_RATIO = 1672 / 941;

// Largeur de reference : la taille en unites de mise en page pour laquelle
// le jeu a ete regle (bureau 1422x800). Sert d'echelle 1 pour le plateau.
const REFERENCE_WIDTH = 1422;

// ------------------------------------------------------------------
// REGLAGES PORTRAIT - c'est ici qu'on ajuste a l'oeil
// ------------------------------------------------------------------
// Part de la largeur d'ecran que la grille doit occuper.
// 0.90 = la grille fait 90% de la largeur. Monter = grille plus grosse et
// cadre plus recadre sur les cotes.
const PORTRAIT_GRID_WIDTH_FRAC = 0.9;

// Hauteur d'ecran ou se place le CENTRE de la grille.
// 0.40 = grille dans la moitie haute, place libre en dessous pour la barre.
// Descendre = grille plus haute ; monter = grille plus basse.
const PORTRAIT_GRID_CENTER_Y_FRAC = 0.4;
// ------------------------------------------------------------------

const GRID_WIDTH_FRAC = GRID_RIGHT_FRAC - GRID_LEFT_FRAC;
const GRID_CENTER_Y_FRAC = (GRID_TOP_FRAC + GRID_BOTTOM_FRAC) / 2;

export type BoardBox = {
	x: number;
	y: number;
	width: number;
	height: number;
	scale: number;
};

export const boardBox = (): BoardBox => {
	const main = stateLayoutDerived.mainLayout();
	const layoutType = stateLayoutDerived.layoutType();

	// Paysage : la boite EST la mise en page, et l'echelle vaut EXACTEMENT 1.
	// C'est ce qui garantit que rien ne bouge : le plateau garde sa taille
	// fixe (BOARD_SIZES) comme avant ce lot.
	if (layoutType === 'desktop' || layoutType === 'landscape') {
		return {
			x: 0,
			y: 0,
			width: main.width,
			height: main.height,
			scale: 1,
		};
	}

	// Portrait / tablette : le decor est dessine PLUS LARGE que l'ecran et
	// deborde sur les cotes, pour que la grille remplisse la largeur. C'est
	// la methode des jeux Hacksaw : on recadre au lieu d'ecraser.
	const width = (PORTRAIT_GRID_WIDTH_FRAC * main.width) / GRID_WIDTH_FRAC;
	const height = width / ART_RATIO;

	return {
		x: (main.width - width) / 2,
		y: PORTRAIT_GRID_CENTER_Y_FRAC * main.height - GRID_CENTER_Y_FRAC * height,
		width,
		height,
		scale: width / REFERENCE_WIDTH,
	};
};
'''


MAIN_W = "context.stateLayoutDerived.mainLayout().width"
MAIN_H = "context.stateLayoutDerived.mainLayout().height"


def cadre(cle):
    """Ancien bloc de positionnement d'un Sprite de cadre."""
    return ("\t\tkey=\"%s\"\n"
            "\t\tx={%s * 0.5}\n"
            "\t\ty={%s * 0.5}\n"
            "\t\tanchor={0.5}\n"
            "\t\twidth={%s}\n"
            "\t\theight={%s}\n" % (cle, MAIN_W, MAIN_H, MAIN_W, MAIN_H))


def cadre_nouveau(cle):
    return ("\t\tkey=\"%s\"\n"
            "\t\tx={boardBox().x + boardBox().width * 0.5}\n"
            "\t\ty={boardBox().y + boardBox().height * 0.5}\n"
            "\t\tanchor={0.5}\n"
            "\t\twidth={boardBox().width}\n"
            "\t\theight={boardBox().height}\n" % cle)


def main():
    # ---------- 1. Le module ----------
    say("===== 1. NOUVEAU MODULE game/boardBox.ts =====")
    cible = "apps/louvo/src/game/boardBox.ts"
    if os.path.exists(cible) and open(cible, encoding="utf-8").read() == BOARD_BOX:
        say("  deja en place, identique")
    else:
        with open(cible, "w", encoding="utf-8") as f:
            f.write(BOARD_BOX)
        say("  ecrit (%d lignes)" % len(BOARD_BOX.split("\n")))

    # ---------- 2. Le cadre ----------
    say("")
    say("===== 2. BoardFrame.svelte : le cadre suit la boite =====")
    b = "apps/louvo/src/components/BoardFrame.svelte"
    load(b)
    rep(b,
        "\timport { getContext } from '../game/context';",
        "\timport { getContext } from '../game/context';\n"
        "\timport { boardBox } from '../game/boardBox';",
        "import boardBox")
    rep(b, cadre("boardFrameOverlay"), cadre_nouveau("boardFrameOverlay"), "cadre de base")
    rep(b, cadre("boardFrameOverlayAfterDark"), cadre_nouveau("boardFrameOverlayAfterDark"),
        "cadre After Dark")

    # ---------- 3. La position et l'echelle du plateau ----------
    say("")
    say("===== 3. stateGame.svelte.ts : boardLayout suit la boite =====")
    s = "apps/louvo/src/game/stateGame.svelte.ts"
    load(s)
    rep(s,
        "import { stateLayoutDerived } from './stateLayout';",
        "import { stateLayoutDerived } from './stateLayout';\nimport { boardBox } from './boardBox';",
        "import boardBox")
    rep(s,
        "\tconst afterDarkYAdjust = stateGame.tier === 'after_dark' ? AFTER_DARK_Y_ADJUST_FRAC : 0;\n"
        "\treturn {\n"
        "\t\tx: stateLayoutDerived.mainLayout().width * ((GRID_LEFT_FRAC + GRID_RIGHT_FRAC) / 2),\n"
        "\t\ty:\n"
        "\t\t\tstateLayoutDerived.mainLayout().height *\n"
        "\t\t\t((GRID_TOP_FRAC + GRID_BOTTOM_FRAC) / 2 + afterDarkYAdjust),\n"
        "\t\tanchor: { x: 0.5, y: 0.5 },",
        "\tconst afterDarkYAdjust = stateGame.tier === 'after_dark' ? AFTER_DARK_Y_ADJUST_FRAC : 0;\n"
        "\t// La grille se repere sur la boite de cadre, plus sur mainLayout().\n"
        "\t// En paysage la boite vaut la mise en page (x=0, y=0, scale=1) :\n"
        "\t// ces formules redonnent exactement les anciennes.\n"
        "\tconst box = boardBox();\n"
        "\treturn {\n"
        "\t\tx: box.x + box.width * ((GRID_LEFT_FRAC + GRID_RIGHT_FRAC) / 2),\n"
        "\t\ty: box.y + box.height * ((GRID_TOP_FRAC + GRID_BOTTOM_FRAC) / 2 + afterDarkYAdjust),\n"
        "\t\tscale: box.scale,\n"
        "\t\tanchor: { x: 0.5, y: 0.5 },",
        "boardLayout sur la boite")

    # ---------- 4. L'echelle du conteneur ----------
    say("")
    say("===== 4. BoardContainer.svelte : le plateau s'echelonne =====")
    c = "apps/louvo/src/components/BoardContainer.svelte"
    load(c)
    rep(c,
        "\tpivot={context.stateGameDerived.boardLayout().pivot}\n\tsortableChildren={true}",
        "\tpivot={context.stateGameDerived.boardLayout().pivot}\n"
        "\tscale={context.stateGameDerived.boardLayout().scale}\n"
        "\tsortableChildren={true}",
        "scale du conteneur")

    # ---------- 5. Les centres de rangee ----------
    say("")
    say("===== 5. utils.ts : centres de rangee sur la boite =====")
    u = "apps/louvo/src/game/utils.ts"
    load(u)
    rep(u,
        "import { stateLayoutDerived } from './stateLayout';",
        "import { stateLayoutDerived } from './stateLayout';\nimport { boardBox } from './boardBox';",
        "import boardBox")
    rep(u,
        "export const getSymbolY = (symbolIndexOfBoard: number) =>\n"
        "\tstateLayoutDerived.mainLayout().height * ROW_CENTERS_FRAC[symbolIndexOfBoard];",
        "export const getSymbolY = (symbolIndexOfBoard: number) =>\n"
        "\tboardBox().y + boardBox().height * ROW_CENTERS_FRAC[symbolIndexOfBoard];",
        "getSymbolY sur la boite")

    for p, contenu in files.items():
        with open(p, "w", encoding="utf-8") as f:
            f.write(contenu)

    # ---------- 6. Verification ----------
    say("")
    say("===== 6. VERIFICATION =====")
    controles = (
        ("apps/louvo/src/game/boardBox.ts", "PORTRAIT_GRID_WIDTH_FRAC"),
        ("apps/louvo/src/components/BoardFrame.svelte", "boardBox().width"),
        ("apps/louvo/src/game/stateGame.svelte.ts", "scale: box.scale"),
        ("apps/louvo/src/components/BoardContainer.svelte", "boardLayout().scale"),
        ("apps/louvo/src/game/utils.ts", "boardBox().y + boardBox().height"),
    )
    tout_ok = True
    for chemin, motif in controles:
        present = motif in open(chemin, encoding="utf-8").read()
        if not present:
            tout_ok = False
        say("  %-58s %s" % (os.path.basename(chemin) + " : " + motif,
                            "OK" if present else "*** ABSENT ***"))

    say("")
    say("  aucune occurrence restante de mainLayout dans BoardFrame : %s"
        % ("oui" if "mainLayout" not in open(
            "apps/louvo/src/components/BoardFrame.svelte", encoding="utf-8").read() else "NON"))
    return tout_ok


try:
    ok = main()
except Exception:
    say("")
    say("===== ERREUR PYTHON - AUCUN FICHIER ENREGISTRE =====")
    for l in traceback.format_exc().rstrip().split("\n"):
        say(l)
    ok = False

say("")
say("VERDICT : %s" % ("OK - build, puis comparez PAYSAGE (doit etre identique) et PORTRAIT"
                     if ok else "ECHEC - envoyez rapport56.txt"))
print("")
print("FINI - envoyez rapport56.txt")
