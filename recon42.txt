# confirmation d'achat de bonus.

#     :global(html) { font-size: 16px;
#       @media screen and (max-width: 500px) { font-size: 50%; } }
# Sous 500 px de large, la police racine est divisee par deux. Tout ce
# qui est dimensionne en rem rapetisse d'autant, zone cliquable comprise.
#
# NE MODIFIE RIEN. Rapport ecrit apres chaque ligne. AUCUN BUILD.

#   cd /workspaces/web-sdk && python3 recon42.txt
#   puis envoyer rapport75.txt

import os
import re
import sys
import traceback

ROOT = os.path.dirname(os.path.abspath(__file__))
os.chdir(ROOT)

RAPPORT = "rapport75.txt"
report = []

BASE = "packages/components-ui-html/src"
IGNORE = {"node_modules", "dist", ".turbo", "build", ".svelte-kit"}


def flush():
    try:
        with open(RAPPORT, "w") as f:
            f.write("\n".join(report) + "\n")
    except OSError:
        pass


def say(line=""):
    report.append(line)
    flush()


def marche(*a):
    print(*a)
    try:
        sys.stdout.flush()
    except Exception:
        pass


def lire(p):
    try:
        with open(p, "r", errors="ignore") as f:
            return f.read()
    except OSError:
        return None


def fichiers():
    for dirpath, dirnames, filenames in os.walk(BASE):
        dirnames[:] = [d for d in dirnames if d not in IGNORE]
        for fn in sorted(filenames):
            if fn.endswith((".svelte", ".ts", ".scss", ".css")) and not fn.endswith(".d.ts"):
                yield os.path.join(dirpath, fn)


def dump(p, max_lignes=300):
    txt = lire(p)
    say("")
    say("--------------------------------------------------------------")
    if txt is None:
        say(">>> INTROUVABLE : %s" % p)
        return
    say(">>> %s" % p)
    say("--------------------------------------------------------------")
    lignes = txt.split("\n")
    for i, l in enumerate(lignes[:max_lignes], 1):
        say("%4d| %s" % (i, l))
    if len(lignes) > max_lignes:
        say("... (%d lignes de plus, tronque)" % (len(lignes) - max_lignes))


def main():
    if not os.path.isdir(BASE):
        say(">>> DOSSIER INTROUVABLE : %s" % BASE)
        return False

    # ---------- 1. inventaire ----------
    marche("1/4 inventaire...")
    say("===== 1. INVENTAIRE DE %s =====" % BASE)
    tous = list(fichiers())
    for p in tous:
        txt = lire(p) or ""
        say("  %-70s %5d lignes" % (p, len(txt.split("\n"))))

    # ---------- 2. la confirmation d'achat et ce qu'elle utilise ----------
    marche("2/4 modales de confirmation...")
    say("")
    say("===== 2. CONFIRMATION D'ACHAT ET BOUTONS PARTAGES =====")
    cibles = [p for p in tous if re.search(
        r"(BuyBonusConfirm|ModalError|AutoSpinMessage|Button|Modal\.svelte|ModalContainer)",
        os.path.basename(p))]
    for p in cibles:
        dump(p)

    # ---------- 3. ou sont les rem et les tailles ----------
    marche("3/4 unites rem et tailles...")
    say("")
    say("===== 3. TOUTES LES TAILLES EN rem / em / px =====")
    say("(c'est ici que se joue la zone cliquable)")
    say("")
    motif = re.compile(r"(width|height|padding|margin|font-size|min-width|min-height|gap)\s*:\s*[^;]*"
                       r"(rem|em|px|%)")
    for p in tous:
        txt = lire(p) or ""
        for i, l in enumerate(txt.split("\n"), 1):
            if motif.search(l):
                say("%s:%d: %s" % (p, i, l.strip()[:150]))

    # ---------- 4. la regle qui divise la police par deux ----------
    marche("4/4 regle des 500 px...")
    say("")
    say("===== 4. LA REGLE QUI DIVISE LA POLICE RACINE =====")
    trouve = 0
    for base in ("packages", "apps/louvo"):
        if not os.path.isdir(base):
            continue
        for dirpath, dirnames, filenames in os.walk(base):
            dirnames[:] = [d for d in dirnames if d not in IGNORE]
            for fn in filenames:
                if not fn.endswith((".svelte", ".scss", ".css", ".html")):
                    continue
                p = os.path.join(dirpath, fn)
                txt = lire(p) or ""
                for i, l in enumerate(txt.split("\n"), 1):
                    if re.search(r"max-width:\s*500px|font-size:\s*50%|:global\(html\)", l):
                        say("%s:%d: %s" % (p, i, l.strip()[:150]))
                        trouve += 1
    say("(%d lignes trouvees)" % trouve)
    return True


try:
    ok = main()
except Exception:
    say("")
    say("===== ERREUR PYTHON =====")
    for l in traceback.format_exc().rstrip().split("\n"):
        say(l)
    ok = False

say("")
say("VERDICT : %s (aucun fichier modifie)" % ("OK" if ok else "ARRET"))
print("")
print("FINI - envoyez rapport75.txt")
sys.exit(0 if ok else 1)
