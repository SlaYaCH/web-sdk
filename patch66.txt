#!/usr/bin/env python3
# ============================================================
# Louvo - lot 69 : zone cliquable des boutons OK doublee
#
# CE QUI DIMENSIONNE LE BOUTON
#   <BaseIcon width="100%" height="3rem" />
# C'est cette hauteur qui donne la taille du bouton, donc sa zone
# cliquable. Avec la regle de Modals.svelte :
#   :global(html) { font-size: 16px;
#     @media screen and (max-width: 500px) { font-size: 50%; } }
# 3rem vaut 48 px sur grand ecran mais seulement 24 px sous 500 px de
# large. D'ou l'impression que le probleme est apparu avec le mobile.
#
# ON DOUBLE : 3rem -> 6rem, dans tous les modes.
#
# UNIQUEMENT LES BOUTONS OK : la modification ne vise que les boutons
# dont le libelle est i18nDerived.confirm(). Les autres boutons
# (autoplay, cartes de bonus, +/- de mise) ne sont pas touches.
#
# ATTENTION, A SAVOIR : le bouton sera aussi deux fois plus HAUT a
# l'oeil, pas seulement au toucher. Si vous vouliez garder exactement
# le meme visuel et n'agrandir que la zone sensible, c'est faisable
# autrement - le rapport contient deja le fichier Button.svelte qu'il
# me faudrait pour ca, dites-le moi et je le fais.
#
# Fichiers vises (dans packages/, donc dans le SDK, pas dans le jeu) :
#   components-ui-html/src/components/ModalBuyBonusConfirm.svelte
#   components-ui-html/src/components/ModalBetMenu.svelte
# Sauvegardes en .avant-lot69.
#
# Sort en erreur si ca echoue, pour bloquer le build. AUCUN BUILD ICI.
#
# Usage :
#   cd /workspaces/web-sdk && python3 patch66.txt
# ============================================================
import os
import re
import shutil
import sys
import traceback

ROOT = os.path.dirname(os.path.abspath(__file__))
os.chdir(ROOT)

RAPPORT = "rapport76.txt"
report = []


def flush():
    try:
        with open(RAPPORT, "w") as f:
            f.write("\n".join(report) + "\n")
    except OSError:
        pass


def say(line=""):
    report.append(line)
    print(line)
    try:
        sys.stdout.flush()
    except Exception:
        pass
    flush()


BASE = "packages/components-ui-html/src/components"
# (nom du fichier, obligatoire ?) - la confirmation d'achat est celle que
# Seb a signalee, le menu de mise porte le meme bouton OK. Si ce dernier a
# une forme differente, on le signale sans bloquer le build.
CIBLES = [("ModalBuyBonusConfirm.svelte", True), ("ModalBetMenu.svelte", False)]

ANCIENNE = "3rem"
NOUVELLE = "6rem"

# Un BaseIcon dont le bouton porte le libelle confirm(). Le .{0,300}? non
# gourmand garantit qu'on ne saute pas d'un bouton a un autre.
MOTIF = re.compile(
    r'(<BaseIcon\b[^>]*height=")' + ANCIENNE + r'("[^>]*/>)((?:.|\n){0,300}?i18nDerived\.confirm\(\))'
)
DEJA = re.compile(
    r'<BaseIcon\b[^>]*height="' + NOUVELLE + r'"[^>]*/>(?:.|\n){0,300}?i18nDerived\.confirm\(\)'
)


def main():
    tout_ok = True
    touches = 0

    say("===== 1. BOUTONS OK =====")
    for nom, obligatoire in CIBLES:
        p = os.path.join(BASE, nom)
        if not os.path.exists(p):
            say("  %-32s INTROUVABLE%s" % (nom, "" if obligatoire else " (optionnel)"))
            if obligatoire:
                tout_ok = False
            continue
        txt = open(p, encoding="utf-8").read()

        if DEJA.search(txt):
            say("  %-32s deja a %s" % (nom, NOUVELLE))
            touches += 1
            continue

        n = len(MOTIF.findall(txt))
        if n == 0:
            say("  %-32s aucun bouton confirm() trouve - NON MODIFIE%s"
                % (nom, "" if obligatoire else " (optionnel, sans gravite)"))
            if obligatoire:
                tout_ok = False
            continue

        if not os.path.exists(p + ".avant-lot69"):
            shutil.copyfile(p, p + ".avant-lot69")
        txt = MOTIF.sub(lambda m: m.group(1) + NOUVELLE + m.group(2) + m.group(3), txt)
        with open(p, "w", encoding="utf-8") as f:
            f.write(txt)
        say("  %-32s %d bouton(s) : %s -> %s" % (nom, n, ANCIENNE, NOUVELLE))
        touches += n

    if touches == 0:
        tout_ok = False

    # ---------- 2. rien d'autre n'a bouge ----------
    say("")
    say("===== 2. LES AUTRES BOUTONS SONT INTACTS =====")
    say("(tout ce qui reste en 3rem ne doit PAS etre un bouton OK)")
    for fn in sorted(os.listdir(BASE)):
        if not fn.endswith(".svelte"):
            continue
        p = os.path.join(BASE, fn)
        txt = open(p, encoding="utf-8", errors="ignore").read()
        for i, l in enumerate(txt.split("\n"), 1):
            if "BaseIcon" in l and "height=" in l:
                say("  %-34s:%-4d %s" % (fn, i, l.strip()[:90]))

    # ---------- 3. verification ----------
    say("")
    say("===== 3. VERIFICATION =====")
    for nom, obligatoire in CIBLES:
        p = os.path.join(BASE, nom)
        if not os.path.exists(p):
            continue
        txt = open(p, encoding="utf-8").read()
        ok = bool(DEJA.search(txt))
        if not ok and obligatoire:
            tout_ok = False
        say("  %-32s bouton OK a %s : %s" % (
            nom, NOUVELLE,
            "OK" if ok else ("*** ECHEC ***" if obligatoire else "non modifie (optionnel)")))

    say("")
    say("===== 4. TAILLE DE LA ZONE CLIQUABLE =====")
    say("  %-26s %14s %14s" % ("largeur d'ecran", "avant", "apres"))
    say("  %-26s %14s %14s" % ("plus de 500 px", "48 px", "96 px"))
    say("  %-26s %14s %14s" % ("500 px ou moins", "24 px", "48 px"))
    say("")
    say("  La largeur ne change pas : elle est plafonnee a 250 px par")
    say("  BaseButtonWrap (type max-width), partage avec d'autres modales.")

    # ---------- 5. de quoi enchainer si le visuel ne va pas ----------
    say("")
    say("===== 5. LECTURE : components-shared Button.svelte =====")
    say("(pour pouvoir, si vous le voulez, agrandir la zone sensible SANS")
    say(" changer le visuel du bouton)")
    for chemin in ("packages/components-shared/src/components/Button.svelte",
                   "packages/components-ui-html/src/components/BaseIcon.svelte"):
        say("")
        say("--------------------------------------------------------------")
        try:
            src = open(chemin, encoding="utf-8", errors="ignore").read()
        except OSError:
            say(">>> INTROUVABLE : %s" % chemin)
            continue
        say(">>> %s" % chemin)
        say("--------------------------------------------------------------")
        for i, l in enumerate(src.split("\n")[:200], 1):
            say("%4d| %s" % (i, l))
    return tout_ok


try:
    ok = main()
except Exception:
    say("")
    say("===== ERREUR PYTHON =====")
    for l in traceback.format_exc().rstrip().split("\n"):
        say(l)
    ok = False

say("")
say("VERDICT : %s" % ("OK" if ok else "ECHEC"))
print("")
print("FINI - envoyez rapport76.txt")
sys.exit(0 if ok else 1)
