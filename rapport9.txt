===== 1. Anticipation.svelte (le cadre jaune a decaler) =====
     1	<script lang="ts">
     2		import { SpineProvider, SpineTrack } from 'pixi-svelte';
     3		import { stateBetDerived } from 'state-shared';
     4	
     5		import { getContext } from '../game/context';
     6		import type { Reel } from '../game/stateGame.svelte';
     7		import { REEL_PADDING, SYMBOL_SIZE } from '../game/constants';
     8	
     9		type Props = {
    10			reel: Reel;
    11			oncomplete: () => void;
    12		};
    13	
    14		const props: Props = $props();
    15		const context = getContext();
    16	
    17		type AnimationName = 'anticipation_intro' | 'anticipation_loop' | 'anticipation_out';
    18	
    19		let animationName = $state<AnimationName>('anticipation_intro');
    20	
    21		$effect(() => {
    22			if (props.reel.reelState.motion === 'stopped') {
    23				animationName = 'anticipation_out';
    24			}
    25		});
    26	</script>
    27	
    28	<SpineProvider
    29		key="anticipation"
    30		width={SYMBOL_SIZE * 1.0}
    31		height={SYMBOL_SIZE * 5}
    32		x={context.stateGameDerived.boardLayout().x -
    33			context.stateGameDerived.boardLayout().width * 0.5 +
    34			(props.reel.reelIndex + REEL_PADDING) * SYMBOL_SIZE}
    35		y={context.stateGameDerived.boardLayout().y - SYMBOL_SIZE * 0.06}
    36	>
    37		<SpineTrack
    38			trackIndex={0}
    39			{animationName}
    40			loop={animationName === 'anticipation_loop'}
    41			timeScale={stateBetDerived.timeScale()}
    42			listener={{
    43				complete: () => {
    44					if (animationName === 'anticipation_intro') {
    45						animationName = 'anticipation_loop';
    46					}
    47	
    48					if (animationName === 'anticipation_out') {
    49						props.oncomplete();
    50					}
    51				},
    52			}}
    53		/>
    54	</SpineProvider>
