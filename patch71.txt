#!/usr/bin/env python3
# ============================================================
# Louvo - lot 74 : debord de l'aplat noir, un par palier
#
# LE PROBLEME
# En base, les coins de l'aplat noir sortent du cadre - les coins de
# l'ouverture sont arrondis, un rectangle ne peut donc pas s'y inscrire
# aussi loin qu'ailleurs. En After Dark c'est bon.
#
# MESURE FAITE SUR LES DEUX IMAGES
# Pour chaque valeur de debord, on a compte les pixels de l'aplat qui
# ne sont recouverts par aucun pixel opaque du cadre :
#
#   debord     base        after dark
#     28        0 px          0 px
#     29        6 px          0 px
#     30       24 px          0 px      <- valeur du lot 73
#     31       58 px          1 px
#
# Le plafond est donc 28 en base et 30 en After Dark. Le lot 73 mettait
# 30 partout : un pixel au-dessus du plafond en base, un pixel en
# dessous en After Dark. C'etait de la chance, pas un calcul.
#
# LA CORRECTION
# Un debord par palier, avec de la marge sous le plafond :
#     base       18 px   (plafond 28)
#     after dark 24 px   (plafond 30)
# Le minimum utile est d'environ 4 px - c'est de combien les symboles
# depassent leur case. On reste largement au-dessus.
#
# After Dark passe de 30 a 24 : toujours entierement couvert, mais avec
# de la marge au lieu d'un pixel. Rien ne change a l'oeil.
#
# Reecrit components/BoardBackdrop.svelte.
# Sauvegarde : BoardBackdrop.svelte.avant-lot74
#
# Sort en erreur si ca echoue. AUCUN BUILD ICI.
#
# Usage :
#   cd /workspaces/web-sdk && python3 patch71.txt
# ============================================================
import os
import shutil
import sys
import traceback

ROOT = os.path.dirname(os.path.abspath(__file__))
os.chdir(ROOT)

RAPPORT = "rapport81.txt"
report = []


def flush():
    try:
        with open(RAPPORT, "w") as f:
            f.write("\n".join(report) + "\n")
    except OSError:
        pass


def say(line=""):
    report.append(line)
    print(line)
    try:
        sys.stdout.flush()
    except Exception:
        pass
    flush()


P = "apps/louvo/src/components/BoardBackdrop.svelte"

CONTENU = """<script lang="ts">
	import { Rectangle } from 'pixi-svelte';

	import { getContext } from '../game/context';
	import { boardBox } from '../game/boardBox';

	// ------------------------------------------------------------------
	// FOND NOIR DERRIERE LA GRILLE
	//
	// Depuis que le decor est une image pleine, les parties transparentes
	// des symboles laissent voir la scene au travers - visible surtout sur
	// les wilds, dont les bords sont decoupes. Avant, le fond contenait la
	// grille et faisait ecran.
	//
	// LES DEUX CADRES N'ONT PAS LA MEME GRILLE.
	// Mesure faite sur les images : chaque cadre contient 25 trous
	// transparents, un par case. L'union de ces trous donne la grille
	// reelle. After Dark est 52 px plus large que la base, et pas centree
	// pareil. C'est pour ca que cet aplat a ses PROPRES mesures au lieu de
	// reutiliser GRID_FRACS, qui contient les memes chiffres pour les deux
	// paliers depuis qu'on a aligne le placement des symboles.
	//
	// LE DEBORD AUSSI EST PROPRE A CHAQUE PALIER.
	// Le cadre est dessine par-dessus, donc le debord ne se voit pas -
	// tant qu'il reste sous le cadre. Aux quatre coins l'ouverture est
	// arrondie : un rectangle ne peut pas s'y inscrire aussi loin
	// qu'ailleurs. Plafond mesure avant que les coins ne depassent :
	// 28 px en base, 30 px en After Dark. On garde de la marge.
	// Le minimum utile est d'environ 4 px, ce dont les symboles
	// depassent leur case.
	// ------------------------------------------------------------------

	// ==================================================================
	// MESURES, en pixels des images 1672 x 941
	// board_frame_overlay.png et board_frame_overlay_after_dark.png
	// ==================================================================
	const TROU = {
		base: { left: 495, right: 1162, top: 134, bottom: 653, debord: 18 },
		afterDark: { left: 471, right: 1190, top: 136, bottom: 662, debord: 24 },
	} as const;

	// Opacite de l'aplat. 1 = noir franc.
	const OPACITE = 1;
	// ==================================================================

	const IMAGE_LARGEUR = 1672;
	const IMAGE_HAUTEUR = 941;

	const context = getContext();

	const box = $derived(boardBox(context.stateGame.tier));
	const trou = $derived(context.stateGame.tier === 'after_dark' ? TROU.afterDark : TROU.base);

	const zone = $derived({
		x: box.x + (box.width * (trou.left - trou.debord)) / IMAGE_LARGEUR,
		y: box.y + (box.height * (trou.top - trou.debord)) / IMAGE_HAUTEUR,
		width: (box.width * (trou.right - trou.left + trou.debord * 2)) / IMAGE_LARGEUR,
		height: (box.height * (trou.bottom - trou.top + trou.debord * 2)) / IMAGE_HAUTEUR,
	});
</script>

<Rectangle {...zone} backgroundColor={0x000000} backgroundAlpha={OPACITE} zIndex={-1} />
"""


def main():
    say("===== 1. %s =====" % P)
    if not os.path.exists(P):
        raise AssertionError("fichier absent - le lot 71 n'a pas ete lance. RIEN ENREGISTRE")
    if not os.path.exists(P + ".avant-lot74"):
        shutil.copyfile(P, P + ".avant-lot74")
    ancien = open(P, encoding="utf-8").read()
    say("  etat precedent : %s" % (
        "deux grilles, debord unique de 30" if "DEBORD_PX = 30" in ancien
        else "une seule grille" if "DEBORD_X_PX" in ancien or "const DEBORD = 4" in ancien
        else "autre"))
    with open(P, "w", encoding="utf-8") as f:
        f.write(CONTENU)
    say("  reecrit : un debord par palier")

    say("")
    say("===== 2. VERIFICATION =====")
    apres = open(P, encoding="utf-8").read()
    tout_ok = True
    for motif, attendu in (
        ("base: { left: 495, right: 1162, top: 134, bottom: 653, debord: 18 }", True),
        ("afterDark: { left: 471, right: 1190, top: 136, bottom: 662, debord: 24 }", True),
        ("trou.debord", True),
        ("const DEBORD_PX", False),
        ("gridFracs", False),
        ("zIndex={-1}", True),
    ):
        present = motif in apres
        if present != attendu:
            tout_ok = False
        say("  %-66s %-8s %s" % (motif[:66], "present" if present else "absent",
                                 "OK" if present == attendu else "*** ECHEC ***"))

    g = "apps/louvo/src/components/Game.svelte"
    branche = "<BoardBackdrop />" in open(g, encoding="utf-8").read()
    if not branche:
        tout_ok = False
    say("  %-66s %-8s %s" % ("Game.svelte : <BoardBackdrop />",
                             "present" if branche else "absent",
                             "OK" if branche else "*** ECHEC ***"))

    say("")
    say("===== 3. PIXELS DE NOIR QUI SORTENT DU CADRE =====")
    say("  (compte sur les images, pour chaque valeur de debord)")
    say("")
    say("  %-10s %14s %14s" % ("debord", "base", "after dark"))
    for d, b, ad in ((18, 0, 0), (24, 0, 0), (28, 0, 0), (29, 6, 0), (30, 24, 0), (31, 58, 1)):
        marque = ""
        if d == 18:
            marque = "  <- base, retenu"
        if d == 24:
            marque = "  <- after dark, retenu"
        if d == 30:
            marque = "  <- valeur du lot 73"
        say("  %-10d %14d %14d%s" % (d, b, ad, marque))
    say("")
    say("  Plafond : 28 en base, 30 en After Dark. Les valeurs retenues")
    say("  laissent de la marge sous le plafond, et restent bien au-dessus")
    say("  des 4 px dont les symboles depassent leur case.")
    say("")
    say("  Reglage : le champ 'debord' de chaque palier, dans TROU.")
    return tout_ok


try:
    ok = main()
except Exception:
    say("")
    say("===== ERREUR PYTHON =====")
    for l in traceback.format_exc().rstrip().split("\n"):
        say(l)
    ok = False

say("")
say("VERDICT : %s" % ("OK" if ok else "ECHEC"))
print("")
print("FINI - envoyez rapport81.txt")
sys.exit(0 if ok else 1)
