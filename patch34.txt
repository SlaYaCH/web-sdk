#!/usr/bin/env python3
# ============================================================
# Louvo - lot 34 : mise a l'ecart des assets de la demo miniere
#                 + audit du dossier audio (lecture seule)
#
# Les dossiers de spines/sprites qui ne sont JAMAIS demandes par le
# bundle deploye sont DEPLACES (pas supprimes) vers
# _demo_assets_backup/ a la racine du depot, hors du build.
#
# VERROUS EN DUR - ces dossiers ne seront JAMAIS touches :
#   - assets/fonts/**      (les 4 polices sont chargees dynamiquement,
#                           l'audit ne peut pas les voir : trop risque)
#   - assets/audio/**      (sons references dans sounds.json)
#   - assets/spines/anticipation/**  (les rochers, demande du joueur)
#
# La liste n'est pas ecrite en dur : elle est RECALCULEE depuis le
# bundle a chaque execution. Rien n'est deplace sans preuve.
#
# Usage :
#   1) deposer ce fichier a la RACINE de web-sdk
#   2) cd /workspaces/web-sdk && python3 patch34.txt
#   3) envoyer rapport40.txt
# Build coupe automatiquement (5 min max), pas de Ctrl+C.
# ============================================================
import os
import re
import shutil
import subprocess
import sys

ROOT = os.path.dirname(os.path.abspath(__file__))
os.chdir(ROOT)

BUILD = "apps/louvo/build"
STATIC = "apps/louvo/static/assets"
BACKUP = "_demo_assets_backup"

PROTEGES = ("fonts", "audio")
PROTEGES_DIRS = ("spines/anticipation",)

report = []


def human(n):
    if n < 1024:
        return "%d o" % n
    if n < 1024 * 1024:
        return "%.0f Ko" % (n / 1024.0)
    return "%.2f Mo" % (n / 1024.0 / 1024.0)


def dir_size(path):
    total = 0
    for dirpath, _d, filenames in os.walk(path):
        for fn in filenames:
            try:
                total += os.path.getsize(os.path.join(dirpath, fn))
            except OSError:
                pass
    return total


# ---------- 1. Ce que le bundle deploye demande vraiment ----------
if not os.path.isdir(BUILD):
    print("ERREUR : dossier build introuvable - lancez d'abord un build")
    sys.exit(1)

referenced = set()
for dirpath, _d, filenames in os.walk(BUILD):
    if os.path.sep + "assets" + os.path.sep in dirpath + os.path.sep:
        continue
    for fn in filenames:
        if not fn.endswith((".js", ".html", ".json", ".css")):
            continue
        try:
            with open(os.path.join(dirpath, fn), "r", errors="ignore") as f:
                content = f.read()
        except OSError:
            continue
        for h in re.findall(r"assets/[A-Za-z0-9_./-]+", content):
            referenced.add(os.path.splitext(h)[0])

report.append("===== 1. REFERENCES TROUVEES DANS LE BUNDLE =====")
report.append("chemins d'assets distincts demandes par le jeu : %d" % len(referenced))
if len(referenced) < 30:
    report.append("ARRET DE SECURITE : trop peu de references, le bundle semble incomplet.")
    with open("rapport40.txt", "w") as f:
        f.write("\n".join(report) + "\n")
    print("\n".join(report))
    sys.exit(1)


def est_utilise(path):
    rel = os.path.relpath(path, os.path.dirname(STATIC)).replace(os.sep, "/")
    return os.path.splitext(rel)[0] in referenced


# ---------- 2. Dossiers jamais demandes (spines/ et sprites/ seulement) ----------
candidats = []
for top in sorted(os.listdir(STATIC)):
    if top in PROTEGES:
        continue
    top_path = os.path.join(STATIC, top)
    if not os.path.isdir(top_path):
        continue
    for sub in sorted(os.listdir(top_path)):
        sub_path = os.path.join(top_path, sub)
        if not os.path.isdir(sub_path):
            continue
        label = "%s/%s" % (top, sub)
        if label in PROTEGES_DIRS:
            continue
        nb = 0
        used = 0
        for dirpath, _d, filenames in os.walk(sub_path):
            for fn in filenames:
                nb += 1
                if est_utilise(os.path.join(dirpath, fn)):
                    used += 1
        if nb > 0 and used == 0:
            candidats.append((label, sub_path, dir_size(sub_path)))

report.append("")
report.append("===== 2. DOSSIERS MIS A L'ECART =====")
total = 0
for label, path, size in sorted(candidats, key=lambda x: -x[2]):
    dest = os.path.join(BACKUP, "assets", label)
    os.makedirs(os.path.dirname(dest), exist_ok=True)
    if os.path.exists(dest):
        shutil.rmtree(dest)
    shutil.move(path, dest)
    total += size
    report.append("%-46s %10s  -> %s" % (label, human(size), dest))
if not candidats:
    report.append("(aucun - deja fait)")
report.append("")
report.append("TOTAL RETIRE DU BUILD : %s" % human(total))

# ---------- 3. index.ts perdus dans static ----------
report.append("")
report.append("===== 3. index.ts (code source jamais execute) =====")
moved_ts = 0
for dirpath, _d, filenames in os.walk(STATIC):
    rel_dir = os.path.relpath(dirpath, STATIC).replace(os.sep, "/")
    if rel_dir.split("/")[0] in PROTEGES or rel_dir in PROTEGES_DIRS:
        continue
    for fn in filenames:
        if fn != "index.ts":
            continue
        src = os.path.join(dirpath, fn)
        dest = os.path.join(BACKUP, "assets", rel_dir, fn)
        os.makedirs(os.path.dirname(dest), exist_ok=True)
        shutil.move(src, dest)
        moved_ts += 1
report.append("deplaces : %d (ceux de fonts/ et anticipation/ sont laisses en place)" % moved_ts)

# ---------- 4. Fichiers non demandes DANS des dossiers utilises (rapport seul) ----------
report.append("")
report.append("===== 4. FICHIERS NON DEMANDES restant dans des dossiers utilises =====")
report.append("(rien n'est touche ici - a decider ensemble)")
restants = []
for dirpath, _d, filenames in os.walk(STATIC):
    rel_dir = os.path.relpath(dirpath, STATIC).replace(os.sep, "/")
    if rel_dir.split("/")[0] in PROTEGES:
        continue
    for fn in filenames:
        fp = os.path.join(dirpath, fn)
        if not est_utilise(fp):
            try:
                restants.append((rel_dir + "/" + fn, os.path.getsize(fp)))
            except OSError:
                pass
for name, size in sorted(restants, key=lambda x: -x[1])[:20]:
    report.append("%-58s %10s" % (name, human(size)))
if not restants:
    report.append("(aucun)")

# ---------- 5. AUDIT AUDIO (lecture seule) ----------
report.append("")
report.append("===== 5. AUDIT DU DOSSIER AUDIO (aucune suppression) =====")
audio_dir = os.path.join(STATIC, "audio")
sounds_json = os.path.join(audio_dir, "sounds.json")
if os.path.exists(sounds_json):
    with open(sounds_json, "r", errors="ignore") as f:
        sj = f.read()
    audio_refs = set(re.findall(r"[A-Za-z0-9_./-]+\.(?:mp3|ogg|m4a|webm|wav|aac|flac)", sj))
    report.append("fichiers audio cites dans sounds.json : %s"
                  % (", ".join(sorted(os.path.basename(a) for a in audio_refs)) or "(aucun)"))
    report.append("nombre d'entrees 'sprite' declarees    : %d" % len(re.findall(r'"[^"]+"\s*:\s*\[\s*\d', sj)))
    report.append("")
    report.append("%-46s %10s   %s" % ("fichier present", "poids", "cite dans sounds.json ?"))
    total_audio = 0
    total_orphelin = 0
    for fn in sorted(os.listdir(audio_dir)):
        fp = os.path.join(audio_dir, fn)
        if not os.path.isfile(fp):
            continue
        size = os.path.getsize(fp)
        total_audio += size
        cite = fn == "sounds.json" or any(os.path.basename(a) == fn for a in audio_refs)
        if not cite:
            total_orphelin += size
        report.append("%-46s %10s   %s" % (fn, human(size), "oui" if cite else "*** NON ***"))
    report.append("")
    report.append("poids total audio        : %s" % human(total_audio))
    report.append("poids des non-cites      : %s" % human(total_orphelin))
else:
    report.append("sounds.json introuvable - audit impossible")

# ---------- 6. .gitignore ----------
try:
    gi = ""
    if os.path.exists(".gitignore"):
        with open(".gitignore") as f:
            gi = f.read()
    if BACKUP not in gi:
        with open(".gitignore", "a") as f:
            f.write("\n# Assets de la demo d'origine, mis a l'ecart (recuperables ici ou via git)\n%s/\n" % BACKUP)
        report.append("")
        report.append(".gitignore : %s/ ajoute" % BACKUP)
except OSError as e:
    report.append("gitignore : %s" % e)

# ---------- 7. git status + build ----------
report.append("")
report.append("===== 6. git status (extrait) =====")
try:
    out = subprocess.run(["git", "status", "--short"], capture_output=True, text=True).stdout
    lignes = out.rstrip().split("\n")
    report.append("%d entrees modifiees/supprimees" % len(lignes))
    report.extend(lignes[:12])
    if len(lignes) > 12:
        report.append("... (%d de plus)" % (len(lignes) - 12))
except OSError as e:
    report.append("git indisponible : %s" % e)

report.append("")
report.append("===== 7. BUILD =====")
timed_out = False
with open("build_log.txt", "w") as logf:
    try:
        code = subprocess.run(["pnpm", "run", "build", "--filter=louvo"],
                              stdout=logf, stderr=subprocess.STDOUT, timeout=300).returncode
    except subprocess.TimeoutExpired:
        timed_out = True
        code = -1
with open("build_log.txt") as f:
    log = f.read()
report.extend(log.rstrip().split("\n")[-6:])
site_ecrit = ("Wrote site to" in log) or ("done" in log)
if timed_out:
    report.append("(build coupe apres 5 min - comportement pnpm connu)")
report.append("site ecrit : %s / exit : %s" % ("OUI" if site_ecrit else "NON", code))

# poids final du build
if os.path.isdir(os.path.join(BUILD, "assets")):
    report.append("poids de build/assets apres nettoyage : %s" % human(dir_size(os.path.join(BUILD, "assets"))))

report.append("")
report.append("VERDICT : %s" % ("TOUT OK - testez le jeu puis commit"
                                if site_ecrit else "ECHEC - envoyez ce rapport a Claude"))

with open("rapport40.txt", "w") as f:
    f.write("\n".join(report) + "\n")
print("\n".join(report[-8:]))
print("")
print("FINI - envoyez rapport40.txt")
sys.exit(0 if site_ecrit else 1)
